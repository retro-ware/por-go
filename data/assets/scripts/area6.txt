OfferDisguiseHelp:
	PrintClear "THE WOMAN SAYS 'TYRANTHRAXUS IS LEADING OUR MEN TO THEIR DEATHS. IF YOU INTEND TO KILL HIM WE WILL HELP. USE THESE CLOTHES AS A DISGUISE AND YOU WILL ESCAPE NOTICE."
	Gosub PressToContinue

AskUseDisguise:
	PrintClear "WILL YOU USE THE CLOTHING AS A DISGUISE?"
	Gosub YesNoMenu
	IfEq
	Or CastleFlags 1 CastleFlags
	And CastleFlags 1 Scratch10
	Return

CheckPatrolSurprise:
	PartySurprise Scratch8 Scratch9
	And CastleFlags 1 Scratch10
	IfNe
	Add 1 Scratch9 Scratch9
	Surprise Scratch8 Scratch9 0 0
	if AlarmEndHour != 0: And Surprise 1 Surprise
	if Surprise == 3: Save 0 Surprise
	Compare Surprise 1
	Return

CheckAlarmExpired:
	if AlarmEndHour < 1: Return
	Save Hours AuxRoll
	if Hours == 0 && AlarmHour == 23: Save 24 AuxRoll
	if AuxRoll < AlarmEndHour: Goto AlarmStillRinging
	if TenMinutes < AlarmMinute: Goto AlarmStillRinging
	Save 0 AlarmEndHour
	Compare 0 1
	Return

AlarmStillRinging:
	if Search == 1: Goto PrintAlarmSounding
	Random 9 AuxRoll
	if 0 != AuxRoll: Goto AlarmReturnActive

PrintAlarmSounding:
	PrintClear "AN ALARM IS SOUNDING."
	Delay

AlarmReturnActive:
	Compare 1 0
	Return

StartAlarm:
	if AlarmEndHour == 0: PrintClear "AN ALARM STARTS RINGING!"
	Delay
	Save Hours AlarmHour
	Save TenMinutes AlarmMinute
	Add 1 AlarmHour AlarmEndHour
	Return

AskTakeIt:
	Print "DO YOU TAKE IT?"

YesNoMenu:
	HMenu Scratch10 2 "YES" "NO"
	PrintClear " "
	Compare Scratch10 0
	Return

PressToContinue:
	HMenu Scratch9 1 "PRESS BUTTON OR <RETURN> TO CONTINUE"
	PrintClear " "
	Return

SetSkyByTime:

	Save 9 SkyInside
	if Hours >= 14: Save 8 Scratch1

	IfLt
	Goto DispatchCell

	IfNe

	Exit

event3:

	LoadCharacter Scratch1

	GetTable AgonyByRace SelAlignment Scratch10
	OnGoto Scratch10 1 BlessNextMember
	PrintClear $SelName
	Print " DOUBLES UP IN AGONY."
	Delay
	Or BaneTempleFlags 4 BaneTempleFlags
	Damage 160 2 10 2 128

	if Scratch1 < 8: Goto BlessMemberLoop
	And BaneTempleFlags 4 Scratch10
	IfEq
	Goto Done
