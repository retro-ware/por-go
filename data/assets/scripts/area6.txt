OfferDisguiseHelp:
	PrintClear "THE WOMAN SAYS 'TYRANTHRAXUS IS LEADING OUR MEN TO THEIR DEATHS. IF YOU INTEND TO KILL HIM WE WILL HELP. USE THESE CLOTHES AS A DISGUISE AND YOU WILL ESCAPE NOTICE."

AskUseDisguise:
	PrintClear "WILL YOU USE THE CLOTHING AS A DISGUISE?"
	Gosub YesNoMenu
	IfEq
	CastleFlags = CastleFlags | 1
	Scratch10 = CastleFlags & 1
	Return

CheckPatrolSurprise:
	PartySurprise Scratch8 Scratch9
	Scratch10 = CastleFlags & 1
	if 0 != Scratch10: Scratch9 += 1
	Surprise Scratch8 Scratch9 0 0
	if AlarmEndHour != 0: Surprise = Surprise & 1
	if Surprise == 3: Surprise = 0
	Compare Surprise 1
	Return

CheckAlarmExpired:
	if AlarmEndHour < 1: Return
	AuxRoll = Hours
	if Hours == 0 && AlarmHour == 23: AuxRoll = 24
	if AuxRoll < AlarmEndHour: Goto AlarmStillRinging
	if TenMinutes < AlarmMinute: Goto AlarmStillRinging
	AlarmEndHour = 0
	Compare 0 1
	Return

AlarmStillRinging:
	if Search == 1: Goto PrintAlarmSounding
	AuxRoll = Random 9
	if 0 != AuxRoll: Goto AlarmReturnActive

PrintAlarmSounding:
	PrintClear "AN ALARM IS SOUNDING."
	Delay

AlarmReturnActive:
	Compare 1 0
	Return

StartAlarm:
	if AlarmEndHour == 0: PrintClear "AN ALARM STARTS RINGING!"
	Delay
	AlarmHour = Hours
	AlarmMinute = TenMinutes
	AlarmEndHour = 1 + AlarmHour
	Return
