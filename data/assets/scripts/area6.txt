def OfferDisguiseHelp() {
    Text("THE WOMAN SAYS 'TYRANTHRAXUS IS LEADING OUR MEN TO THEIR DEATHS. IF YOU INTEND TO KILL HIM WE WILL HELP. USE THESE CLOTHES AS A DISGUISE AND YOU WILL ESCAPE NOTICE.")
    PressToContinue()
    goto AskUseDisguise
}

AskUseDisguise:
    Text("WILL YOU USE THE CLOTHING AS A DISGUISE?")
    if YesNoMenu() == "YES" { CastleFlags |= Disguised }
    Scratch10 = CastleFlags & Disguised
    return Scratch10

def CheckPatrolSurprise() {
    PartySurprise(Scratch8, Scratch9)
    if CastleFlags has Disguised { Scratch9 += 1 }
    Surprise(Scratch8, Scratch9, 0, 0)
    if AlarmEndHour != 0 { Surprise &= PartyIsSurprised }
    if Surprise == BothSurprised { Surprise = NoSurprise }
    return Surprise
}

def CheckAlarmExpired() {
    if AlarmEndHour < 1 { return }
    AuxRoll = Hours
    if Hours == 0 and AlarmHour == 23 { AuxRoll = 24 }
    if AuxRoll < AlarmEndHour { goto AlarmStillRinging }
    if TenMinutes < AlarmMinute { goto AlarmStillRinging }
    AlarmEndHour = 0
    return 0 < 1

    AlarmStillRinging:
    if Search == Searching { goto PrintAlarmSounding }
    AuxRoll = Random(9)
    if 0 != AuxRoll { goto AlarmReturnActive }

    PrintAlarmSounding:
    Text("AN ALARM IS SOUNDING.")
    Delay()

    AlarmReturnActive:
    return 1 < 0
}

def StartAlarm() {
    if AlarmEndHour == 0 { Text("AN ALARM STARTS RINGING!") }
    Delay()
    AlarmHour = Hours
    AlarmMinute = TenMinutes
    AlarmEndHour = 1 + AlarmHour
    return
}

def AskTakeIt() {
    Print("DO YOU TAKE IT?")
    goto YesNoMenu
}

YesNoMenu:
    Scratch10 = Menu("YES", "NO")
    Text(" ")
    return Scratch10

def PressToContinue() {
    Scratch9 = Menu("PRESS BUTTON OR <RETURN> TO CONTINUE")
    Text(" ")
    return
}

def SetSkyByTime() {
    Scratch1 = 11
    SkyInside = 9
    if Hours >= 14 { Scratch1 = 8 }
    Sky = Scratch1
    return
}

on move {
    SetSkyByTime()
    CastleRoomFlags &= KitchenOverseer|Scribes|EmptyScriptorium|CadornaEvidenceFound|TyranthraxusNote|BarbarianReportSeen|BarracksGiantsMet
    if BaneTempleFlags has OfferingBoxStolen|CrossedSwordsStolen {
        CastleFlags &= HallSeen|PatrolTalkedAway|LooseStoneFound|SleepingGiantsFought|SwordInWell|FakeTyranthraxus|GiantsPassed
    }
    if LeftMap == 0 { exit }
    Scratch1 = 3
    if Facing != South { goto StepOutToCity }
    Scratch1 = 9
    SymbolSet = 2
    WallSplitA = 0
    WallSplitB = 0

    StepOutToCity:
    CallStepForward()
    HoldMove = SkipMove
    NewECL(Scratch1)
}

on step {
    if CheckAlarmExpired() { goto DispatchCell }
    if Cell has Special { goto DispatchCell }
    Scratch1 = 2
    if Search == NotSearching { Scratch1 = 20 }
    if Cell has 32 { Scratch1 += Scratch1 }
    Scratch10 = Random(Scratch1)
    if Scratch10 == 0 { goto GiantPatrol }

    DispatchCell:
    Scratch1 = Cell & 31
    goto [Done, LaundryEnter, SmithyEnter, Done, Smokehouse, BaneTempleEnter, Done, CoalStoreroom, AltarEnter][Scratch1]
    goto event2
}

on camp {
    RestEvery = 0
    RestChance = 10
    if AlarmEndHour == 0 { exit }
    if Cell has Special { exit }
    RestEvery = 1
    RestChance = 17
    goto Done
}

Done:
    RedrawView()
    CallRedraw()
    exit

on rest {
    goto GiantPatrol
}

on load {
    SetSkyByTime()
    NoCoords = 1
    if AreaKind == ValjevoCastle6 { exit }
    WallSplitA = 1
    WallSplitB = 1
    LoadMap(ValjevoCastle6)
    LoadWalls(Castle, Castle2, Stone)
    exit
}

LaundryEnter:

    Text("YOU ENTER A BUILDING WHERE WOMEN ARE WASHING CLOTHES. AS YOU COME IN, THEY RETREAT TO THE CORNER LOOKING FEARFULLY AT YOU.")
    PressToContinue()
    Text("YOU HEAR ONE OF THEM WHISPER, 'THEY DON'T BELONG HERE; THEY MUST HAVE COME TO KILL TYRANTHRAXUS'. A HALF-ORC WOMAN STEPS FORWARD 'WHAT DO YOU WANT OF US?'")
    Scratch1 = Parlay(0, 1, 1, 1, 2)
    goto [LaundryOfferDisguise, LaundryOfferDisguise][Scratch1]
    Text("'PLEASE DON'T KILL US,' SHE SAYS. 'IF YOU DON'T WE WILL GIVE YOU THESE CLOTHES TO USE AS A DISGUISE. WE SWEAR WE WON'T SAY ANYTHING!")
    if AskUseDisguise() == 0 {
        goto LaundryPutOnClothes
    } else {
        goto LaundryGoodLuck
    }

    if OfferDisguiseHelp() == 0 { goto LaundryPutOnClothes }

PriestMenu:

    Scratch1 = 0

    SelectMember(Scratch1)
    if SelStatus != Okay { goto BlessNextMember }
    Scratch10 = AgonyByRace[SelAlignment]
    goto [BlessNextMember][Scratch10]
    Text($SelName)
    Print(" DOUBLES UP IN AGONY.")

    BaneTempleFlags |= DoubledInAgony

    Scratch1 = Scratch1 + 1

    if BaneTempleFlags lacks DoubledInAgony { goto Done }

    Text("'I HAVE NO MORE TIME TO TALK WITH YOU, THE LORD BANE REQUIRES MY CONSTANT DEVOTION. COME, BEFORE YOU LEAVE ACCEPT THE BLESSING OF OUR LORD.")

    Text("'HIS IMMENSITY, TYRANTHRAXUS, HAS SEEN FIT TO ESTABLISH THIS TEMPLE TO OUR LORD BANE. THEREFORE IT IS RIGHT THAT YOU ACCEPT THE LORD BANE'S BLESSING.'")
    BaneTempleFlags |= PriestBlessing
    goto PriestMenu

AltarEnter:
    Text("YOU ARE BEFORE THE ALTER. ")
    if BaneTempleFlags has OfferingBoxStolen { goto AltarSwords }
    Print("NEAR IT IS THE OFFERING BOX WITH SOME CHANGE IN IT. ")
    if AskTakeIt() != "YES" { goto AltarBoxLeft }
    BaneTempleFlags |= OfferingBoxStolen
    ClearMonsters()
    Treasure(silver=37)
    Combat()
    Text("THERE IS A FEELING OF SOMETHING IMMENSE AND EVIL STARING DOWN AT YOU...")
    PressToContinue()
    Damage(224, 2, 10, 2, 0)

    AltarBoxLeft:
    Text("")

    AltarSwords:
    if BaneTempleFlags has CrossedSwordsStolen { exit }
    Text("BEHIND THE ALTER ARE TWO CROSSED SWORDS. DO YOU TAKE THEM?")
    if YesNoMenu() != "YES" { exit }
    BaneTempleFlags |= CrossedSwordsStolen
    CastleFlags &= HallSeen|PatrolTalkedAway|LooseStoneFound|SleepingGiantsFought|SwordInWell|FakeTyranthraxus|GiantsPassed
    ClearMonsters()
    Treasure(items=[LongSword3, LongSword3])
    Combat()
    exit

FleeAfterCombat:
    CombatEnd = PartyFled

AfterCombat:
    if CombatEnd == FightOver { return }
    StartAlarm()
    CastleFlags &= HallSeen|PatrolTalkedAway|LooseStoneFound|SleepingGiantsFought|SwordInWell|FakeTyranthraxus|GiantsPassed
    if CombatEnd != PartyFled { return }

BackOut:
    HoldMove = SkipMove
    PartyX = PrevX
    PartyY = PrevY
    Facing = Facing + South
    if Facing > West { Facing -= 4 }
    goto Done

GiantPatrol:
    $AuxEncText = "A PATROL OF GIANTS LED BY A HUMAN FIGHTER"
    Portrait = Bearded
    SetupMonster(Fighter16, distance=2, body=PlateArmor)
    if CheckPatrolSurprise() != PartyIsSurprised { goto PatrolSighted }
    Text($AuxEncText)
    Print(" SURPRISES YOU.")
    if BaneTempleFlags has OfferingBoxStolen|CrossedSwordsStolen { goto PatrolBlasphemers }
    if CastleFlags has Disguised { goto PatrolQuestion }
    Print(" 'THAT'S THEM! GET THEM!'")
    goto PatrolEncounterMenu

    PatrolBlasphemers:
    Print("'BLASPHEMERS! GET THEM!'")
    Portrait = None
    goto PatrolFight

    PatrolSighted:
    Text("YOU SEE A")
    Print($AuxSightText)
    if BaneTempleFlags lacks OfferingBoxStolen|CrossedSwordsStolen { goto PatrolCheckDisguise }
    Print(". THE LEADER YELLS 'SACRILEGIOUS SCUM!'")
    goto PatrolEncounterMenu

    PatrolCheckDisguise:
    if CastleFlags has Disguised { goto PatrolQuestion }
    Print(". THE LEADER SEES YOU. 'THAT'S THEM!'")

    PatrolEncounterMenu:
    PressToContinue()
    EncounterMenu(Fighter16, distance=2, body=PlateArmor, result=Scratch10, combat=Attack, wait=Attack,
        flee=Attack, advance=Mirror, parlay=Attack, near="THEY ADVANCE, WEAPONS READY", escape=12,
        strength=9)
    Portrait = None
    goto [Done, PatrolFight, FleeAfterCombat, PatrolFight][Scratch10]

    PatrolQuestion:
    Surprise = NoSurprise
    Text("YOU SEE A")
    Print($AuxSightText)
    PressToContinue()
    EncounterMenu(Fighter16, distance=2, body=PlateArmor, result=Scratch10, combat=Attack, wait=CloseIn,
        flee=Attack, advance=CloseIn, parlay=Talk, near="'WHAT ARE YOU DOING HERE?'",
        mid="THE LEADER ADVANCES.", escape=12, strength=9)
    Portrait = None
    goto [Done, PatrolFight, FleeAfterCombat][Scratch10]
    Text("WHAT ARE YOU DOING HERE?")
    Scratch10 = Parlay(0, 1, 1, 2, 0)
    goto [PatrolAskAuthority, PatrolIntruders][Scratch10]
    Text("'ALL RIGHT, THEN. WHAT IS TODAY'S PASSWORD?'")
    $AuxInputText = InputString(6)
    Compare("HARASH", $AuxInputText)
    goto PatrolCheckAnswer

PatrolAskAuthority:
    Text("'I'M SORRY, SIR, THERE'S AN ALERT. ON WHOSE AUTHORITY ARE YOU HERE?'")
    NewLine()
    $AuxInputText = InputString(12)
    Compare("TYRANTHRAXUS", $AuxInputText)

PatrolCheckAnswer:
    if (!=) { goto PatrolImposters } # "TYRANTHRAXUS" != $AuxInputText
    Text("WITH A NOD OF THE HEAD YOU ARE LEFT ALONE.")
    goto Done

    PatrolImposters:
    Text("'WHAT? I THINK THESE SCUM ARE IMPOSTERS!'")
    goto PatrolFight

PatrolIntruders:
    Text("'WELL BOYS, I THINK WE HAVE THE INTRUDERS HERE. GET THEM!'")

PatrolFight:
    Delay()
    Morale = 60
    Portrait = None
    Delay()
    RedrawView()
    ClearMonsters()
    LoadMonster(ThLvlFighter75, count=1, icon=ThLvlFighter)
    LoadMonster(HillGiant, count=5, icon=HillGiant)
    Combat()
    AfterCombat()
    exit
