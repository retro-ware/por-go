	PartySurprise Scratch8 Scratch9

CheckAlarmExpired:
	Compare AlarmEndHour 1
	IfLt
	Return
	Save Hours AuxRoll
	CompareAnd Hours 0 AlarmHour 23
	IfEq
	Save 24 AuxRoll
	Compare AuxRoll AlarmEndHour
	IfLt
	Goto AlarmStillRinging
	Compare TenMinutes AlarmMinute
	IfLt
	Goto AlarmStillRinging
	Save 0 AlarmEndHour
	Compare 0 1
	Return

AlarmStillRinging:
	Compare Search 1
	IfEq
	Goto PrintAlarmSounding
	Random 9 AuxRoll
	Compare 0 AuxRoll
	IfNe
	Goto AlarmReturnActive

PrintAlarmSounding:
	Print "AN ALARM IS SOUNDING."
	Delay

AlarmReturnActive:
	Compare 1 0
	Return

StartAlarm:
	Compare AlarmEndHour 0
	IfEq
	Print "AN ALARM STARTS RINGING!"
	Delay
	Save Hours AlarmHour
	Save TenMinutes AlarmMinute
	Arith 1 AlarmHour AlarmEndHour
	Return
