on move {
    Scratch6 = 11
    if Hours >= 14 { Scratch6 = 8 }
    Sky = Scratch6
    SkyInside = 10
    if LeftMap == 0 { exit }
    QuestZhentilOutpost = LeftOutpost
    Portrait = None
    NewECL(Mountains)
}

on step {
    if AlarmLevel != 0 { goto AlarmCheck }
    goto CellDispatch
}

CellDispatch:
    Scratch7 = Cell & ~Inside
    goto [Done, EscortIn, RestrictedGuards, QuartersEnter, ClimbableWall, DwarfAttackCheck][Scratch7]
    goto AlarmCheck

AlarmCheck:
    if OutpostConquered != 0 { goto CellDispatch }
    Scratch1 = Random(14)
    if Scratch1 >= AlarmLevel { goto CellDispatch }
    Scratch1 = 5 * AlarmLevel
    AlarmLevel += 1
    Portrait = NoseguardHelmet
    if Scratch1 > 30 { goto DwarfAndCommandant }
    if Scratch1 == 10 {
        Text("THERE IS GREAT CONFUSION AS THE GUARDS RUN TO THEIR POSTS.")
        goto CellDispatch
    }
    goto GuardsSpotYou

GuardsSpotYou:
    SetupMonster(Fighter16, distance=2, body=LeatherHarness)
    ApproachTwice()
    Text("SOME GUARDS SPOT YOU AND ATTACK.")
    ClearMonsters()
    LoadMonster(Guard, count=Scratch1, icon=Soldier)
    LoadMonster(Corporal, count=AlarmLevel, icon=Soldier)
    Combat()
    goto CellDispatch

EscortIn:
    if EscortedIn != 0 { goto FrontGateCheck }
    Text("YOU ARE BEING ESCORTED INTO ZHENTIL OUTPOST.")
    WaitReturnPressed()
    Scratch6 = 0
    EscortedIn = Escorted
    goto EscortStep

    EscortStep:
    Scratch1 = RouteStepTable[Scratch6]
    Facing = DestFacing[Scratch6]
    WalkSteps()
    Scratch1 = Table994C[Scratch6]
    gosub [TourNothing, TourInnerWall, TourToKeep, CommandantOffice, TourBarracks, TourWatchTowers,
        TourQuarters][Scratch1]
    if Scratch1 == 6 { goto Done }
    Scratch6 += 1
    goto EscortStep

def WalkSteps() {
    Scratch2 = 1
    if Scratch1 == 0 {
        CallRedraw()
        Delay()
        return
    }
    goto WalkStepLoop

    WalkStepLoop:
    if Scratch2 > Scratch1 { return }
    gosub [StepNorth, StepEast, StepSouth, StepWest][Facing]
    CallRedraw()
    Delay()
    Scratch2 += 1
    goto WalkStepLoop
}

def StepNorth() {
    PartyY -= 1
    return
}

def StepEast() {
    PartyX += 1
    return
}

def StepSouth() {
    PartyY += 1
    return
}

def StepWest() {
    PartyX -= 1
    goto TourNothing
}

TourNothing:
    return

def TourInnerWall() {
    Text("FROM HERE YOU SEE A LARGE INNER WALL AROUND THE CENTRAL KEEP.  THIS AREA IS BUSY WITH MEN ENTERING AND LEAVING BARRACKS.")
    WaitReturnPressed()
    return
}

def TourToKeep() {
    Text("YOU ARE LED TOWARD THE CENTRAL KEEP.")
    WaitReturnPressed()
    return
}

def CommandantOffice() {
    Picture(50)
    Text("YOU ARE INSIDE THE COMMANDANT'S OFFICE.  HE GREETS YOU AND YOU HAND HIM THE PAPERS.")
    WaitReturnPressed()
    Text("AFTER SKIMMING THROUGH THE PAPERS THE COMMANDANT SAYS, 'I WELCOME YOU AND HOPE YOU WILL JOIN ME FOR DINNER TONIGHT.'  HE TELLS A GUARD TO GIVE YOU A TOUR OF THE OUTPOST.")
    WaitReturnPressed()
    InvitedToDinner = 1
    return
}

def TourBarracks() {
    Portrait = NoseguardHelmet
    SetupMonster(Fighter16, distance=1, body=LeatherHarness)
    Text("THE GUARD GIVING THE TOUR SAYS 'HERE YOU SEE ONE OF OUR FINE BARRACKS.  THERE ARE SIX OF THESE HERE AND EACH CAN HOLD OVER A 100 MEN.'")
    RedrawView()
    WaitReturnPressed()
    return
}

def TourWatchTowers() {
    SetupMonster(Fighter16, distance=1, body=LeatherHarness)
    Text("'HERE YOU SEE ONE OF OUR FINE WATCH TOWERS WHICH WERE BUILT BY THE FINEST STONE MASONS IN THE LAND.'")
    WaitReturnPressed()
    Text("'FROM HERE YOU CAN ALSO SEE OUR SOLIDLY BUILT OUTSIDE WALL.  THIS WALL HAS SURVIVED MANY ATTACKS, INCLUDING ONE FROM A DRAGON.'")
    WaitReturnPressed()
    Text("THE GUARD SAYS 'THAT CONCLUDES THE TOUR.  I WILL SHOW YOU TO YOUR QUARTERS.'")
    RedrawView()
    WaitReturnPressed()
    return
}

def TourQuarters() {
    SetupMonster(Fighter16, distance=1, body=LeatherHarness)
    Text("THE GUARD SAYS 'HERE ARE YOUR QUARTERS.  DO NOT LEAVE THIS AREA.'  HE THEN LEAVES.")
    RedrawView()
    Portrait = None
    return
}

RestrictedGuards:
    if QuestZhentilOutpost != 0 or AttackedGuards != 0 { goto Done }
    if NightInQuarters == 1 { goto Done }
    Portrait = NoseguardHelmet
    SetupMonster(Fighter16, distance=0, body=LeatherHarness)
    Text("THERE ARE FOUR GUARDS HERE.  THEIR LEADER SAYS, 'YOU ARE RESTRICTED TO YOUR QUARTERS.   RETURN TO THEM NOW.'  WHAT WILL YOU DO?")
    Scratch1 = Menu("RETURN", "ATTACK", "BRIBE")
    goto [BackToQuarters, AttackGuards, AngeredGuards][Scratch1]
    goto BackToQuarters

BackToQuarters:
    Text("YOU GO BACK TO YOUR QUARTERS.")
    PartyX = 5
    PartyY = 0
    Facing = East
    CallRedraw()
    goto Done

AngeredGuards:
    Text("YOU'VE ANGERED THE GUARDS.  THEY DEMAND THAT YOU RETURN TO YOUR QUARTERS.  WHAT WILL YOU DO?")
    Scratch1 = Menu("GO BACK INSIDE.", "ATTACK GUARDS.")
    goto [BackToQuarters, AttackGuards][Scratch1]
    goto AttackGuards

AttackGuards:
    AlarmFightScore = 0
    Text("YOU ATTACK THE GUARDS.  THE GUARDS SOUND AN ALARM AND ATTACK.")
    AttackedGuards = 1
    Morale = 75
    Portrait = NoseguardHelmet
    SetupMonster(Fighter16, distance=2, body=LeatherHarness)
    ApproachTwice()
    ClearMonsters()
    LoadMonster(Guard, count=4, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto FleeStopped }
    goto SurrenderOrDie

    FleeStopped:
    Text("YOU TRY TO FLEE BUT ARE STOPPED BY SOME GUARDS.")
    goto GuardWaveFight

    SurrenderOrDie:
    SetupMonster(Fighter16, distance=2, body=LeatherHarness)
    Text("MORE GUARDS COMMING RUNNING TOWARD YOU.  THEIR LEADER YELLS 'SURRENDER OR DIE!'  WHAT WILL YOU DO?")
    Scratch1 = Menu("SURRENDER", "ATTACK")
    goto [Surrender, GuardWaveFight][Scratch1]
    goto Surrender

Surrender:
    Text("YOU SURRENDER.  THEY STRIP YOU OF ALL POSSESSIONS AND TOSS YOU OUT INTO THE WILDERNESS.'")
    goto StripAndExpel

StripAndExpel:
    Rob(1, 100, 255)
    Portrait = None
    NewECL(Mountains)

GuardWaveFight:
    Portrait = NoseguardHelmet
    SetupMonster(Fighter16, distance=2, body=LeatherHarness)
    ApproachTwice()
    ClearMonsters()
    LoadMonster(Guard, count=16, icon=Soldier)
    LoadMonster(Corporal, count=4, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto RunForGate }
    AlarmFightScore += 20
    if AlarmFightScore >= 40 { goto Done }
    Text("MORE GUARDS RUSH UP TO ATTACK.")
    goto GuardWaveFight

DwarfAttackCheck:
    if AttackedGuards != 1 or OutpostFought != 0 { goto Done }
    goto DwarfAndCommandant

DwarfAndCommandant:
    if OutpostConquered != 0 { exit }
    Text("A LARGE GROUP OF MEN LED BY A DWARF RUNS FOWARD TO ATTACK")
    Portrait = WhiteBeard
    SetupMonster(SmallFighter, distance=2, body=LeatherHarness)
    ApproachTwice()
    ClearMonsters()
    LoadMonster(DwarvenFighter, count=1, icon=DwarvenFighter)
    LoadMonster(Aides, count=10, icon=Soldier)
    LoadMonster(Corporal, count=12, icon=Soldier)
    LoadMonster(Guard, count=25, icon=Soldier)
    Combat()
    OutpostFought = 1
    if CombatEnd > PartyDied { goto RunForGate }
    Text("A LARGE GROUP OF MEN LED BY THE COMMANDANT RUSH FOWARD AND ATTACK.")
    Portrait = None
    SetupMonster(Fighter16, distance=2, closeup=EyepatchMan)
    ApproachTwice()
    ClearMonsters()
    LoadMonster(AlHyamDazid, count=1, icon=AlHyamDazid)
    LoadMonster(Commandant, count=1, icon=Captain)
    LoadMonster(Corporal, count=24, icon=Soldier)
    LoadMonster(Aides, count=20, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto RunForGate }
    Text("AFTER KILLING THE COMMANDANT THE REST OF THE OUTPOST EITHER SURRENDERS OR RUNS AWAY.")
    QuestZhentilOutpost = CommandantKilled
    OutpostConquered = 1
    goto Done

RunForGate:
    PartyX = 7
    PartyY = 0
    Facing = North
    CallRedraw()
    Portrait = NoseguardHelmet
    SetupMonster(Fighter16, distance=2, body=NoBody11)
    Text("YOU RUN FOR YOUR LIVES AS THE GARRISON CHASES BEHIND YOU.  YOU RUN FOR THE FRONT GATE BUT ARE STOPPED BY SOME GUARDS.")
    ClearMonsters()
    LoadMonster(Guard, count=6, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto CaughtFleeing }
    Text("YOU ESCAPE FROM THE OUTPOST.")
    NewECL(Mountains)

    CaughtFleeing:
    Text("YOU TRY TO RUN BUT ARE CAUGHT.  YOU ARE CAPTURED, STRIPPED AND TOSSED OUT INTO THE WILDERNESS.")
    goto StripAndExpel

QuartersEnter:
    if GivenQuarters != 1 { goto RelaxOrWalk }
    if NightInQuarters != 0 { goto Done }
    NightInQuarters = 1
    Text("IT IS NIGHT AND YOU ARE IN YOUR QUARTERS.")
    Delay()
    Scratch1 = VMenu("HOW DO YOU SPEND THE NIGHT?", "GOTO SLEEP", "SLEEP WITH A WATCH")
    goto [SleepNoWatch, SleepWithWatch][Scratch1]
    goto SleepNoWatch

SleepNoWatch:
    Delay()
    Hours += 2
    Text("YOU ARE ATTACKED BY FOUR GUARDS IN YOUR SLEEP")
    Damage(1, 8, 4, 8, 53)
    Morale = 80
    goto NightAmbush

SleepWithWatch:
    Delay()
    Hours += 2
    Text("YOU WAIT IN YOUR ROOM FOR ABOUT TWO HOURS WHEN SUDDENLY FOUR ARMED GUARDS BURST IN AND ATTACK YOU.")
    WaitReturnPressed()
    Morale = 70
    goto NightAmbush

NightAmbush:
    Portrait = NoseguardHelmet
    SetupMonster(Fighter16, distance=0, body=LeatherHarness)
    ClearMonsters()
    LoadMonster(Aides, count=4, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto NightFleeGuards }
    goto MoreMenAttack

    NightFleeGuards:
    Facing = West
    Scratch1 = 1
    WalkSteps()
    Text("YOU RUN INTO SOME MORE GUARDS.")
    goto SecondWaveFight

    MoreMenAttack:
    Text("AS YOU DEFEAT THE GUARDS SOME MORE MEN RUN IN AND ATTACK.")
    Portrait = NoseguardHelmet
    SetupMonster(Fighter16, distance=0, body=LeatherHarness)
    ClearMonsters()
    goto SecondWaveFight

    SecondWaveFight:
    LoadMonster(Corporal, count=5, icon=Soldier)
    LoadMonster(Aides, count=6, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto RanAndHid }
    Text("AN ALARM GOES OFF IN THE DISTANCE.")
    AlarmLevel = 1
    goto Done

    RanAndHid:
    Text("YOU MANAGE TO RUN AWAY AND HIDE.  YOU HEAR AN ALARM IN THE DISTANCE AS GUARDS ARE ISSUING OUT OF BARRACKS.")
    PartyX = 3
    PartyY = 12
    CallRedraw()
    AlarmLevel = 1
    goto Done

RelaxOrWalk:
    Text("YOU ARE IN YOUR QUARTERS.  WILL YOU WALK AROUND OR WILL YOU JUST RELAX UNTIL DINNER?")
    Scratch1 = Menu("WALK AROUND", "RELAX")
    goto [Done, EscortToDinner][Scratch1]
    exit # the original runs on into data at A6C9

EscortToDinner:
    Hours = 14
    Sky = 8
    Text("AT DUSK A GUARD COMES BY TO ESCORT YOU TO DINNER.")
    for Scratch6 = 0 to 10 {
        Scratch1 = WalkStepTable[Scratch6]
        Facing = WalkFacingTable[Scratch6]
        WalkSteps()
    }
    ShowPortrait(None, 50)
    Text("YOU ARRIVE AT THE OFFICERS MESS AND ARE SEATED WITH THE COMMANDANT AND HIS ADVISORS.  A FINE MEAL OF ROAST BOAR IS SERVED WITH A HEARTY RED WINE.")
    WaitReturnPressed()
    Text("THE COMMANDANT TURNS TO ONE OF THE CHARACTERS AND SAYS, 'SO, HOWS EVERYTHING IN PHLAN?'  WHAT WILL YOU TALK ABOUT?")
    goto DinnerTopicMenu

DinnerTopicMenu:
    Scratch1 = Menu("NOTHING", "CITY", "MAGIC", "POLITICS")
    goto [TopicNothing, TopicCity, TopicMagic, TopicPolitics][Scratch1]
    goto TopicCity

TopicCity:
    Text("ARE YOU GOING TO TALK ABOUT NEW OR OLD PHLAN?")
    Scratch1 = Menu("NEW", "OLD")
    goto [TalkNewPhlan, TalkOldPhlan][Scratch1]
    goto TopicNothing

TopicNothing:
    if TopicsDiscussed != 0 { goto ConversationDies }
    Text("THE CONVERSATION BEGINS TO DIE DOWN.  THE COMMANDANT  STARTS A CONVERSATION BY SAYING 'WELL, ARE YOU HAVING MUCH TROUBLE WITH THE CREATURES FROM OLD PHLAN?'")
    Scratch1 = Menu("TALK", "SAY LITTLE OR NOTHING")
    goto [TalkOldPhlan, ConversationDies][Scratch1]
    goto ConversationDies

ConversationDies:
    Text("THE CONVERSATION COMPLETELY DIES AND EVERYONE IS STARING AT EACH OTHER.  THE COMMANDANT EXCUSES HIMSELF AND LEAVES.  A GUARD TAKES YOU BACK TO YOUR QUARTERS.")
    goto EscortToQuarters

TyranthraxusTrouble:
    Text("THE COMMANDANT SAYS 'I EXPECT THERE WOULD BE TROUBLE; I'VE HEARD TYRANTHRAXUS IS A POWERFUL FIRE DEMON.  YOUR CITY'S DEFENCES MUST BE QUITE STRONG, HOW STRONG ARE THEY?'")
    Scratch1 = Menu("DON'T TALK", "STRONG", "AVERAGE", "WEAK")
    goto [NoAnswer, ForcesStrong, ForcesAverage, ForcesWeak][Scratch1]
    goto NoTyranthraxusTrouble

NoTyranthraxusTrouble:
    Text("THE COMMANDANT IS SURPRISED THAT YOUR CITY IS HAVING NO TROUBLE WITH TYRANTHRAXUS.  HE THEN SAYS, 'YOUR TOWN DEFENSES MUST BE STRONG IN ORDER TO HANDLE HIM.  HOW STRONG ARE THEY?'")
    Scratch1 = Menu("DON'T TALK", "STRONG", "AVERAGE", "WEAK")
    goto [NoAnswer, ForcesStrong, ForcesAverage, ForcesWeak][Scratch1]
    goto TalkNewPhlan

TalkNewPhlan:
    if TalkedNewPhlan != 0 { goto ConversationStale }
    TalkedNewPhlan = 1
    Text("AS YOU TALK ABOUT NEW PHLAN THE COMMANDANT LISTENS CAREFULLY.  THE COMMANDANT SAYS 'IS IT TRUE THAT THE REAL POWER IN NEW PHLAN IS BISHOP BRACCIO?'")
    Scratch1 = Menu("YES", "NO", "DON'T KNOW")
    goto [BraccioIsPower, BraccioNotPower, DontKnowAnswer][Scratch1]
    goto BraccioIsPower

BraccioIsPower:
    Text("YOU SAY THAT BISHOP BRACCIO IS THE REAL POWER IN PHLAN AND THE COMMANDANT RESPONDS BY SAYING 'PHLAN MUST BE A CITY OF FOOLS IF A BISHOP OF TYR HAS POWER!'")
    goto TopicDone

BraccioNotPower:
    Text("YOU SAY THAT BISHOP BRACCIO IS NOT THE REAL POWER IN PHLAN AND THE COMMANDANT RESPONDS BY SAYING 'THE CITIZENS OF PHLAN ARE WISE TO NOT FOLLOW AN IDIOT FROM THE TEMPLE OF TYR!'")
    goto TopicDone

DontKnowAnswer:
    Text("YOU SAY THAT YOU DON'T KNOW ANYTHING ABOUT THAT MATTER AND THE COMMANDANT RESPONDS BY SAYING 'YES, OF COURSE YOU DON'T.'")
    goto TopicDone

TalkOldPhlan:
    if TalkedOldPhlan != 0 { goto ConversationStale }
    TalkedOldPhlan = 1
    Text("AS YOU TALK ABOUT OLD PHLAN'S INHABITANTS THE COMMANDANT SAYS 'I KNOW A LEGEND ABOUT A FIRE BEING CALLED TIRANTIKUS.  I BELIEVE, THAT TIRANTIKUS AND TYRANTHRAXUS ARE ONE AND THE SAME'.")
    WaitReturnPressed()
    Text("YOU CONTINUE TO TALK ABOUT OLD PHLAN AND THE COMMANDANT SAYS 'IS NEW PHLAN HAVING ANY TROUBLE WITH TYRANTHRAXUS?'")
    Scratch1 = Menu("DON'T TALK", "TROUBLE", "NO TROUBLE")
    goto [TopicNothing, TyranthraxusTrouble, NoTyranthraxusTrouble][Scratch1]
    goto TopicMagic

TopicMagic:
    if AdvisorAsked != 0 { goto ConversationStale }
    AdvisorAsked = 1
    Text("ONE OF THE COMMANDANT'S ADVISORS ASKS YOU IF YOU'VE HEARD OF ANY UNIQUE MAGICAL ITEMS OR PLACES?")
    Scratch1 = Menu("DON'T TALK", "LIE", "MENTION POOL OF RADIANCE")
    goto [TopicNothing, LieAboutMagic, MentionPool][Scratch1]
    goto TopicPolitics

TopicPolitics:
    if TalkedPolitics != 0 { goto ConversationStale }
    TalkedPolitics = 1
    Text("AS YOU MENTION POLITICS THE COMMANDANT SAYS 'I'VE HEARD THAT PHLAN'S TOWN COUNCIL IS CORRUPT, SPECIFICALLY ULRICH EBERHARD.  IS THIS TRUE?")
    Scratch1 = Menu("YES", "NO", "DON'T KNOW")
    goto [EberhardCorrupt, EberhardHonest, DontKnowAnswer][Scratch1]
    goto EberhardCorrupt

EberhardCorrupt:
    Text("YOU SAY THAT ULRICH EBERHARD IS CORRUPT AND HE RESPONDS BY SAYING 'I WOULD ADVISE YOU TO BE CAREFUL WHEN DEALING WITH THE TOWN COUNCIL.'")
    goto TopicDone

EberhardHonest:
    Text("YOU SAY THAT ULRICH EBERHARD IS A LOYAL HONEST MAN.  THE COMMANDANT SMIRKS AND SAYS 'YES, OF COURSE HE IS.'")
    goto TopicDone

ForcesStrong:
    Text("YOU TELL THE COMMANDANT THAT NEW PHLAN'S FORCES ARE STRONG.  HE SEEMS IMPRESSED, AND SAYS 'IT SOUNDS LIKE YOUR CITY CAN HANDLE THAT FIRE DEMON TYRANTHRAXUS'")
    goto TopicDone

ForcesAverage:
    Text("YOU TELL THE COMMANDANT THAT NEW PHLAN'S FORCES ARE CAPABLE.  HE LOOKS AT YOU AND SAYS, 'I HOPE, FOR YOUR SAKE, THAT YOUR CITY'S FORCES CAN DEFEND THEMSELVES FROM TYRANTHRAXUS'S HORDE.'")
    goto TopicDone

ForcesWeak:
    Text("YOU TELL THE COMMANDANT THAT NEW PHLAN'S FORCES ARE WEAK.  HE SMILES AND SAYS, 'I HOPE THAT PHLAN CAN DEFEND ITSELF.'")
    goto TopicDone

NoAnswer:
    Text("AS YOU DON'T ANSWER THE LAST QUESTION THE COMMANDANT SMILES SLIGHTLY AND SAYS, 'I SEE THAT YOU ARE NOT FOOLISH AND LOOSE TONGUED.'")
    goto TopicDone

LieAboutMagic:
    Text("YOU MAKE UP STORIES ABOUT WONDERFUL MAGICAL ITEMS AND PLACES.  YOU CAPTIVATE EVERYONE SITTING AROUND YOU.")
    goto TopicDone

MentionPool:
    Text("AS YOU MENTION THE POOL OF RADIANCE THE COMMANDANT TELLS YOU A STORY ABOUT THAT VERY POOL.  YOU ENJOY THE STORY AND COPY IT INTO YOUR JOURNAL UNDER ENTRY 46.")
    goto TopicDone

TopicDone:
    WaitReturnPressed()
    if TopicsDiscussed == 2 { goto CommandantLeaves }
    Text("YOU CONTINUE YOUR CONVERSATION WITH THE COMMANDANT.  WHAT WILL BE YOUR NEXT TOPIC OF DISSCUSION?")
    TopicsDiscussed += 1
    goto DinnerTopicMenu

ConversationStale:
    Text("THE CONVERSATION GOES STALE AND THE COMMANDANT EXCUSES HIMSELF FROM THE TABLE.  YOU ARE ESCORTED BACK TO YOUR QUARTERS.")
    goto EscortToQuarters

CommandantLeaves:
    Text("YOU TALK A LITTLE WHILE LONGER THEN THE COMMANDANT STATES THAT HE HAS OTHER DUTIES TO ATTEND TO.  HE BIDS YOU GOOD NIGHT.")
    goto EscortToQuarters

EscortToQuarters:
    Scratch6 = 0
    GivenQuarters = 1
    goto QuartersWalkStep

    QuartersWalkStep:
    Scratch1 = QuartersStepTable[Scratch6]
    Facing = QuartersFacingTable[Scratch6]
    WalkSteps()
    Scratch6 += 1
    if Scratch6 <= 10 { goto QuartersWalkStep }
    Text("YOU ARRIVE BACK AT YOUR QUARTERS.")
    Delay()
    goto QuartersEnter

ClimbableWall:
    Text("THE WALL IS CLIMBABLE HERE, WILL YOU SCALE OVER IT?")
    Scratch1 = Menu("CLIMB OVER WALL", "LEAVE")
    goto [ClimbOverWall, Done][Scratch1]
    goto ClimbOverWall

ClimbOverWall:
    Text("YOU CLIMB OVER THE WALL AND LEAVE THE OUTPOST.")
    goto LeaveOutpost

FrontGateCheck:
    if QuestZhentilOutpost != 0 { goto NearFrontGate }
    if ReachedGate != 0 { goto Done }
    ReachedGate = 1
    Text("THE COMMOTION ALLOWS YOU TO GET TO THE FRONT GATE!")
    WaitReturnPressed()
    Scratch1 = Random(99)
    if Scratch1 >= 75 { goto NearFrontGate }
    Text("OUT OF THE COMMOTION A GROUP OF GUARDS ATTACK THE PARTY.")
    Portrait = NoseguardHelmet
    SetupMonster(Fighter16, distance=2, body=LeatherHarness)
    ApproachTwice()
    ClearMonsters()
    LoadMonster(Guard, count=11, icon=Soldier)
    Combat()
    goto Done

    NearFrontGate:
    Text("YOU ARE NEAR THE FRONT GATE.")
    goto Done

Done:
    RedrawView()
    Picture(None)
    exit

def ApproachTwice() {
    Delay()
    Approach()
    Delay()
    Approach()
    return
}

LeaveOutpost:
    QuestZhentilOutpost = LeftOutpost
    Portrait = None
    NewECL(Mountains)

def WaitReturnPressed() {
    Scratch8 = Menu("PRESS BUTTON OR <RETURN> TO CONTINUE")
    return
}

on camp {
    if QuestZhentilOutpost != 0 { goto RestCheckNight }
    if AttackedGuards == 0 { goto RestCheckNight }
    goto RestAlarmed
}

RestAlarmed:
    RestChance = 100
    RestEvery = 1
    exit

RestCheckNight:
    if QuestZhentilOutpost != 0 { goto RestNormal }
    if NightInQuarters != 0 { goto RestAlarmed }
    RestEvery = 168
    RestChance = 100
    exit

    RestNormal:
    RestEvery = 0
    RestChance = 1
    exit

on rest {
    RestChance = 0
    if NightInQuarters == 0 { goto EscortToDinner }
    Portrait = NoseguardHelmet
    goto GuardsSpotYou
}

on load {
    Scratch6 = 11
    LoadWalls(Stone, Planks, BlueTile)
    LoadMap(ZhentilOutpost)
    if Hours >= 14 { Scratch6 = 8 }
    Sky = Scratch6
    SkyInside = 10
    NoCoords = 0
    if AreaKind == ZhentilOutpost { exit }
    PartyX = 7
    PartyY = 0
    Facing = South
    EntryPoint = 2
    CallRedraw()
    Hours = 0
    exit
}
