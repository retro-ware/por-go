def PickEncounter() {
    Scratch1 = Random(20)
    if Scratch1 == 20 and ForcedEncounter == 0 { return }
    Scratch1 = Random(17)
    gosub [PickKobolds, PickKobolds, PickKobolds, PickKobolds, PickDisplacers, PickDisplacers,
        PickDriders, PickStirges, PickStirges, PickStirges, PickLizardMen, PickLizardMen,
        PickLizardMen, PickBandits, PickNomads, PickNomads, PickMerchants, PickMerchants][Scratch1]
    AuxTemp9 = MonsterPicTable[EncounterKind]
    AuxParlay = LeaderPicTable[EncounterKind]
    AuxMember = EncounterParam2[EncounterKind]
    AuxRoll = EncounterParam3[EncounterKind]
    AuxMonster = MonsterIdTable[EncounterKind]
    AuxEncResult = EncounterParam5[EncounterKind]
    AuxEncResult2 = MoraleTable[EncounterKind]
    AuxEncTemp = EncounterParam7[EncounterKind]
    AuxPartyStrength = LeaderIdTable[EncounterKind]
    AuxCounter = StrengthDivTable[EncounterKind]
    Scratch1 = PartyStrength()
    Scratch1 = Scratch1 / AuxCounter
    Scratch2 = Random(5)
    goto [CountThird, CountHalf, CountHalf, CountHalf, CountLeaders][Scratch2]
    goto CountThird
}

CountThird:
    Scratch1 = Scratch1 / 3
    if Scratch1 <= 0 { Scratch1 = 1 }
    goto CountLeaders

CountHalf:
    Scratch1 = Scratch1 / 2
    if Scratch1 <= 0 { Scratch1 = 1 }
    goto CountLeaders

CountLeaders:
    LeaderCount = 0
    Scratch3 = Scratch1
    AuxMenuChoice = LeaderEveryTable[EncounterKind]
    if AuxMenuChoice == 0 { goto CapMonsterCount }
    goto LeaderLoop

    LeaderLoop:
    if Scratch3 < AuxMenuChoice { goto CapMonsterCount }
    LeaderCount += 1
    Scratch3 -= AuxMenuChoice
    goto LeaderLoop

    CapMonsterCount:
    if Scratch1 >= 56 { Scratch1 = 56 - LeaderCount }
    MonsterCount = Scratch1
    return

def DrawRiver() {
    Scratch1 = 0
    Scratch2 = 0
    RiverTileIndex = 0
    goto RiverDrawBegin
}

RiverDrawBegin:
    CallBeginDraw()
    RiverFrame = 0
    goto RiverRowLoop

RiverRowLoop:
    if Scratch1 == 36 { goto RiverDrawEnd }
    Scratch7 = RiverRowTable[Scratch1]
    &Scratch3 = 8 * Scratch7
    Scratch6 = 0
    goto RiverTileLoop

    RiverTileLoop:
    if Scratch6 == 8 { goto RiverRowDone }
    if Scratch2 <= 231 {
        Scratch5 = RiverTileTable[Scratch2]
        Scratch2 += 1
    } else {
        Scratch5 = RiverTileTable2[RiverTileIndex]
        RiverTileIndex += 1
    }
    &RiverTileOffset = Scratch6 + &Scratch3
    if RiverOpen == 0 { MapPatchA[&RiverTileOffset] = Scratch5 }
    MapPatchB[&RiverTileOffset] = Scratch5
    Scratch6 += 1
    goto RiverTileLoop

    RiverRowDone:
    if Scratch1 == 8 and RiverOpen == 0 { goto RiverDrawEnd }
    if RiverState == Flowing { RiverAnimFrame() }
    Scratch1 += 1
    goto RiverRowLoop

    RiverDrawEnd:
    CallEndDraw()
    return

def RiverAnimFrame() {
    RiverFrame += 1
    if RiverFrame != 2 { return }
    DelayCounter = 0
    goto RiverDelayLoop

    RiverDelayLoop:
    if DelayCounter == 100 { goto RiverFrameShow }
    DelayCounter += 1
    goto RiverDelayLoop

    RiverFrameShow:
    CallEndDraw()
    if Scratch1 > 21 { goto RiverRowCheck29 }
    goto RiverCallC003
}

RiverCallC003:
    CallUnknownC003()
    goto RiverFrameReset

RiverRowCheck29:
    if Scratch1 > 29 { goto RiverCallC003 }
    CallUnknownC009()
    goto RiverFrameReset

RiverFrameReset:
    RiverFrame = 0
    CallBeginDraw()
    return
