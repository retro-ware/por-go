on move {
    Scratch6 = 11
    if Hours >= 14 { Scratch6 = 8 }
    Sky = Scratch6
    if LeftMap == 0 { exit }
    Scratch10 = Facing
    if TempleOutside == Geo { Scratch10 += 4 }
    CallStepForward()
    HoldMove = SkipMove
    Scratch1 = Table99A8[Scratch10]
    if Scratch1 == 26 { SymbolSet = 7 }
    if Scratch1 == 14 { SymbolSet = 3 }
    LoadMap(Scratch1)
    if Scratch10 == 1 { goto Done }
    if Scratch10 == 7 { goto Done }
    Scratch5 = Table99B0[Scratch10]
    NewECL(Scratch5)

    Done:
    exit
}

on step {
    Scratch7 = Cell & ~Inside
    Scratch1 = 0
    if Geo != TempleOutside { goto TempleSquare }
    goto [RandomEncounter, MansionRoom, MansionRoom, MansionEntry, MansionEntry, MansionEntry,
        OgreAndOrcs, GoblinSlaves, OutsideMansion, OrcGuardRoom, HayBeds, CellRoom, ScratchedMessage][Scratch7]
    goto MansionEntry
}

MansionEntry:
    if QuestEntryOrcs != 0 { goto MansionRoom }
    if OrcsKnowParty == 1 {
        AuxMonsterCount = 20
    } else {
        AuxMonsterCount = 10
    }
    if Carried(LeatherHolySymbol) { goto OrcsConfused }
    SetupMonster(Orc, distance=2, closeup=Orc)
    ApproachTwice()
    Text("AS YOU ENTER THE BUILDING YOU SEE A LARGE GROUP OF FOUL ORCS.  THEY SNARL AT YOU AND ATTACK!")
    goto FightEntryOrcs

    OrcsConfused:
    Text("AS YOU ENTER THE BUILDING YOU SEE A LARGE GROUP OF ORCS WHO ARE SURPRISED THAT YOU'RE HERE.  WITH CONFUSED LOOKS ON THEIR FACES THEY STARE AT YOU")
    EncounterMenu(Goblin, distance=1, closeup=Orc, result=Scratch1, combat=Mirror, wait=Mirror, flee=Mirror,
        advance=Mirror, parlay=Mirror, escape=0, strength=0)
    goto [ExitEvent, FightEntryOrcs, FightEntryOrcs, FightEntryOrcs][Scratch1]
    goto FightEntryOrcs

FightEntryOrcs:
    ClearMonsters()
    LoadMonster(Orc4, count=AuxMonsterCount, icon=Orc)
    Morale = 99
    Combat()
    Scratch10 = 0
    if CombatEnd > PartyDied { goto StepBack }
    QuestEntryOrcs = RewardDue
    Scratch1 = Scratch7 - 3
    goto [GreatHall, BodiesRoom, ExitEvent][Scratch1]
    goto MansionRoom

MansionRoom:
    goto [ExitEvent, RubbleRoom, DiningRoom, GreatHall, BodiesRoom, ExitEvent][Scratch7]
    goto RubbleRoom

RubbleRoom:
    if LastRoomVisited == InRubbleRoom { goto ExitEvent }
    LastRoomVisited = InRubbleRoom
    Text("YOU ARE IN A ROOM FILLED WITH RUBBLE AND WASTE.")
    if Search == NotSearching { goto ExitEvent }
    Scratch1 = MansionRoomBits & Tapestry
    if Scratch1 != 0 { goto ExitEvent }
    Print("  IN THE RUBBLE YOU SPOT A BEAUTIFUL TAPESTRY.  WILL YOU TAKE IT?")
    Scratch1 = Menu("TAKE", "LEAVE")
    goto [TakeTapestry, ExitEvent][Scratch1]
    goto TakeTapestry

TakeTapestry:
    MansionRoomBits |= Tapestry
    Scratch1 = 4
    goto GiveTreasure

DiningRoom:
    if LastRoomVisited == InDiningRoom { goto ExitEvent }
    LastRoomVisited = InDiningRoom
    Text("YOU ARE IN THE REMAINS OF A ONCE GREAT DINING ROOM.  FURNITURE LIES BURNED AND BROKEN THROUGHOUT THE ROOM.  ")
    Scratch1 = MansionRoomBits & SkeletonJewelry
    if Scratch1 != 0 { goto ExitEvent }
    if Search == NotSearching { goto ExitEvent }
    NewLine()
    Print("UNDERNEATH A BROKEN TABLE YOU FIND A CRUSHED SKELETON.  IT HAS SOME JEWELRY ON IT; WILL YOU TAKE IT?")
    Scratch2 = Menu("TAKE", "LEAVE")
    goto [TakeSkeletonJewelry, ExitEvent][Scratch2]
    goto TakeSkeletonJewelry

TakeSkeletonJewelry:
    MansionRoomBits |= SkeletonJewelry
    ClearMonsters()
    Treasure(jewelry=3)
    Combat()
    goto ExitEvent

GreatHall:
    if LastRoomVisited == InGreatHall { goto ExitEvent }
    Text("BEFORE YOU IS A GREAT HALL.  THE FLOOR IS LITTERED WITH ORC TRASH AND THE WALLS HAVE CRUDE BLACK HANDS PAINTED ON THEM.  ")
    LastRoomVisited = InGreatHall
    Scratch1 = MansionRoomBits & TrapDoor
    if Scratch1 != 0 { goto ExitEvent }
    SearchFindSecret()
    if AuxFlag != 1 { goto ExitEvent }
    Print("YOU FIND A TRAP DOOR.  WILL YOU OPEN IT?")
    Scratch3 = Menu("OPEN", "LEAVE")
    goto [OpenHallTrapDoor, ExitEvent][Scratch3]
    goto OpenHallTrapDoor

OpenHallTrapDoor:
    MansionRoomBits |= TrapDoor
    Scratch1 = 6
    goto GiveTreasure

BodiesRoom:
    if LastRoomVisited == InBodiesRoom { goto ExitEvent }
    Text("YOU ARE IN A ROOM THAT IS FILLED WITH THE DEAD BODIES OF THE ORC'S PREVIOUS VICTIMS.  ")
    LastRoomVisited = InBodiesRoom
    Scratch1 = MansionRoomBits & BodyJewelry
    if Scratch1 != 0 { goto ExitEvent }
    if Search == NotSearching { goto ExitEvent }
    Print("ADMIST THE BODIES YOU FIND SOME JEWELRY.  WILL YOU TAKE IT?")
    Scratch2 = Menu("TAKE", "LEAVE")
    goto [TakeBodiesJewelry, ExitEvent][Scratch2]
    goto TakeBodiesJewelry

TakeBodiesJewelry:
    MansionRoomBits |= BodyJewelry
    ClearMonsters()
    Treasure(jewelry=3)
    Combat()
    goto ExitEvent

    if Scratch1 != 0 { goto ExitEvent }

GoblinSlaves:
    Scratch1 = MansionRoomBits & GoblinSlavesMet
    if Scratch1 != 0 { goto ExitEvent }
    SetupMonster(Goblin, distance=0, closeup=Goblin)
    Text("IN THIS ROOM ARE SOME GOBLIN SLAVES THAT ARE COWERING IN THE CORNER.  THEY ALL SCREAM 'PLEASE DON'T KILL US.'  WHAT WILL YOU DO?")
    Scratch1 = Menu("KILL THEM", "LET THEM GO")
    goto [KillGoblins, FreeGoblins][Scratch1]
    goto KillGoblins

KillGoblins:
    Text("YOU COLDLY KILL THE DEFENCELESS GOBLINS.")

FreeGoblins:
    Text("YOU LET THE GOBLINS GO.  AS THEY LEAVE ONE OF THEM SAYS 'TO ENTER THE TEMPLE YOU NEED A HOLY SYMBOL OF BANE.'")

OutsideMansion:

HayBeds:

    Text("YOU ENTER A ROOM THAT HAS SOME CRUDE HAY BEDS. WHAT WILL YOU DO?")
    Scratch1 = Menu("SEARCH THE BEDS", "LEAVE THEM")
    goto [SearchBeds, ExitEvent][Scratch1]

SearchBeds:
    Scratch1 = MansionFightBits & HayBedOrcsFight

    SetupMonster(Orc, distance=0, closeup=Orc)
    Text("AS YOU SEARCH THE BEDS A GROUP OF ORCS COMES INTO THE ROOM.  THEY ATTACK IMMEDIATELY.")

CellRoom:
    if LastMansionRoom == InCellRoom { goto ExitEvent }
    LastMansionRoom = InCellRoom
    Text("YOU ENTER A ROOM THAT HAS BEEN COVERTED INTO A CELL.")
    goto ExitEvent

ScratchedMessage:
    Text("ON THE WALL HERE IS A POORLY SCRATCHED OUT MESSAGE.  YOU CAN BARELEY MAKE OUT THE WORDS")
    NewLine()
    Print("'NORTH WALL.  RI  T FRONT GLAS . SE'")
    goto ExitEvent

TempleSquare:
    Scratch1 = 0
    goto [RandomEncounter, TempleInside, TempleInside, TempleInside, TempleEntrance, SetGuardsPassed,
        ExitEvent, TempleInside][Scratch7]
    goto SetGuardsPassed

SetGuardsPassed:
    TempleGuardState = GuardsPassed
    goto ExitEvent

TempleEntrance:
    if QuestTempleOfBane >= RewardDue { goto ExitEvent }
    if TempleGuardState == GuardsFought {
        goto ExitEvent
    } else if TempleGuardState > GuardsFought {
        goto ExitAmbush
    }
    SetupMonster(Orc, distance=0, closeup=Orc)
    Text("YOU STAND IN FRONT OF THE ENTRANCE TO A LARGE SHADOWY TEMPLE.  AN OLD BLIND DECREPIT ORC STANDS OUTSIDE WITH EIGHT ORC GUARDS.")
    if OrcsKnowParty == 1 { goto GuardsLetYouIn }
    Print("  THE AGED ORC COMES GRASPING TOWARD YOU.  WILL YOU LET HIM TOUCH YOU?")
    Scratch1 = 0
    Scratch1 = Menu("LET HIM TOUCH YOU", "BACK OFF", "ATTACK")
    goto [BlindOrcChecks, BackOffOnce, FightTempleGuards][Scratch1]
    goto BackOffOnce

BackOffOnce:
    Text("AS YOU BACK AWAY THE BLIND ORC SAYS, 'I MUST CHECK IF YOU ARE WORTHY TO ENTER THE TEMPLE OF BANE, DO NEED NOT BE AFRAID.'")
    Scratch1 = Menu("LET HIM CHECK", "BACK OFF AGAIN", "ATTACK")
    goto [BlindOrcChecks, BackOffTwice, FightTempleGuards][Scratch1]
    goto GuardsLetYouIn

GuardsLetYouIn:
    Print("  AS YOU APPROACH THEY MOVE, ALLOWING YOU ENTRANCE INTO THE TEMPLE.")
    goto ExitEvent

BlindOrcChecks:
    if Carried(LeatherHolySymbol) { goto HolySymbolWelcome }
    Text("THE OLD BLIND ORC PAWS THE PARTY AS IF LOOKING FOR SOMETHING.  AFTER A CAREFULL PAWING, HE BACKS AWAY AND YELLS 'GUARDS, IN THE NAME OF BANE, KILL THESE UNBELIEVERS!'")
    goto FightTempleGuards

BackOffTwice:
    Text("AS YOU BACK AWAY AGAIN THE OLD BLIND ORC IS CONVINCED YOU ARE NOT WORSHIPPERS.  HE GRUNTS AN ORDER TO ATTACK.")
    goto FightTempleGuards

FightTempleGuards:
    ClearMonsters()
    LoadMonster(Orc4, count=8, icon=Orc)
    Morale = 99
    Combat()
    if CombatEnd > PartyDied { goto StepBack }
    TempleGuardState = GuardsFought
    goto ExitEvent

HolySymbolWelcome:
    Text("THE OLD BLIND ORC PAWS THE CHARACTERS LOOKING FOR SOMETHING.  FINDING THE LEATHER HOLY SYMBOL, HE SMILES THEN GRUNTS 'WELCOME TO THE TEMPLE OF BANE, SUPREME GOD OF THE REALMS.'")
    goto ExitEvent

TempleInside:
    if Scratch7 == 7 { goto BloodyAltar }
    if Search != Searching { goto ExitEvent }
    TempleSearchFlag = 1
    goto [ExitEvent, TrapDoorOne, TrapDoorTwo, TrapDoorThree, ExitEvent, ExitEvent, ExitEvent][Scratch7]
    goto TrapDoorOne

TrapDoorOne:
    Scratch2 = TempleTreasureBits & Spot1
    if Scratch2 != 0 { goto ExitEvent }
    SearchFindSecret()
    if AuxFlag != 1 { goto ExitEvent }
    AskOpenTrapDoor()
    if Scratch3 == 1 { goto ExitEvent }
    TempleTreasureBits = Spot1 | TempleTreasureBits
    Scratch1 = 8
    goto GiveTreasure

TrapDoorTwo:
    Scratch2 = TempleTreasureBits & Spot2
    if Scratch2 != 0 { goto ExitEvent }
    SearchFindSecret()
    if AuxFlag != 1 { goto ExitEvent }
    AskOpenTrapDoor()
    if Scratch3 == 1 { goto ExitEvent }
    TempleTreasureBits |= Spot2
    Scratch1 = 9
    goto GiveTreasure

TrapDoorThree:
    Scratch2 = TempleTreasureBits & Spot3
    if Scratch2 != 0 { goto ExitEvent }
    SearchFindSecret()
    if AuxFlag != 1 { goto ExitEvent }
    AskOpenTrapDoor()
    if Scratch3 == 1 { goto ExitEvent }
    TempleTreasureBits |= Spot3
    Scratch1 = 17
    goto GiveTreasure

def AskOpenTrapDoor() {
    Text("ON THE FLOOR YOU FIND A TRAP DOOR.  WILL YOU OPEN IT?")
    Scratch3 = Menu("OPEN TRAPDOOR", "LEAVE TRAPDOOR")
    return
}

BloodyAltar:
    if AltarDestroyed == 1 { goto AltarRemains }
    Text("IN FRONT OF YOU IS AN ALTAR THAT IS SMEARED WITH BLOOD AND COVERED WITH CRUDE BLACK HAND PRINTS.  WHAT WILL YOU DO?")
    Scratch1 = Menu("DESTROY ALTAR", "LEAVE IT")
    goto [DestroyAltar, ExitEvent][Scratch1]
    goto DestroyAltar

DestroyAltar:
    if QuestTempleOfBane >= RewardDue { goto AltarGone }
    Text("AS YOU DESTROY THE ALTAR, THE TEMPLE DOORS FLY OPEN AND A SCREAMING HORDE OF ORCS LED BY A HALF-ORC CLERIC ATTACK.  YOU TURN TO FACE THEM.")
    Facing = West
    CallRedraw()
    SetupMonster(Orc, distance=2, closeup=Orc)
    ApproachTwice()
    AltarDestroyed = 1
    goto FightHalfOrcCleric

    AltarGone:
    Text("YOU DESTROY THE EVIL ALTAR AND THE UNEASY FEELING LEAVES YOU.")
    AltarDestroyed = 1
    goto ExitEvent

AltarRemains:
    Text("THE REMAINS OF AN EVIL ALTAR ARE NEARBY.")
    goto ExitEvent

GiveTreasure:
    Picture(TreasureChest)
    ClearMonsters()
    Treasure(items=Scratch1)
    Combat()
    goto [ExitEvent, CheckAllLooted][TempleSearchFlag]
    goto CheckAllLooted

CheckAllLooted:
    if TempleTreasureBits == 7 and QuestTempleOfBane == 0 { goto LootAmbush }
    goto ExitEvent

ExitAmbush:
    SetupMonster(Orc, distance=0, closeup=Orc)
    Text("AS YOU EXIT THE TEMPLE YOU ARE ATTACKED BY A VERY LARGE GROUP OF ORCS.  LEADING THEM IS A HALF-ORC WHO IS SCREAMING 'KILL!  KILL, THE UNBELIEVING TOADS.  KILL THEM OR SUFFER MY WRATH!'")
    goto FightHalfOrcCleric

LootAmbush:
    SetupMonster(Orc, distance=0, closeup=Orc)
    Text("THE MINUTE YOU FIND THE TREASURE A LARGE GROUP OF ORCS CHARGE THROUGH THE TEMPLE DOORS.  LEADING THEM IS A HALF-ORC WHO IS SCREAMING 'KILL!  KILL, THESE SCUM WHO WOULD STEAL FROM OUR TEMPLE.'")
    goto FightHalfOrcCleric

FightHalfOrcCleric:
    Morale = 99
    ClearMonsters()
    LoadMonster(OrcLeader5, count=5, icon=OrcLeader)
    LoadMonster(Mace, count=1, icon=Mace)
    LoadMonster(Orc44, count=20, icon=OrcLeader)
    LoadMonster(Orc4, count=30, icon=Orc)
    Combat()
    if CombatEnd > PartyDied { goto StepBack }
    QuestTempleOfBane = RewardDue
    Text("ON THE DEAD BODY OF THE HALF-ORC CLERIC YOU FIND A PIECE OF PAPER.  YOU PUT THE PAPER IN YOUR JOURNAL UNDER ENTRY 25.")
    Delay()
    Delay()
    Scratch2 = 0
    goto FindDirten

FindDirten:
    SelectMember(Scratch2)
    if $SelName == "DIRTEN" { goto DirtenFound }
    goto NextMember

NextMember:
    Scratch2 += 1
    if Scratch2 < 8 { goto FindDirten }
    goto ExitEvent

DirtenFound:
    if SelControl < 128 { goto NextMember }
    if SelStatus > 127 { goto DirtenLeaves }
    Text("DIRTEN SAYS, 'THANK YOU FOR HELPING US REGAIN OUR TEMPLE.  I MUST STAY HERE NOW.  I BID YOU FAREWELL.'")
    goto DirtenLeaves

    DirtenLeaves:
    SelName = 0
    SelStatus = Died
    Scratch2 += 128
    SelectMember(Scratch2)
    goto ExitEvent

def SearchFindSecret() {
    AuxText = 0
    goto CheckMemberFinds
}

CheckMemberFinds:
    SelectMember(AuxText)
    if SelStatus != Okay { goto NextSearcher }
    AuxSearchTries = Search
    if SelRace == Elf { AuxSearchTries += 1 }
    if SelRace == HalfElf { AuxSearchTries += 1 }
    AuxSearchTarget = Random(5)
    if AuxSearchTries > AuxSearchTarget { goto SecretFound }
    goto NextSearcher

    NextSearcher:
    AuxText += 1
    if AuxText < 8 { goto CheckMemberFinds }
    AuxFlag = 0
    return

    SecretFound:
    AuxFlag = 1
    return

RandomEncounter:
    if QuestTempleOfBane >= RewardDue { goto CheckEntryCleared }
    goto RollEncounter

    CheckEntryCleared:
    if QuestEntryOrcs >= RewardDue { goto ExitEvent }
    goto RollEncounter

    RollEncounter:
    Scratch1 = Random(59)
    if Search < Searching { goto RollNoSearch }
    if Scratch1 > 49 { goto StartEncounter }
    goto RollNoSearch

    RollNoSearch:
    if Scratch1 > 56 { goto StartEncounter }
    goto ExitEvent

    StartEncounter:
    PickEncounter()
    PartySurprise(Scratch1, Scratch2)
    Surprise(Scratch1, Scratch2, AuxEncSurpriseA, AuxEncSurpriseB)
    goto [SpotGroup, SurprisedByGroup, SurpriseGroup, FaceToFace][Surprise]
    goto SpotGroup

SpotGroup:
    Text("YOU SPOT A GROUP OF ")
    Print($AuxCounter)
    Print(".")
    EncounterMenu(AuxEncMonster, distance=MonsterDist, closeup=AuxEncDist, result=Scratch1, combat=Flee,
        wait=CloseIn, flee=Attack, advance=Mirror, parlay=Talk, far="THE MONSTERS STAND READY",
        escape=AuxEncChoice, strength=AuxEncChoice2)
    AuxEncValue = 130
    goto [ExitEvent, FightGroup, StepBack, IntimidateGroup][Scratch1]
    goto OrcPatrol

OrcPatrol:
    Text("YOU SEE A GROUP OF FOUR ORCS WHOSE SHIELDS BEAR A BLACK HAND ON A RED FIELD.  THEY CAUTIOSLY LOOK YOU OVER.")
    Scratch1 = Menu("PRESS <RETURN> TO CONTINUE")
    EncounterCount1 = 4
    EncounterTemp = 0
    EncounterMenu(AuxEncMonster, distance=MonsterDist, closeup=AuxEncDist, result=Scratch1, combat=Attack,
        wait=Mirror, flee=Mirror, advance=Mirror, parlay=Talk, far="THE ORCS READY THEIR SPEARS",
        escape=AuxEncChoice, strength=AuxEncChoice2)
    AuxEncValue = 10
    AuxEncDist = 4
    goto [OrcsSawParty, FightGroup, OrcsSawParty, ParlayOrcPatrol][Scratch1]
    goto SurprisedByGroup

SurprisedByGroup:
    if 1 == EncounterIsOrcs { goto OrcPatrol }
    SetupMonster(AuxEncMonster, distance=0, closeup=AuxEncDist)
    Text("YOU ARE SURPRISED BY ")
    Print($AuxCounter)
    Print(".")
    goto FightGroup

OrcsSawParty:
    OrcsKnowParty = 1
    goto ExitEvent

SurpriseGroup:
    Text("YOU SPOT A GROUP OF ")
    Print($AuxCounter)
    Print(".")
    EncounterMenu(AuxEncMonster, distance=MonsterDist, closeup=AuxEncDist, result=Scratch1, combat=Mirror,
        wait=Mirror, flee=Mirror, advance=Mirror, parlay=Mirror, near="YOU HAVE SURPRISED THEM.",
        escape=AuxEncChoice, strength=AuxEncChoice2)
    AuxEncValue = 50
    if 1 == EncounterIsOrcs { goto OrcsSurprisedMenu }
    goto [ExitEvent, PartySurprises, ExitEvent, IntimidateGroup][Scratch1]
    goto OrcsSurprisedMenu

    OrcsSurprisedMenu:
    goto [ExitEvent, PartySurprises, StepBack, FightGroup][Scratch1]
    goto FaceToFace

FaceToFace:
    Text("YOU ARE SUDDENLY FACE TO FACE WITH A GROUP OF ")
    Print($AuxCounter)
    Print(".")
    EncounterMenu(AuxEncMonster, distance=0, closeup=AuxEncDist, result=Scratch1, combat=Mirror, wait=Mirror,
        flee=Mirror, advance=Mirror, parlay=Mirror, near="THEY ARE AS SURPRISED AS YOU.",
        escape=AuxEncChoice, strength=AuxEncChoice2)
    AuxEncValue = 50
    goto [ExitEvent, FightGroup, StepBack, FightGroup][Scratch1]
    goto PartySurprises

PartySurprises:
    Surprise = MonstersAreSurprised
    goto FightGroup

FightGroup:
    Morale = AuxEncMorale
    Scratch4 = 1 + AuxSightText
    ClearMonsters()
    LoadMonster(AuxSightText, count=EncounterCount1, icon=AuxEncDist)
    if EncounterCount2 == 0 { goto GroupCombat }
    LoadMonster(Scratch4, count=EncounterCount2, icon=Scratch4)
    goto GroupCombat

    GroupCombat:
    if EncounterIsOrcs == 1 { Treasure(items=[LeatherHolySymbol*6]) }
    Combat()
    if EncounterIsOrcs == 1 and CombatEnd == 1 { OrcsKnowParty = 1 }
    if CombatEnd > PartyDied { goto StepBack }
    goto ExitEvent

ExitEvent:
    TempleSearchFlag = 0
    RedrawView()
    Picture(None)
    SavedGeo24 = Geo
    exit

StepBack:
    PartyX = PrevX
    PartyY = PrevY
    Geo = SavedGeo24
    CallRedraw()
    goto ExitEvent

IntimidateGroup:
    Scratch1 = Parlay(30, 30, 30, 5, 60)
    Scratch2 = Random(30)
    Scratch1 = Scratch1 + Scratch2
    Scratch1 = Scratch1 + SelReaction
    Scratch1 = Scratch1 + AuxEncBonus
    if Scratch1 < 140 { goto NonbelieversLeave }
    Text("YOU INTIMIDATE THE MONSTERS SO MUCH THEY RUN AWAY!")
    goto ExitEvent

    NonbelieversLeave:
    if Scratch1 < 100 { goto GetOuttaHere }
    Text("'WE GOTTA KEEP ALL YOU NONBELIVERS OUTA HERE' THE MONSTERS SAY AS THEY BEGIN TO LEAVE.  'SO HIT THE ROAD YA BUMS.'  THE MONSTERS TURN AND LEAVE.")
    goto ExitEvent

    GetOuttaHere:
    if Scratch1 < 75 { goto TeachLesson }
    Text("'GET OUTA HERE BEFORE WE MAKE YOU LEAVE' THE MONSTERS YELL.  'AND YOU WONT COME BACK IF YOU KNOW WHATS GOOD FER YA'  THE MONSTERS THEN LEAVE.")
    goto ExitEvent

    TeachLesson:
    Text("THE MONSTERS SCREAM 'OH YEAH, WELL I GUESS WERE JUST GONNA HAVE TO TEACH YOUSE GUYS A LESSON OR TWO'  THEY RUSH FOWARD TO ATTACK.")
    goto FightGroup

ParlayOrcPatrol:
    Scratch1 = Parlay(1, 2, 0, 0, 1)
    goto [PatrolHarmless, PatrolThreatens, PatrolKnowsTreasure][Scratch1]
    goto PatrolHarmless

PatrolHarmless:
    Text("THE ORCS YELL OUT, 'WE HAVE ORDERS TO CHECK OUT SUSPICIOUS INDIVIDUALS, BUT YOUSE GUYS LOOK HARMLESS ENOUGH.'  AFTER TALKING THE FOUR ORCS WANDER AWAY.")
    goto OrcsSawParty

PatrolThreatens:
    Text("IN A DEEP THREATENING VOICE ONE OF THE ORCS SAYS 'YOU MOLES BETTER WATCH YOUR STEP OR YOU SHALL FEEL THE WRATH OF BANE!'  AFTER SAYING THIS HE SPITS AND LEAVES.")
    goto OrcsSawParty

PatrolKnowsTreasure:
    Text("ONE OF THE ORCS SAYS, 'OH, YEAH YOUSE THEM GUYS WHO KNOW WHERE DE TREASURE IS.  WE NOT SUPPOSED TO KILL YOU YET.'  THE ORCS THEN WANDER AWAY.")
    goto OrcsSawParty

def PickEncounter() {
    EncounterIsOrcs = 0
    Scratch1 = Random(9)
    goto [PickPatrolOrcs, PickPatrolOrcs, PickPatrolOrcs, PickPatrolOrcs, PickOrcs, PickOrcs,
        PickGoblins, PickGoblins, PickHobgoblins, PickOgres][Scratch1]
    goto PickGoblins
}

PickGoblins:
    $AuxCounter = "GOBLINS"
    AuxEncIndex = 0
    goto LoadEncounterData

PickPatrolOrcs:
    $AuxCounter = "ORCS"
    EncounterIsOrcs = 1
    AuxEncIndex = 1
    goto LoadEncounterData

PickOrcs:
    $AuxCounter = "ORCS"
    AuxEncIndex = 1
    goto LoadEncounterData

PickHobgoblins:
    $AuxCounter = "HOBGOBLINS"
    AuxEncIndex = 2
    goto LoadEncounterData

PickOgres:
    $AuxCounter = "OGRES"
    AuxEncIndex = 3
    goto LoadEncounterData

LoadEncounterData:
    MonsterDist = EncDistTable[AuxEncIndex]
    AuxSightText = EncMonsterTable[AuxEncIndex]
    AuxEncMonster = EncPictureTable[AuxEncIndex]
    AuxEncDist = EncTable4[AuxEncIndex]
    AuxEncMorale = EncMoraleTable[AuxEncIndex]
    AuxEncChoice = EncTable6[AuxEncIndex]
    AuxEncChoice2 = EncTable7[AuxEncIndex]
    AuxEncTblA = EncStrengthAdjTable[AuxEncIndex]
    AuxEncTblB = EncGroupStepTable[AuxEncIndex]
    AuxEncTblC = EncGroupIncTable[AuxEncIndex]
    AuxEncSurpriseA = EncTable11[AuxEncIndex]
    AuxEncSurpriseB = EncTable12[AuxEncIndex]
    AuxEncBonus = EncTable13[AuxEncIndex]
    Scratch2 = 0
    EncounterCount2 = 0
    if 1 == EncounterIsOrcs { goto OrcPatrol }
    Scratch1 = PartyStrength()
    Scratch1 = Scratch1 / AuxEncTblA
    if Scratch1 < 4 { Scratch1 = 4 }
    Scratch4 = Random(9)
    goto [AddThree, AddTwo, AddTwo, AddTwo, CountGroups][Scratch4]
    goto AddThree

AddThree:
    Scratch1 = Scratch1 / 3
    goto CountGroups

AddTwo:
    Scratch1 = Scratch1 / 2
    goto CountGroups

CountGroups:
    Scratch3 = Scratch1
    goto CountGroupsLoop

    CountGroupsLoop:
    if Scratch3 < AuxEncTblB { goto ClampGroupSize }
    Scratch2 += AuxEncTblC
    Scratch3 -= AuxEncTblB
    goto CountGroupsLoop

    ClampGroupSize:
    if Scratch1 >= AuxEncTblB { SetSecondGroup() }
    if Scratch1 > 56 { Scratch1 = 56 - Scratch2 }
    EncounterCount1 = Scratch1
    return

def SetSecondGroup() {
    EncounterCount2 = Scratch2
    return
}

def ApproachTwice() {
    Delay()
    Approach()
    Delay()
    Approach()
    return
}
