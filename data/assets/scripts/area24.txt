LoadEncounterData:
	MonsterDist = EncDistTable[AuxEncIndex]
	AuxSightText = EncMonsterTable[AuxEncIndex]
	AuxEncMonster = EncPictureTable[AuxEncIndex]
	AuxEncDist = EncTable4[AuxEncIndex]
	AuxEncMorale = EncMoraleTable[AuxEncIndex]
	AuxEncChoice = EncTable6[AuxEncIndex]
	AuxEncChoice2 = EncTable7[AuxEncIndex]
	AuxEncTblA = EncStrengthAdjTable[AuxEncIndex]
	AuxEncTblB = EncGroupStepTable[AuxEncIndex]
	AuxEncTblC = EncGroupIncTable[AuxEncIndex]
	AuxEncSurpriseA = EncTable11[AuxEncIndex]
	AuxEncSurpriseB = EncTable12[AuxEncIndex]
	AuxEncBonus = EncTable13[AuxEncIndex]
	Scratch2 = 0
	EncounterCount2 = 0
	if 1 == EncounterIsOrcs: Goto OrcPatrol
	PartyStrength Scratch1
	Scratch1 = Scratch1 / AuxEncTblA
	if Scratch1 < 4: Scratch1 = 4
	Scratch4 = Random 9
	OnGoto Scratch4 AddThree AddTwo AddTwo AddTwo CountGroups

AddThree:
	Scratch1 = Scratch1 / 3
	Goto CountGroups

AddTwo:
	Scratch1 = Scratch1 / 2
	Goto CountGroups

CountGroups:
	Scratch3 = Scratch1

CountGroupsLoop:
	if Scratch3 < AuxEncTblB: Goto ClampGroupSize
	Scratch2 += AuxEncTblC
	Scratch3 -= AuxEncTblB
	Goto CountGroupsLoop

ClampGroupSize:
	if Scratch1 >= AuxEncTblB: Gosub SetSecondGroup
	if Scratch1 > 56: Scratch1 = 56 - Scratch2
	EncounterCount1 = Scratch1
	Return

SetSecondGroup:
	EncounterCount2 = Scratch2
	Return
