def WalkSteps() {
    Scratch2 = 1
    if Scratch1 == 0 {
        CallRedraw()
        Delay()
        return

    WalkStepLoop:
    if Scratch2 > Scratch1 { return }
    gosub [StepNorth, StepEast, StepSouth, StepWest][Facing]
    CallRedraw()
    Delay()
    Scratch2 += 1
    goto WalkStepLoop
}
