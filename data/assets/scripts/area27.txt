def CheckTerrainAhead() {
    ProbeX = MapX
    ProbeY = MapY
    gosub [ProbeN, ProbeNE, ProbeE, ProbeSE, ProbeS, ProbeSW, ProbeW, ProbeNW][FacingRaw]
    CallUnknownC01B()
    Scratch1 = 0
    Scratch3 = 8

    CheckTerrainLoop:
    if Scratch1 == Scratch3 { return }
    Scratch2 = BlockedTerrain[Scratch1]
    if ProbeWall == Scratch2 {
        HoldMove = SkipMove
        return
    }
    Scratch1 += 1
    goto CheckTerrainLoop
}

def PickEncounter() {
    Scratch1 = Random(20)
    if Scratch1 == 20 and ForceEncounter27 == 0 { return }
    Scratch1 = Random(19)
    gosub [PickKobolds, PickKobolds, PickKobolds, PickKobolds, PickGnolls, PickGnolls, PickGnolls,
        PickLizardMen, PickLizardMen, PickLizardMen, PickGiantLizards, PickGiantLizards,
        PickHippogriffs, PickHippogriffs, PickWyverns, PickTrolls, PickTrolls, PickPhaseSpiders,
        PickMinotaurs, PickMinotaurs][Scratch1]
    AuxMember = EncData2[EncounterType27]
    AuxRoll = EncData3[EncounterType27]
    AuxMonster = EncMonster[EncounterType27]
    AuxEncResult = EncData5[EncounterType27]
    AuxEncResult2 = EncMorale[EncounterType27]
    AuxEncTemp = EncData7[EncounterType27]
    AuxPartyStrength = EncMonster2[EncounterType27]
    AuxCounter = EncStrength[EncounterType27]
    Scratch1 = PartyStrength()
    Scratch1 = Scratch1 / AuxCounter
    Scratch2 = Random(5)
    goto [GroupSizeLarge, GroupSizeMedium, GroupSizeMedium, GroupSizeMedium, SplitGroups][Scratch2]
} # falls into GroupSizeLarge

GroupSizeLarge:
    Scratch1 = Scratch1 / 3
    if Scratch1 <= 0 { Scratch1 = 1 }
    goto SplitGroups

GroupSizeMedium:
    Scratch1 = Scratch1 / 2
    if Scratch1 <= 0 { Scratch1 = 1 }

SplitGroups:
    EncounterExtraCount27 = 0
    Scratch3 = Scratch1
    AuxMenuChoice = EncGroupSize[EncounterType27]
    if AuxMenuChoice == 0 { goto CapGroupSize }

    SplitGroupsLoop:
    if Scratch3 < AuxMenuChoice { goto CapGroupSize }
    EncounterExtraCount27 += 1
    Scratch3 -= AuxMenuChoice
    goto SplitGroupsLoop

    CapGroupSize:
    if Scratch1 >= 56 { Scratch1 = 56 - EncounterExtraCount27 }
    EncounterCount27 = Scratch1
    return
