on move {
    Scratch6 = 11
    if Hours >= 14 { Scratch6 = 8 }
    Sky = Scratch6
    if LeftMap == 0 { exit }
    if QuestBivant == BoyPurchased { QuestBivant = RewardDue }
    NewECL(Mountains)
}

on step {
    if BaseCaptured == Captured { goto Done }
    if QuestBivant >= RewardDue { goto GuardsHuntBoy }
    goto CheckAnimals
}

CheckAnimals:
    if AnimalsLoose == 0 { goto DispatchCell }
    Scratch1 = Cell & Inside
    if Scratch1 != 0 { goto CheckRecapture }
    Text("ANIMALS ARE RUNNING ALL OVER THE COMPOUND.")
    goto CheckRecapture

    CheckRecapture:
    if BaseAlerted != 0 { goto DispatchCell }
    StampedeTimer += 1
    if Search != NotSearching { StampedeTimer += 9 }
    if StampedeTimer >= 60 { RecaptureAnimals() }
    goto DispatchCell

    DispatchCell:
    Scratch10 = Cell & ~Inside
    if AnimalsLoose != 0 { goto DispatchCellStampede }
    goto [Done, MainGateEnter, MerchantsCamp, SlavePenOutside, CaptainDoorGuards, CaptainRoom,
        SeedyManPass, GuardTower, BarracksEnter, AnimalPenPacked, AnimalPenGate, SlavePenGateCheck,
        InsideSlavePen][Scratch10]
    goto DispatchCellStampede

    DispatchCellStampede:
    goto [Done, MainGateEmpty, MerchantsCamp, SlavePenOutsideStampede, CaptainDoorGuards, CaptainRoom,
        Done, GuardTower, EmptyBarracks, AnimalPensEmpty, AnimalPenDoorOpen, SlavePenGate,
        InsideSlavePen][Scratch10]
    goto RecaptureAnimals

RecaptureAnimals:
    Text("THE ANIMALS HAVE BEEN CAPTURED.")
    AnimalsLoose = 0
    StampedeTimer = 0
    return

GuardsHuntBoy:
    if BoyBought != 0 { goto CheckAnimals }
    if BaseAlerted != 0 { goto CheckAnimals }
    GuardSpotChance += 1
    Scratch1 = Random(11)
    if AnimalsLoose != 0 { Scratch1 += 1 }
    if Scratch1 > GuardSpotChance { goto CheckAnimals }
    GuardSpotChance = 1 + GuardPatrolCount
    Portrait = LongHaired
    SetupMonster(Fighter16, distance=2, body=BareChest)
    Delay()
    Approach()
    Delay()
    Approach()
    Portrait = None
    goto [GuardsSpotBoy, AlarmRunForGate, AnotherWaveAttacks, CaptainAppears][GuardPatrolCount]
    goto GuardsSpotBoy

GuardsSpotBoy:
    Text("SOME OF THE GUARDS SPOT YOU WITH THE BOY.")
    ClearMonsters()
    LoadMonster(Buccaneer, count=20, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto CutOffWithBoy }
    GuardPatrolCount += 1
    goto CheckAnimals

AlarmRunForGate:
    Text("AN ALARM GOES OFF IN THE DISTANCE.  YOUR ONLY HOPE IS TO RUN FOR THE FRONT GATE.")
    ClearMonsters()
    LoadMonster(RdLvlFighter, count=4, icon=Soldier)
    LoadMonster(Buccaneer, count=30, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto CutOffWithBoy }
    GuardPatrolCount += 1
    goto CheckAnimals

AnotherWaveAttacks:
    Text("ANOTHER WAVE OF MEN ATTACK YOU.")
    ClearMonsters()
    LoadMonster(RdLvlFighter, count=4, icon=Soldier)
    LoadMonster(Buccaneer, count=40, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto CutOffWithBoy }
    GuardPatrolCount += 1
    goto CheckAnimals

MainGateEnter:
    if EnteredBase != 0 { goto MainGateCheck }
    Text("YOU ARE JUST INSIDE THE MAIN GATE OF THE BUCCANEER'S BASE.  YOU ARE BEING TAKEN TO THE CENTER OF THE BASE WHERE THE MERCHANTS MAKE CAMP.")
    EnteredBase = 1
    Cell = 2
    Scratch3 = 0
    Facing = South
    goto EscortToCampLoop

    EscortToCampLoop:
    if Scratch3 == 10 { goto MerchantsCamp }
    PartyY += 1
    CallRedraw()
    DelayLoop()
    Scratch3 += 1
    goto EscortToCampLoop

    MainGateCheck:
    if AnimalsLoose != 0 { goto MainGateEmpty }
    if BoyBought != 0 { goto GateGuardsStand }
    if BaseCaptured == Captured { goto MainGateEmpty }
    if QuestBivant != 0 { goto GateGuardsAttack }
    goto GateGuardsStand

    GateGuardsStand:
    Text("YOU ARE NEAR THE MAIN GATE TO THE COMPOUND.  SOME BUCCANEERS STAND GUARD NEAR THE ENTRANCE.")
    exit

MainGateEmpty:
    Text("YOU ARE AT THE MAIN GATE TO THE COMPOUND.")
    exit

GateGuardsAttack:
    if GateGuardsDefeated != 0 { goto Done }
    Portrait = LongHaired
    SetupMonster(Fighter16, distance=0, body=BareChest)
    Portrait = None
    Text("SOME GUARDS RUSH FOWARD AND ATTACK.")
    Morale = 100
    ClearMonsters()
    LoadMonster(Buccaneer, count=8, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto FleeToCamp }
    GateGuardsDefeated = 1
    exit

MerchantsCamp:
    Scratch10 = 2
    Text("YOU ARE IN THE MERCHANTS' CAMP AREA.")
    goto Done

SlavePenOutside:
    if WallAhead != 8 { goto Done }
    if QuestBivant != 0 { goto SlavePenOutsideSeen }
    if SlavePenSeen > 0 { goto SlavePenOutsideSeen }
    if SlavePenBlocked != 0 { goto SlavePenOutsideSeen }
    SlavePenSeen = 1
    Picture(Prisoners)
    Text("YOU ARE JUST OUTSIDE A PEN FOR HOLDING SLAVES.  THE PEOPLE HERE ARE WORN AND BEATEN, DESTINED TO A LIFE OF SLAVERY.  AROUND THE PEN THERE ARE MANY GUARDS AND SLEAZY MERCHANTS LOOKING AT THE SLAVES.")
    PressToContinue()
    Text("YOU SEE A BOY THAT HAS A HEART-SHAPED BIRTHMARK ON HIS LEFT SHOULDER.  AS YOU STARE AT THE BOY, A SLEAZY FAT MERCHANT APPROACHES YOU AND SAYS ...")
    PressToContinue()
    Text("'THAT YOUNG BOY WILL BRING A GOOD PRICE AT ZHENTIL KEEP.  I THINK I SHALL BUY HIM.'")
    PressToContinue()
    Scratch1 = VMenu("WHAT WILL YOU DO?", "PUNCH MERCHANT", "AGREE", "SAY NOTHING", "RESCUE BOY")
    goto [PunchMerchant, AgreeWithMerchant, Done, RescueBoyMenu][Scratch1]
    goto PunchMerchant

PunchMerchant:
    Text("YOU PUNCH THE MERCHANT IN THE NOSE AND HE FALLS TO THE GROUND, BADLY BLEEDING.  THE MERCHANT YELLS FOR HIS GUARDS AND THEY COME RUNNING.")
    Morale = 60
    ClearPicture()
    Portrait = RedFaced
    SetupMonster(Fighter16, distance=2, body=PlateArmor)
    Delay()
    Approach()
    Delay()
    Approach()
    Portrait = None
    ClearMonsters()
    LoadMonster(Buccaneer, count=6, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto FleeToCamp }
    Text("AFTER YOU DEFEAT THE GUARDS YOU SEE A CROWD HAS FORMED.  FROM WITHIN THE CROWD A LARGE MAN IN GLEAMING CHAIN MAIL STEPS OUT.  HE SAYS 'THAT WAS AMUSING BUT DO NOT CAUSE ANYMORE TROUBLE!'")
    PressToContinue()
    Text("THE MAN THEN LEAVES AND BEFORE YOU IS THE SLEAZY MERCHANT.  HE BEGS FORGIVENESS THEN WIMPERS AWAY.")
    goto Done

FleeToCamp:
    Text("YOU RUN AWAY FROM THE BATTLE BACK TO THE MERCHANT'S AREA.  FROM BEHIND YOU HEAR PEOPLE CALLING YOU NAMES LIKE 'PANSY' AND 'WIMP.'")
    PartyX = 8
    PartyY = 10
    CallRedraw()
    goto Done

AgreeWithMerchant:
    Text("YOU AGREE WITH THE MERCHANT THAT THE YOUNG BOY IS VALUABLE.  THE MERCHANT LAUGHS THEN SAYS 'YOU'LL HAVE TO OUTBID ME FOR THE WAIF.'  THE SLEAZY MERCHANT THEN WALKS AWAY.")
    goto Done

RescueBoyMenu:
    Text("YOU IGNORE THE SLEAZY MERCHANT AND CONCENTRATE ON RESCUING THE BOY.")
    Scratch1 = VMenu("HOW WILL YOU RESCUE THE BOY?", "SNEAK INTO PEN", "ATTACK GUARDS", "DO NOT RESCUE BOY NOW")
    goto [SneakToPenGate, AttackPenGuards, Done][Scratch1]
    goto SneakToPenGate

SneakToPenGate:
    Text("YOU MOVE THROUGH THE CROWD NONCHALANTLY UNTIL YOU ARE NEAR THE FRONT GATE OF THE SLAVE PEN.")
    Facing = South
    PartyX = 7
    PartyY = 6
    CallRedraw()
    if PenGuardsDefeated != 0 { goto Done }
    goto PenGateGuards

PenGateGuards:
    Portrait = LongHaired
    SetupMonster(Fighter16, distance=0, body=BareChest)
    Text("YOU ARE NEAR THE FRONT GATE TO THE SLAVE PEN.  TWO GUARDS GUARD THE DOOR.  WHAT WILL YOU DO?")
    Scratch1 = Menu("ATTACK", "TRY TO SNEAK BY", "LEAVE")
    goto [AttackPenGuards, SneakPastFails, StepBack][Scratch1]
    goto AttackPenGuards

AttackPenGuards:
    ClearPicture()
    Text("YOU ATTACK THE GUARDS.")
    Morale = 65
    ClearMonsters()
    LoadMonster(Buccaneer, count=4, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto CutOffWithBoy }
    PenGuardsDefeated = 1
    goto Done

CutOffWithBoy:
    Portrait = LongHaired
    SetupMonster(Fighter16, distance=2, body=BareChest)
    Delay()
    Approach()
    Delay()
    Approach()
    Text("YOU TRY TO RUN FROM THE GUARDS BUT ARE CUT OFF BY NEW ARRIVALS.  THE YOUNG BOY IS SLOWING YOU DOWN.  IF YOU RUN AGAIN YOU WILL LEAVE HIM BEHIND.")
    goto BuccaneerWaves

BuccaneerWaves:
    Morale = 80
    ClearMonsters()
    LoadMonster(Buccaneer, count=20, icon=Soldier)
    LoadMonster(RdLvlFighter, count=1, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto FleeLeaveBoy }
    SetupMonster(Fighter16, distance=2, body=BareChest)
    Delay()
    Approach()
    Delay()
    Approach()
    Text("AFTER DEFEATING THE FIRST WAVE OF BUCCANEERS,  YOU SEE ANOTHER WAVE FORMING IN THE DISTANCE.  WITH A LOUD ROAR THEY ATTACK YOU IN FULL FORCE.")
    ClearMonsters()
    LoadMonster(RdLvlFighter, count=1, icon=Soldier)
    LoadMonster(ThLvlFighter101, count=1, icon=Soldier)
    LoadMonster(Buccaneer, count=50, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto FleeLeaveBoy }
    SetupMonster(Fighter16, distance=2, body=BareChest)
    Text("AFTER DEFEATING THE LAST, WAVE YOU SEE ANOTHER WAVE ATTACK FROM THE SOUTH OF THE COMPOUND.")
    Morale = 100
    ClearMonsters()
    LoadMonster(ThLvlFighter101, count=1, icon=Soldier)
    LoadMonster(RdLvlFighter, count=2, icon=Soldier)
    LoadMonster(Buccaneer, count=53, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto FleeLeaveBoy }
    goto CaptainAppears

CaptainAppears:
    if CaptainKilled != 0 { goto FinalWave }
    Text("THE SLAVES CHEER YOUR VICTORY.  A LARGE GROUP OF BUCCANEERS STANDS IN DISBELIEF.  THE BUCCANEERS PART AS A LARGE MAN IN GLEAMING CHAIN MAIL STEPS FORTH.  THE SLAVES FALL SILENT.")
    Portrait = Bearded
    SetupMonster(Fighter16, distance=2, body=HeraldTabard)
    Delay()
    Approach()
    Delay()
    Approach()
    Text("THE CAPTAIN SAYS 'YOU HAVE FOUGHT WELL.  YOU MAY RANSOM YOURSELVES OR DIE!'  WHAT WILL YOU DO?")
    Scratch1 = Menu("SURRENDER", "ATTACK")
    goto [SurrenderToCaptain, AttackCaptain][Scratch1]
    goto SurrenderToCaptain

SurrenderToCaptain:
    Text("THE CAPTAIN SAYS 'YOU HAVE CHOSEN WISELY.'  SOME OF THE BUCCANEERS COME FORWARD AND STRIP YOU OF SOME OF YOUR ITEMS AND MONEY.  THEY ESCORT YOU OUT OF THE COMPOUND.")
    Rob(1, 75, 50)
    goto LeaveBase

AttackCaptain:
    Text("YOU THROW CAUTION TO THE WIND AND ATTACK.  THE CAPTAIN SMILES AND THEN RUSHES TO BATTLE.")
    Morale = 100
    ClearMonsters()
    LoadMonster(Captain, count=1, icon=Captain)
    goto FinalBattle

FinalBattle:
    LoadMonster(ThLvlFighter41, count=15, icon=Soldier)
    LoadMonster(Buccaneer, count=40, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto FleeLeaveBoy }
    Text("AFTER DEFEATING THE LAST OF THE BUCCANNEERS THE REST SURRENDER OR FLEE.  YOU RESCUE THE SLAVES AND CAPTURE THE FORT.")
    BaseCaptured = Captured
    QuestBivant = RewardDue
    goto Done

FinalWave:
    Portrait = LongHaired
    SetupMonster(Fighter16, distance=2, body=BareChest)
    Text("THE LAST OF THE BUCCANNEERS FORM INTO A FINAL WAVE AND ATTACK.")
    Morale = 60
    ClearMonsters()
    goto FinalBattle

FleeLeaveBoy:
    Text("YOU RUN FROM THE BATTLE AND OUT THE OUTPOST GATE.  THE YOUNG BOY COULD NOT KEEP UP AND WAS LEFT BEHIND.")
    goto LeaveBase

LeaveBase:
    QuestBivant = Bungled
    BaseCaptured = FledWithoutBoy
    NewECL(Mountains)

SneakPastFails:
    Text("YOU TRY TO SNEAK UP PAST THE GUARDS BUT THEY ARE TOO ALERT.  YOU CANNOT GET BY THE GUARDS INTO THE SLAVE PEN.  WHAT WILL YOU DO?")
    Scratch1 = Menu("ATTACK GUARDS", "LEAVE")
    goto [AttackPenGuards, StepBack][Scratch1]
    goto SlavePenOutsideSeen

SlavePenOutsideSeen:
    Text("YOU ARE JUST OUTSIDE THE SLAVE PEN.")
    goto Done

SlavePenOutsideStampede:
    if SlavePenSeen > 0 { goto SlavePenUnguarded }
    if SlavePenBlocked != 0 { goto SlavePenUnguarded }
    if QuestBivant != 0 { goto SlavePenUnguarded }
    Text("YOU ARE JUST OUTSIDE THE SLAVE PEN.  AMOUNG THE SLAVES IS A YOUNG BOY WITH A HEART-SHAPED BIRTHMARK.")
    SlavePenSeen = 1
    goto Done

    SlavePenUnguarded:
    Text("YOU ARE JUST OUTSIDE AN UNGUARDED SLAVE PEN.")
    goto Done

QuartersAlarm:
    Text("AS YOU APPROACH THE CAPTAIN'S QUARTERS THE ELITE GUARDS SOUND AN ALARM AND GUARDS RUSH FORTH AND ATTACK.")
    Portrait = LongHaired
    goto BuccaneerWaves

CaptainDoorGuards:
    if BaseAlerted != 0 { goto Done }
    if DoorGuardsKilled != 0 { goto Done }
    Portrait = LongHaired
    SetupMonster(Fighter16, distance=0, body=BareChest)
    Portrait = None
    if QuestBivant >= RewardDue { goto QuartersAlarm }
    Text("STANDING IN FRONT OF THE DOOR ARE TWO LARGE GUARDS.  ONE OF THEM SAYS 'DO YOU HAVE A PASS TO SEE THE CAPTAIN?'  WHAT WILL YOU DO?")
    Scratch1 = Menu("SHOW PASS", "LEAVE", "BRIBE", "ATTACK")
    goto [ShowPass, LeaveDoorGuards, BribeRefused, AttackDoorGuards][Scratch1]
    goto BribeRefused

BribeRefused:
    Text("YOU OFFER THE GUARDS SOME MONEY BUT THEY REFUSE.  ONE OF THE GUARDS SAYS 'GET OTTA' HERE YA SCUM'")
    Scratch1 = Menu("ATTACK", "LEAVE")
    goto [AttackDoorGuards, LeaveDoorGuards][Scratch1]
    goto AttackDoorGuards

AttackDoorGuards:
    Morale = 100
    ClearMonsters()
    LoadMonster(ThLvlFighter41, count=2, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto CutOffWithBoy }
    Text("YOU KILL THE GUARDS BUT ARE CAUGHT DOING IT.  A LARGE GROUP OF GUARDS RUN TOWARD YOU AND ATTACK.")
    DoorGuardsKilled = 1
    goto BuccaneerWaves

ShowPass:
    if not Carried(Pass) { goto NoPass }
    Text("THE GUARDS LOOK AT YOUR PASS AND LET YOU PASS.")
    goto Done

    NoPass:
    Text("YOU DO NOT HAVE A PASS TO SHOW THE GUARDS!")
    goto CaptainDoorGuards

CaptainRoom:
    if BaseAlerted != 0 { goto Done }
    if CaptainFightDone != 0 { goto Done }
    if CaptainKilled == 1 { goto Done }
    Portrait = LongHaired
    SetupMonster(Fighter16, distance=0, body=BareChest)
    Text("YOU ENTER A ROOM WITH FOUR GUARDS IN GLEAMING CHAIN MAIL.  AS YOU ENTER, ONE OF THEM GOES THROUGH THE SOUTH DOOR AND RETURNS WITH THE BUCCANEER CAPTAIN.")
    PressToContinue()
    Portrait = Bearded
    SetupMonster(Fighter16, distance=0, body=HeraldTabard)
    Text("THE CAPTAIN LOOKS YOU OVER AND SAYS 'WHAT DO YOU WANT?'")
    Scratch1 = Menu("ATTACK", "MENTION YOUNG BOY", "LEAVE")
    goto [CaptainFight, MentionBoy, StepBack][Scratch1]
    goto CaptainFight

CaptainFight:
    Text("YOU DRAW YOUR WEAPONS AND ATTACK.  THE CAPTAIN DRAWS HIS SWORD AND YELLS FOR HIS GUARDS TO ATTACK!")
    Morale = 100
    ClearMonsters()
    LoadMonster(Captain, count=1, icon=Captain)
    LoadMonster(ThLvlFighter41, count=4, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto FleeQuarters }
    Text("YOU KILL THE CAPTAIN.")
    CaptainKilled = 1
    goto Done

    FleeQuarters:
    PartyX = 7
    PartyY = 11
    Facing = North
    CallRedraw()
    Text("YOU HAVE RUN OUTSIDE THE CAPTAINS QUARTERS.  AN ALARM GOES OFF IN THE DISTANCE.  YOU SEE A LARGE GROUP OF ANGRY BUCCANEERS CHARGING TOWARD YOU!")
    goto BuccaneerWaves

MentionBoy:
    Text("YOU MENTION THE YOUNG BOY IN THE SLAVE PEN AND THE CAPTAIN SEEMS INTERESTED.  HE SAYS 'I CAN LET YOU BUY THE YOUNG BOY BUT IT WILL COST YOU ALOT!'  WHAT WILL YOU DO?")
    Scratch1 = Menu("BUY THE BOY", "ATTACK")
    goto [BuyTheBoy, CaptainFight][Scratch1]
    goto BuyTheBoy

BuyTheBoy:
    Text("THE CAPTAIN SAYS 'YOU HAVE MADE A GOOD DEAL.'  AFTER TALKING THE CAPTAIN MAKES A MOTION AND THE BOY IS BROUGHT TO YOU.")
    QuestBivant = BoyPurchased
    Rob(1, 75, 10)
    PressToContinue()
    BoyBought = 1
    Text("THE CAPTAIN THEN LEAVES THE ROOM AND YOU ARE ESCORTED BACK OUTSIDE.")
    PartyX = 7
    PartyY = 11
    Facing = North
    CallRedraw()
    goto Done

SeedyManPass:
    if PassBought != 0 { goto Done }
    if QuestBivant >= RewardDue { goto Done }
    Portrait = EyepatchHelmet
    SetupMonster(Fighter16, distance=0, body=ShopkeeperApron)
    Text("THERE IS A SEEDY LITTLE MAN HERE WHO WHISPERS 'PSSST, BUDDY.  WANT TO BUY A PASS TO SEE THE CAPTAIN?'")
    Scratch2 = Menu("BUY PASS", "LEAVE")
    goto [BuyPass, Done][Scratch2]
    goto BuyPass

BuyPass:
    Text("YOU BUY THE PASS.  THE SEEDY LITTLE MAN DISAPPEARS AROUND A CORNER.")
    Rob(0, 25, 0)
    PassBought = Yes
    ClearMonsters()
    Treasure(items=[Pass])
    Combat()
    goto Done

GuardTower:
    if BaseAlerted != 0 { goto Done }
    Portrait = LongHaired
    SetupMonster(Fighter16, distance=0, body=BareChest)
    if QuestBivant >= RewardDue { goto TowerGuardsAttack }
    Text("YOU HAVE ENTERED ONE OF THE GUARD TOWERS.  INSIDE ARE FOUR BUCCANEERS.  ONE OF THEM SAYS 'THIS IS A RESTRICTED AREA, LEAVE NOW!'  WHAT WILL YOU DO?")
    Scratch2 = Menu("ATTACK", "LEAVE")
    goto [AttackTowerGuards, StepBack][Scratch2]
    goto TowerGuardsAttack

    TowerGuardsAttack:
    Text("THE TOWER GUARDS ATTACK.")
    goto TowerFight

AttackTowerGuards:
    Text("YOU ATTACK THE TOWER GUARDS.")
    goto TowerFight

TowerFight:
    Morale = 60
    ClearMonsters()
    LoadMonster(Buccaneer, count=4, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto CutOffWithBoy }
    goto Done

BarracksEnter:
    if BaseAlerted != 0 { goto Done }
    Portrait = LongHaired
    SetupMonster(Fighter16, distance=0, body=BareChest)
    if QuestBivant >= RewardDue { goto BarracksGuardsAttack }
    Text("YOU ENTER A BARRACKS FILED WITH TWENTY BUCCANEERS.  ONE OF THE BUCCANEERS SAYS 'THIS AREA IS OFF LIMITS.  LEAVE NOW.' WHAT WILL YOU DO?")
    Scratch1 = Menu("ATTACK", "LEAVE")
    goto [AttackBarracks, StepBack][Scratch1]
    goto BarracksGuardsAttack

    BarracksGuardsAttack:
    Text("THE GUARDS IN THE BARRACKS ATTACK.")
    goto BarracksFight

AttackBarracks:
    Text("YOU ATTACK THE BUCCANEERS.")
    goto BarracksFight

BarracksFight:
    Morale = 70
    ClearMonsters()
    LoadMonster(Buccaneer, count=20, icon=Soldier)
    Combat()
    if CombatEnd > PartyDied { goto CutOffWithBoy }
    goto BuccaneerWaves

EmptyBarracks:
    Text("YOU ENTER AN EMPTY BARRACKS.  IT SEEMS MOST OF THE MEN ARE OUTSIDE CHASING THE ANIMALS.")
    goto Done

AnimalPenPacked:
    Text("YOU ARE NEAR A DENSLY PACKED ANIMAL PEN.")
    goto Done

AnimalPensEmpty:
    Text("YOU ARE JUST OUTSIDE SOME EMPTY ANIMAL PENS.")
    goto Done

AnimalPenGate:
    if BaseAlerted != 0 { goto AnimalPensEmpty }
    Text("YOU ARE AT THE GATE TO THE ANIMAL PENS.  THE ANIMALS ARE RESTLESS.  WHAT WILL YOU DO?")
    Scratch1 = Menu("RELEASE ANIMALS", "LEAVE")
    goto [ReleaseAnimals, Done][Scratch1]
    goto ReleaseAnimals

ReleaseAnimals:
    Text("AS SOON AS YOU OPEN THE GATE THE ANIMALS INSIDE CHARGE OUT.  THEY STAMPEDE ACROSS THE COMPOUND.")
    PressToContinue()
    if BaseAlerted != 0 { goto Done }
    Text("THERE IS MASS CONFUSION AS THE ANIMALS RUN AROUND THE COMPOUND.  GUARDS ARE LEAVING THEIR POSTS IN ORDER TO CONTROL THE ANIMALS.")
    AnimalsLoose = Yes
    goto Done

AnimalPenDoorOpen:
    Text("YOU ARE NEAR AN OPEN DOOR TO THE ANIMAL PENS.")
    goto Done

SlavePenGateCheck:
    if BaseAlerted != 0 { goto SlavePenGate }
    if PenGuardsDefeated == 0 { goto PenGateGuards }
    goto SlavePenGate

SlavePenGate:
    Text("YOU ARE NEAR THE GATE TO THE SLAVE PEN.")
    goto Done

InsideSlavePen:
    if SlavePenBlocked != 0 { goto Done }
    ShowPortrait(None, 42)
    Text("YOU ARE IN THE SLAVE PEN.  STANDING NEAR YOU IS THE YOUNG BOY.  WHAT WILL YOU DO?")
    Scratch1 = Menu("GET HIM", "LEAVE")
    goto [TakeBoy, StepBack][Scratch1]
    goto TakeBoy

TakeBoy:
    Text("YOU TAKE THE YOUNG BOY AND LEAVE.  THE OTHER SLAVES ALSO RUN OUT THE SLAVE PEN.")
    QuestBivant = RewardDue
    PartyX = 7
    PartyY = 6
    Facing = North
    CallRedraw()
    goto Done

Done:
    Picture(None)
    RedrawView()
    Portrait = None
    exit

LeaveDoorGuards:
    if PrevY != 12 { goto StepBack }
    PartyY = 10
    CallRedraw()
    goto Done

StepBack:
    PartyX = PrevX
    PartyY = PrevY
    CallRedraw()
    goto Done

def PressToContinue() {
    Scratch2 = Menu("PRESS BUTTON OR <RETURN> TO CONTINUE")
    return
}

def DelayLoop() {
    Scratch4 = 0
    goto DelayLoopNext

    DelayLoopNext:
    if Scratch4 > 200 { return }
    Scratch4 += 1
    goto DelayLoopNext
}

def ClearPicture() {
    RedrawView()
    Picture(None)
    return
}

on camp {
    if BaseCaptured == Captured { goto RestSafe }
    if Scratch10 != 2 { goto RestDangerous }
    if QuestBivant >= RewardDue { goto RestDangerous }
    goto RestSafe

    RestSafe:
    RestChance = 1
    RestEvery = 0
    exit

    RestDangerous:
    RestChance = 100
    RestEvery = 5
    exit
}

on rest {
    if QuestBivant >= RewardDue { goto GuardsSpotBoy }
    RestChance = 0
    Text("A GUARD COMES BY AND SAYS 'IF YOU WISH TO CAMP, GO TO THE MERCHANTS AREA!'")
    exit
}

on load {
    if AreaKind == BuccaneerBase { goto LoadAreaFiles }
    PartyY = 0
    PartyX = 8
    Facing = South
    CallRedraw()
    Scratch1 = 0
    goto LoadAreaFiles

    LoadAreaFiles:
    LoadWalls(Planks, Palisade, Palisade)
    LoadMap(BuccaneerBase)
    Scratch6 = 11
    if Hours >= 14 { Scratch6 = 8 }
    Sky = Scratch6
    SkyInside = 10
    exit
}
