CheckAnimals:
	if AnimalsLoose == 0: Goto DispatchCell
	Scratch1 = Cell & 128
	if Scratch1 != 0: Goto CheckRecapture
	PrintClear "ANIMALS ARE RUNNING ALL OVER THE COMPOUND."

CheckRecapture:
	if BaseAlerted != 0: Goto DispatchCell
	StampedeTimer += 1
	if Search != 0: StampedeTimer += 9
	if StampedeTimer >= 60: Gosub RecaptureAnimals

DispatchCell:
	Scratch10 = Cell & 127
	if AnimalsLoose != 0: Goto DispatchCellStampede
	OnGoto Scratch10 Done MainGateEnter MerchantsCamp SlavePenOutside CaptainDoorGuards CaptainRoom SeedyManPass GuardTower BarracksEnter AnimalPenPacked AnimalPenGate SlavePenGateCheck InsideSlavePen

	OnGoto Scratch10 Done MainGateEmpty MerchantsCamp SlavePenOutsideStampede CaptainDoorGuards CaptainRoom Done GuardTower EmptyBarracks AnimalPensEmpty AnimalPenDoorOpen SlavePenGate InsideSlavePen

	PrintClear "THE ANIMALS HAVE BEEN CAPTURED."
	AnimalsLoose = 0
	StampedeTimer = 0
	Return

	Approach
