RollSurprise:
	PartySurprise Scratch8 Scratch9
	And CastleFlags 1 Scratch10
	IfNe
	Add 1 Scratch9 Scratch9
	Surprise Scratch8 Scratch9 0 0
	if AlarmEndHour != 0: And Surprise 1 Surprise
	if Surprise == 3: Save 0 Surprise
	Compare Surprise 1
	Return

CheckAlarmSounding:
	if AlarmEndHour < 1: Return
	Save Hours AuxRoll
	if Hours == 0 && AlarmHour == 23: Save 24 AuxRoll
	if AuxRoll < AlarmEndHour: Goto AlarmStillOn
	if TenMinutes < AlarmMinute: Goto AlarmStillOn
	Save 0 AlarmEndHour
	Compare 0 1
	Return

AlarmStillOn:
	if Search == 1: Goto SayAlarmSounding
	Random 9 AuxRoll
	if AuxRoll != 0: Goto AlarmActive

SayAlarmSounding:
	PrintClear "AN ALARM IS SOUNDING."
	Delay

AlarmActive:
	Compare 1 0
	Return

StartAlarm:
	if AlarmEndHour == 0: PrintClear "AN ALARM STARTS RINGING!"
	Delay
	Save Hours AlarmHour
	Save TenMinutes AlarmMinute
	Add 1 AlarmHour AlarmEndHour
	Return

AskTakeIt:
	Print "DO YOU TAKE IT?"

YesNoMenu:
	HMenu Scratch10 2 "YES" "NO"
	PrintClear " "
	Compare Scratch10 0
	Return

PressReturn:
	HMenu Scratch9 1 "PRESS BUTTON OR <RETURN> TO CONTINUE"
	PrintClear " "
	Return

SetSkyByHour:
	Save 11 Scratch1
	Save 9 SkyInside
	if Hours >= 14: Save 8 Scratch1
	Save Scratch1 Sky
	Return

event0:
	And CastleRoomFlags 127 CastleRoomFlags
	if LeftMap == 0: Goto OnStep
	if Geo == 3: GetTable ByFacing9AEB Facing Scratch10
	if Geo == 4: GetTable ByFacing9AEF Facing Scratch10
	if Geo == 5: GetTable ByFacing9AF3 Facing Scratch10
	if Geo == 6: GetTable ByFacing9AF7 Facing Scratch10

	Save 255 HoldMove
	Exit

ByFacing9AEF:

ByFacing9AF3:

ByFacing9AF7:

OnStep:
	Gosub SetSkyByHour
	And Cell 31 Scratch1
	if Scratch1 == 21 && Facing == 3: Goto CastleGate
	if Scratch1 == 22 && Facing == 1: Goto CastleGate
	if Scratch1 == 4 && Facing == 1: Goto GoUpstairs

	Save 255 HoldMove
	Exit
