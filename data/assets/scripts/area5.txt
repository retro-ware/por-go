def RollSurprise() {
    PartySurprise(Scratch8, Scratch9)
    if CastleFlags has Disguised { Scratch9 += 1 }
    Surprise(Scratch8, Scratch9, 0, 0)
    if AlarmEndHour != 0 { Surprise &= PartyIsSurprised }
    if Surprise == BothSurprised { Surprise = NoSurprise }
    return Surprise
}

def CheckAlarmSounding() {
    if AlarmEndHour < 1 { return }
    AuxRoll = Hours
    if Hours == 0 and AlarmHour == 23 { AuxRoll = 24 }
    if AuxRoll < AlarmEndHour { goto AlarmStillOn }
    if TenMinutes < AlarmMinute { goto AlarmStillOn }
    AlarmEndHour = 0
    return 0 < 1

    AlarmStillOn:
    if Search == Searching { goto SayAlarmSounding }
    AuxRoll = Random(9)
    if AuxRoll != 0 { goto AlarmActive }
    goto SayAlarmSounding

    SayAlarmSounding:
    Text("AN ALARM IS SOUNDING.")
    Delay()
    goto AlarmActive

    AlarmActive:
    return 1 < 0
}

def StartAlarm() {
    if AlarmEndHour == 0 { Text("AN ALARM STARTS RINGING!") }
    Delay()
    AlarmHour = Hours
    AlarmMinute = TenMinutes
    AlarmEndHour = 1 + AlarmHour
    return
}

def AskTakeIt() {
    Print("DO YOU TAKE IT?")
    goto YesNoMenu
}

YesNoMenu:
    Scratch10 = Menu("YES", "NO")
    Text(" ")
    return Scratch10

def WaitReturnPressed() {
    Scratch9 = Menu("PRESS BUTTON OR <RETURN> TO CONTINUE")
    Text(" ")
    return
}

def SetSkyByHour() {
    Scratch1 = 11
    SkyInside = 9
    if Hours >= 14 { Scratch1 = 8 }
    Sky = Scratch1
    return
}

on move {
    CastleRoomFlags &= KitchenOverseer|Scribes|EmptyScriptorium|CadornaEvidenceFound|TyranthraxusNote|BarbarianReportSeen|BarracksGiantsMet
    if LeftMap == 0 { goto OnStep }
    if Geo == ValjevoCastle3 { Scratch10 = ByFacing9AEB[Facing] }
    if Geo == ValjevoCastle4 { Scratch10 = ByFacing9AEF[Facing] }
    if Geo == ValjevoGrounds { Scratch10 = ByFacing9AF3[Facing] }
    if Geo == ValjevoCastle6 { Scratch10 = ByFacing9AF7[Facing] }
    LoadMap(Scratch10)
    CallStepForward()
    HoldMove = SkipMove
    exit

    OnStep:
    SetSkyByHour()
    Scratch1 = Cell & 31
    if Scratch1 == 21 and Facing == West { goto CastleGate }
    if Scratch1 == 22 and Facing == East { goto CastleGate }
    if Scratch1 == 4 and Facing == East { goto GoUpstairs }
    if WallAhead == 7 { goto ThornHedge }
    if WallAhead != 9 { exit }
    goto ThornHedge

    ThornHedge:
    Text("THERE IS A HEDGE WITH POISONOUS THORNS IN THAT DIRECTION. DO YOU WANT TO TRY TO CUT THROUGH?")
    if YesNoMenu() != "YES" { exit }
    Print(" AS YOU SLICE THROUGH SOMEONE IS PRICKED BY A THORN.")
    Damage(128, 1, 1, 200, 0)
    CallStepForward()
    HoldMove = SkipMove
    exit
}

on step {
    if CheckAlarmSounding() { goto SpecialSquare }
    if Cell has Special { goto SpecialSquare }
    Scratch1 = 2
    if Search != Searching { Scratch1 = 20 }
    Scratch10 = Random(Scratch1)
    if Scratch10 == 0 { goto L_ABF0 }
    goto SpecialSquare

    SpecialSquare:
    Scratch1 = Cell & 31
    if Scratch1 > 29 { NewECL(ValjevoCastle3) }
    if Scratch1 > 20 { exit }
    goto [Done, HarmlessMage, ArguingGiants, GateGiants, Done, ThroneRoom, TowerTrolls, Disorient][Scratch1]
    goto event2
}

on camp {
    RestEvery = 0
    RestChance = 10
    if AlarmEndHour == 0 { exit }
    if Cell has Special { exit }
    RestEvery = 1
    RestChance = 17
    goto Done
}

Done:
    RedrawView()
    CallRedraw()
    exit

on load {
    SetSkyByHour()
    NoCoords = 1
    exit
}

Disorient:
    if CastleRoomFlags has VisitFlag { exit }
    Text("YOU FEEL A SUDDEN DISORIENTATION.")
    Scratch10 = Random(3)
    Scratch1 = 3 + Scratch10
    PartyX = DestX[Scratch10]
    PartyY = DestY[Scratch10]
    LoadMap(Scratch1)
    CallRedraw()
    exit

FledCombat:
    CombatEnd = PartyFled
    goto AfterCombat

AfterCombat:
    if CombatEnd == FightOver { return }
    StartAlarm()
    CastleFlags &= HallSeen|PatrolTalkedAway|LooseStoneFound|SleepingGiantsFought|SwordInWell|FakeTyranthraxus|GiantsPassed
    if CombatEnd != PartyFled { return }
    goto StepBack

StepBack:
    HoldMove = SkipMove
    PartyX = PrevX
    PartyY = PrevY
    Facing = Facing + South
    if Facing > West { Facing -= 4 }
    goto Done

GateGiants:
    if CastleDeedFlags has PasswordGiants { exit }
    if RollSurprise() > PartyIsSurprised { goto SurpriseGiants }
    Surprise = NoSurprise
    Text("THREE GIANTS STEP UP TO YOU.")
    goto GiantsMenu

    SurpriseGiants:
    Text("YOU SURPRISE THREE GIANTS!")
    goto GiantsMenu

    GiantsMenu:
    Delay()
    EncounterMenu(HillGiant, distance=2, closeup=HillGiant, result=Scratch10, combat=Attack, wait=Talk,
        flee=Attack, advance=Talk, parlay=Talk, near="THEY HOLD THEIR WEAPONS READY.", escape=12,
        strength=12)
    goto [Done, FightGiants, FledCombat][Scratch10]
    Scratch10 = Parlay(0, 1, 1, 1, 0)
    goto [WhoSentYou][Scratch10]
    Text("'ALL RIGHT, THEN. WHAT TODAY'S PASSWORD?'")
    goto AskPassword

    AskPassword:
    $AuxInputText = InputString(6)
    Scratch10 = Random(4)
    if Scratch10 == 0 { goto NoPasswordAttack }
    Text("'WRONG! GUESS AGAIN!")
    goto AskPassword

    NoPasswordAttack:
    Text("'WRONG! THERE IS NO PASSWORD! HA! HA! HA!' THE GIANTS ATTACK.")
    goto FightGiants

WhoSentYou:
    Text("'OH, YEAH? WHO SEND YOU?'")
    $AuxInputText = InputString(12)

    Text("'OH, WE SORRY! JUST DOING JOB! PLEASE NO TELL? WE LET YOU... WE STAY AT POST. MAKE SURE NO ONE GO BY!'")
    goto Done

    WrongMasterName:

    goto FightGiants

FightGiants:

    Morale = 60
    RedrawView()
    ClearMonsters()
    LoadMonster(HillGiant, count=3, icon=HillGiant)
    Combat()
    AfterCombat()
    CastleDeedFlags |= PasswordGiants
    exit

    if AlarmEndHour == 0 { exit }
    Scratch1 = 64
    if Scratch10 == 22 { Scratch1 = 128 }
    if BaneTempleFlags has Scratch1 { goto GateRecognizesYou }
    if BaneTempleFlags has OfferingBoxStolen|CrossedSwordsStolen { goto GateCallsThief }
    Scratch10 = VMenu("A DEEP VOICE CALLS DOWN, 'WHAT IS THE PASSWORD?'", "GIVE THE PASSWORD", "FORCE THE GATE OPEN")

    GateRecognizesYou:

    GateCallsThief:
    Text("YOU HEAR A DEEP VOICE, 'THIEF! YOU STOLE FROM LORD BANE!'")
    goto BouldersThrown

    WaitReturnPressed()

    Text("A VOLLEY OF BOULDERS ARE THROWN AT YOU...")

    Text("'FROGS AND TOADS NOT SAME THING. THEY NOT SAME, THEY DIF... THEY NOT SAME! ME RIGHT?'")
    Scratch10 = Menu("YES", "NO", "ATTACK", "LEAVE")
    goto [AnswerYes, AnswerNo, FightArguingGiants][Scratch10]
    Text("AS YOU QUICKLY LEAVE THE ROOM, YOU HEAR ONE OF THE GIANTS SAY, 'AH, US RIGHT. THEM JUST AFRAID TO SAY 'CAUSE KNOW YOU SMASH 'EM.' THE ARGUMENT RESUMES.")
    goto StepBack

AnswerNo:

    Text("BEFORE HE CAN ATTACK THE OTHERS GRAB HIM AND SAY, 'YOU BETTER GO LITTLE MEN.")
    Scratch10 = Menu("STAY", "LEAVE")

AnswerYes:
    Text("AS THE GIANT TURNS IN TRIUMPH TO HIS COMPANIONS, THE OTHERS PICK UP THEIR CLUBS. 'WHAT THEY KNOW? WE KNOCK DUMB LITTLE MEN'S BRAINS OUT!'")

    if CastleFlags has FakeTyranthraxus { exit }
    Text("YOU ENTER A CHAMBER WHICH IS LAVISHLY APPOINTED WITH HANGINGS, CARPETS AND FURNITURE. SITTING ON A THRONE IS A POWERFUL LOOKING MAN CLAD IN BARBARIC SPLENDOR.")
    WaitReturnPressed()
    Portrait = HornedHelmet
    SetupMonster(Fighter16, distance=1, body=PlateArmor)
    Text("'WHAT IS YOUR BUSINESS WITH TYRANTHRAXUS, LORD OF PHLAN?'")
    Approach()
    Delay()
    EncounterMenu(Fighter16, distance=0, body=PlateArmor, result=Scratch10, combat=Attack, wait=Attack,
        flee=Mirror, advance=Attack, parlay=Talk, escape=0, strength=0)
    Portrait = None
    goto [Done, FightImpostor, FledCombat][Scratch10]
    Scratch10 = Parlay(0, 1, 2, 2, 3)
    goto [ArrogantKnave, HeavyCrown, BegoneOrWrath][Scratch10]
    Scratch10 = PartyStrength()
    Scratch10 += SelReaction
    if Scratch10 >= 85 { goto ArrogantKnave }
    Text("HE SEEMS TO COLLAPSE. 'ALL RIGHT. I'M NOT TYRANTHRAXUS. HE JUST HIRED ME TO IMPERSONATE HIM. IT'S NOT A BAD LIFE, BUT, HEY! I CAN TELL WHICH WAY THE WIND'S BLOWING.'")
    WaitReturnPressed()
    Text("'JUST LET ME GO, OKAY?")
    if YesNoMenu() != "YES" { goto FightImpostor }
    Text("HE RUNS OUT OF THE HALL. THERE IS TREASURE ALL ABOUT. ")
    CastleFlags |= FakeTyranthraxus
    if AskTakeIt() != "YES" { goto Done }
    ClearMonsters()
    Treasure(gems=8, jewelry=1)
    Combat()
    exit

ArrogantKnave:

    goto FightImpostor
