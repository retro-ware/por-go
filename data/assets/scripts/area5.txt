RollSurprise:
	PartySurprise Scratch8 Scratch9
	Scratch10 = CastleFlags & 1
	if 0 != Scratch10: Scratch9 += 1
	Surprise Scratch8 Scratch9 0 0
	if AlarmEndHour != 0: Surprise = Surprise & 1
	if Surprise == 3: Surprise = 0
	Compare Surprise 1
	Return

CheckAlarmSounding:
	if AlarmEndHour < 1: Return
	AuxRoll = Hours
	if Hours == 0 && AlarmHour == 23: AuxRoll = 24
	if AuxRoll < AlarmEndHour: Goto AlarmStillOn
	if TenMinutes < AlarmMinute: Goto AlarmStillOn
	AlarmEndHour = 0
	Compare 0 1
	Return

AlarmStillOn:
	if Search == 1: Goto SayAlarmSounding
	AuxRoll = Random 9
	if AuxRoll != 0: Goto AlarmActive

SayAlarmSounding:
	PrintClear "AN ALARM IS SOUNDING."
	Delay

AlarmActive:
	Compare 1 0
	Return

StartAlarm:
	if AlarmEndHour == 0: PrintClear "AN ALARM STARTS RINGING!"
	Delay
	AlarmHour = Hours
	AlarmMinute = TenMinutes
	AlarmEndHour = 1 + AlarmHour
	Return

AskTakeIt:
	Print "DO YOU TAKE IT?"

YesNoMenu:

	Choice = Menu "YES" "NO"
	PrintClear " "
	Compare Choice 0
	Return

	Return

	SkyInside = 9
