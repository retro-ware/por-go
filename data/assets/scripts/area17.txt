def KoboldEncounterMenu() {
    EncounterMenu(Kobold, distance=MaxDist, closeup=Kobold, result=Scratch10, combat=Attack, wait=Attack,
        flee=Attack, advance=Attack, parlay=Attack, escape=6, strength=6)
    return
}

def NomadEncounterMenu() {
    Portrait = NomadBeard
    EncounterMenu(Swordsman11, distance=MaxDist, body=WhiteRobe, result=Scratch10, combat=Attack, wait=Attack,
        flee=Attack, advance=Attack, parlay=Attack, escape=12, strength=12)
    Portrait = None
    return
}

def GetCampState() {
    if MarkFlags4 lacks KoboldRaidWon { goto CampNotHostile }
    Scratch1 = 0
    if MarkFlags4 has FoughtTribe { return }
    if MarkFlags4 has AttackedNomadAlly { Scratch1 = 1 }
    return

    CampNotHostile:
    Scratch1 = 2
    if MarkFlags4 lacks FoughtTribe { goto CampFriendlyCheck }
    goto CampDayCheck
}

CampDayCheck:
    if Hours < 18 { Scratch1 = 3 }
    return

CampFriendlyCheck:
    if MarkFlags4 lacks AttackedNomadAlly { goto CampByStage }
    if Hours < 18 { Scratch1 = 1 }
    return

    CampByStage:
    goto [StageLodging, StageGames, StageKoboldWait, CampDayCheck][CampStage]

StageLodging:
    Scratch1 = 4
    if Cell != 1 { Scratch1 = 0 }
    return

StageGames:
    Scratch1 = 6
    goto StageNightCheck

StageKoboldWait:
    Scratch1 = 7

StageNightCheck:
    if Hours > 1 { Scratch1 = 5 }
    return

def SetRestRules() {
    GetCampState()
    RestEvery = 0
    RestChance = 10
    if Scratch1 == 1 and Cell == 4 { Scratch1 = 0 }
    Scratch10 = RestChanceTable[Scratch1]
    if Scratch10 == 0 { return }
    RestEvery = 1
    RestChance = 50
    if Scratch10 == 1 { return }
    Scratch10 -= Hours
    RestEvery = Scratch10 * 12
    RestChance = 100
    return
}

def MakeAlliesNomads() {
    Scratch1 = PartyCount
    Scratch4 = NomadAllies + PartyCount
    if NomadAllies == 0 { return }

    MakeAlliesLoop:
    SelectMember(Scratch1)
    SelSide = PartyControlled
    Scratch2 = Scratch1 + 128
    SelectMember(Scratch2)
    Scratch1 = Scratch1 + 1
    if Scratch1 < Scratch4 { goto MakeAlliesLoop }
    return
}

def CheckAttackedAlly() {
    if AttackedAlly == 0 { return }
    MarkFlags4 |= AttackedNomadAlly
    MarkFlags4 &= ~FeastAccepted
    return
}

def RollNomadAllies() {
    CallLookAhead()
    NomadAllies = 0
    if MarkFlags4 has FoughtTribe { goto SizeKoboldBand }
    if Cell == 4 { goto SizeKoboldBand }
    Scratch10 = Random(6)
    NomadAllies = 1 + Scratch10
    Scratch10 = Random(6)
    Scratch10 += 1
    NomadAllies += Scratch10
    goto SizeKoboldBand
}

SizeKoboldBand:
    CallLookAhead()
    Scratch10 = PartyStrength()
    Scratch9 = 56 - NomadAllies
    if Scratch10 > Scratch9 { Scratch10 = Scratch9 }
    Scratch9 = Scratch9 / 2
    return

def RetreatOrFight() {
    if PartyX < 11 { goto FindRetreatZone }
    Text("THERE IS NOWHERE LEFT TO RETREAT. YOU MUST FIGHT.")
    PressToContinue()
    return 1 == 0

    FindRetreatZone:
    Scratch10 = 0
    goto RetreatZoneLoop
}

RetreatZoneLoop:
    Scratch9 = ZoneXMinTable[Scratch10]
    Scratch8 = ZoneYMinTable[Scratch10]
    Scratch7 = ZoneXMaxTable[Scratch10]
    Scratch6 = ZoneYMaxTable[Scratch10]
    if PartyX < Scratch9 { goto NextRetreatZone }
    if PartyY < Scratch8 { goto NextRetreatZone }
    if PartyX > Scratch7 { goto NextRetreatZone }
    if PartyY <= Scratch6 { goto RetreatToZone }

    NextRetreatZone:
    Scratch10 += 1
    if Scratch10 < 12 { goto RetreatZoneLoop }
    goto ExitToWilderness

    RetreatToZone:
    PartyX = ZoneDestXTable[Scratch10]
    PartyY = ZoneDestYTable[Scratch10]
    Facing = East
    RedrawView()
    CallRedraw()
    HearWarCries()
    return 1 == 1

def HearWarCries() {
    Text("YOU HEAR WAR CRIES AND THE CLANG OF METAL ")
    gosub [CriesLeft, CriesBehind, CriesRight, CriesAhead][Facing]
    goto FaceTheSound
}

FaceTheSound:
    if Facing == West { goto PressToContinue }
    Print(" YOU SPIN TO FACE THE SOUND.")
    PressToContinue()
    Facing = West
    CallRedraw()
    return

def CriesLeft() {
    Print("TO YOUR LEFT.")
    return
}

def CriesBehind() {
    Print("BEHIND YOU.")
    return
}

def CriesRight() {
    Print("TO YOUR RIGHT.")
    return
}

def CriesAhead() {
    Print("IN FRONT OF YOU.")
    return
}

WalkToHut:
    Scratch10 = 1
    if PartyX < 4 { goto WalkToHutStep }
    Scratch1 = PartyY * 12
    Scratch1 += PartyX
    Scratch10 = WalkPathToHut[Scratch1]

    WalkToHutStep:
    if not WalkStep() { goto WalkToHut }
    return

WalkToClearing:
    Scratch10 = 1
    if PartyX < 4 { goto WalkToClearingStep }
    Scratch1 = PartyY * 12
    Scratch1 += PartyX
    Scratch10 = WalkPathToClearing[Scratch1]

    WalkToClearingStep:
    if not WalkStep() { goto WalkToClearing }
    return

def WalkStep() {
    Clock(1)
    SetSky()
    if Scratch10 == 255 { return }
    if Scratch10 == Facing { CallStepForward() }
    Facing = Scratch10
    CallRedraw()
    Delay()
    Delay()
    return 1 != 0
}

def YesNoMenu() {
    Scratch10 = Menu("YES", "NO")
    Text(" ")
    return Scratch10
}

PressToContinue:
    Scratch9 = Menu("PRESS BUTTON OR <RETURN> TO CONTINUE")
    Text(" ")
    return

def SetSky() {
    AuxText = 11
    if Hours >= 14 { AuxText = 8 }
    Sky = AuxText
    SkyInside = 13
    return
}

def ShowNomads() {
    Portrait = NomadBeard
    SetupMonster(Swordsman11, distance=2, body=WhiteRobe)
    goto ApproachPause
}

ApproachPause:
    Delay()
    Approach()
    return

def ApproachTwice() {
    ApproachPause()
    goto ApproachHidePortrait
}

ApproachHidePortrait:
    ApproachPause()
    Portrait = None
    return

on move {
    if LeftMap == 0 { goto CheckLodgingExit }
    Text("THIS DIRECTION WILL TAKE YOU BACK INTO THE WILDERNESS. DO YOU STILL WANT TO GO THIS WAY?")
    if YesNoMenu() == "YES" { goto ExitToWilderness }
    HoldMove = SkipMove

    CheckLodgingExit:
    SetSky()
    SetRestRules()
    if MarkFlags4 has FoughtTribe { exit }
    Scratch1 = Cell & ~Inside
    if Scratch1 > 1 { exit }
    if CampStage != Lodged { exit }
    if WallAhead != 2 { exit }
    goto GuardEscort
}

on step {
    SetRestRules()
    Scratch10 = Random(3)
    if Scratch10 == 0 { goto RoomEvents }
    if Scratch1 == 1 { exit }
    if Scratch1 == 2 { exit }
    goto RoomEvents
}

RoomEvents:
    goto [Done, NomadsHostile, KoboldRaid, Done, CampOutskirts, Done, MorningGreeting, HeadmanAsks][Scratch1]

on rest {
    SetSky()
    SetRestRules()
    goto RoomEvents
}

on camp {
    SetRestRules()
    goto Done
}

Done:
    RedrawView()
    CallRedraw()
    exit

on load {
    NoCoords = 1
    SetSky()
    if AreaKind == NomadCamp { exit }
    LoadMap(NomadCamp)
    LoadWalls(Reeds, Trees, Marsh)
    PartyX = 1
    PartyY = 14
    Facing = East
    exit
}

CampOutskirts:
    if MarkFlags4 has FeastAccepted { exit }
    if Hours <= 14 { goto NomadsGather }
    Scratch10 = Random(99)
    if Scratch10 >= 50 { goto NomadsGather }
    Text("YOU TRIP ON A WIRE TO THE ACCOMPANIMENT OF BELLS TINKLING. AFTER A MOMENT THERE IS A COMMOTION AHEAD OF YOU.")
    Scratch10 = Menu("FLEE", "ADVANCE CAUTIOUSLY", "ATTACK")
    goto [ExitToWilderness, NomadsGather][Scratch10]
    Text("YOU CHARGE INTO A GROUP OF SCRUFFY LOOKING NOMADS.")
    ShowNomads()

NomadFightNoSurprise:
    Surprise = NoSurprise

NomadFight:
    Morale = 100
    MarkFlags4 |= AttackedNomadAlly
    NomadAllies = 0
    SizeKoboldBand()
    Scratch9 = Scratch9 / 2
    ClearMonsters()
    LoadMonster(Nomad, count=Scratch9, icon=Nomad)

    CheckFled()
    NomadFightCount = NomadFightCount + 1
    exit

    Portrait = None
    goto [NomadsSpotYou, NomadsSpotYou, NomadsSpotYou, HeadmanRallies][NomadFightCount]

NomadsSpotYou:
    Text("A GROUP OF NOMADS SPOT YOU. THEY SHOUT AND CHARGE. AS THEY COME CLOSER THEIR NUMBERS SWELL WITH ARRIVALS.")
    PressToContinue()
    NomadEncounterMenu()
    goto [Done, NomadFightNoSurprise, ExitToWilderness][Scratch10]

HeadmanRallies:
    Text("THE HEROIC DEATHS OF THOSE YOU HAVE JUST BATTLED HAS BOUGHT TIME FOR THE HEADMAN TO RALLY HIS PEOPLE. WITH A TREMENDOUS CRY HE LEADS HIS TRIBE INTO BATTLE.")
    PressToContinue()
    Portrait = NomadElder
    EncounterMenu(Swordsman11, distance=MaxDist, body=WhiteRobe, result=Scratch10, combat=Attack, wait=Attack,
        flee=Attack, advance=Attack, parlay=Attack, escape=12, strength=12)
    Portrait = None
    goto [Done, TribeFight, ExitToWilderness][Scratch10]

TribeFight:
    Surprise = NoSurprise
    Morale = 100
    MonsterHit = 254
    ClearMonsters()
    LoadMonster(ThLvlFighter41, count=5, icon=ThLvlFighter)
    LoadMonster(Hassad, count=1, icon=Hassad)
    LoadMonster(Shaman, count=1, icon=Shaman)
    LoadMonster(Nomad, count=49, icon=Nomad)
    Treasure(gold=5000)

TribeCombat:
    Combat()
    CheckFled()
    MarkFlags4 |= FoughtTribe

def CheckFled() {
    if CombatEnd != PartyFled { return }
    goto LeaveCamp
}

LeaveCamp:
    Mark4 = 1

ExitToWilderness:
    RedrawView()
    CallRedraw()
    Portrait = None
    if MarkFlags4 has FoughtTribe { goto MarkLeftCamp }
    Scratch10 = MarkFlags4 & FeastAccepted
    if Scratch10 != 1 { goto LoadWilderness }

    MarkLeftCamp:
    if QuestNomads != Rewarded { QuestNomads = RewardDue }

    LoadWilderness:
    NewECL(Wilderness)

    SetupMonster(Swordsman11, distance=2, body=WhiteRobe)

    ApproachPause()
    Text("ALL THE NOMADS HOLD THEIR SWORDS AT THE READY, WATCHING YOU WARILY. MORE NOMADS APPEAR FROM OUT OF TENTS IN THE CAMP.")
    ApproachPause()
    Text("YOU CAN SEE CHILDREN PEERING OUT OF THE TENT OPENINGS, ONLY TO BE PUSHED BACK INSIDE BY THEIR MOTHERS.")
    PressToContinue()
    EncounterMenu(Swordsman11, distance=0, body=WhiteRobe, result=Scratch10, combat=Attack, wait=Mirror,
        flee=Mirror, advance=Mirror, parlay=Talk,
        near="THE NOMADS WAIT FOR YOU TO MAKE THE FIRST MOVE.", escape=12, strength=12)
    Portrait = None
    goto [Done, NomadFightNoSurprise, ExitToWilderness][Scratch10]

    Text("THE BIGGEST, MEANEST, DIRTIEST NOMAD PUSHES HIS WAY TO THE FRONT OF THE CROWD. FIVE OTHER NOMADS STAND AROUND HIM AS IF TO PROTECT HIM.")
    PressToContinue()
    $AuxCounter = "TOMORROW"
    if Hours < 14 { $AuxCounter = "TONIGHT" }
    Text("THE BIGGEST NOMAD, APPARENTLY THE CHIEF OR HEADMAN, ACKNOWLEDGES YOU WITH A NOD, 'WE WILL GIVE A FEAST IN YOUR HONOR ")
    Print($AuxCounter)
    Print(". WILL YOU JOIN US?'")
    if YesNoMenu() == "YES" { goto AcceptFeast }
    Text("'THEN YOU MUST LEAVE US,' HE RESPONDS WITH AS MUCH DIGNITY AS HE CAN MUSTER. DO YOU LEAVE?")
    if YesNoMenu() == "YES" { goto LeaveCamp }
    Text("'YOUR INSULTS ARE TOO MANY! KILL THEM!'")
    Surprise = PartyIsSurprised
    goto NomadFight

    AcceptFeast:
    MarkFlags4 |= FeastAccepted
    if Hours < 14 { goto CampGames }
    Text("'YOU MUST BE TIRED FROM YOUR TRAVELS. WE WILL PROVIDE YOU WITH LODGING.' HE LEADS THE WAY TO A HUT WHICH IS HURRIEDLY BEING CLEARED OF ITS OCCUPANTS.")
    PressToContinue()
    CampStage = Lodged

    WalkToHut()
    Text("'HERE YOU MAY REST.'")
    PressToContinue()
    SetRestRules()
    RestSaveX = PartyX
    RestSaveY = PartyY
    RestSaveFacing = Facing
    Program(9)

MorningGreeting:

    Facing = East

    Text("YOU ARE TREATED AS HONORED GUESTS FOR THE REST OF THE DAY. THERE ARE CONTESTS OF STRENGTH AND SKILL IN YOUR HONOR.")

    Text("AS NIGHT FALLS, THE NOMADS BUILD A HUGE CAMPFIRE IN THE CENTER OF THE CAMP. THE WHOLE TRIBE GATHERS FOR THE FEAST. AFTER THE FEAST THERE IS DANCING AND WILD CELEBRATIONS.")
    PressToContinue()
    ShowPortrait(NomadElder, WhiteRobe)

    Text("THE HEADMAN SEEMS VERY UNEASY AS HE CONTINUES TO SPEAK. HE PRACTICALLY BEGS YOU TO STAY FOR A FEW MORE DAYS, RATHER THAN LEAVE IN THE MORNING AS YOU PLANNED.")

    PressToContinue()

    goto [LodgingRest][Scratch10]

    RedrawView()
    CallRedraw()
    Text("HOWEVER, AWAKENED BY THE TALK SOMEONE IN THE TENT HAS SEEN YOUR ACT AND A SCREAM GOES OUT 'TREACHERY! THEY HAVE SLAIN SAMIR!' THE CAMP BEGINS TO STIR.")

    MaxDist = 2
    NomadEncounterMenu()
    goto [Done, NomadFight, ExitToWilderness][Scratch10]

    Sky = 11
    ApproachTwice()
    CampStage = HeadmanRequest
    Text("THE HEADMAN COMES TO YOU AND SAYS, 'ABOUT MY REQUEST OF LAST NIGHT, WHAT IS YOUR DECISION?'")
    Scratch10 = Menu("STAY AND FIGHT", "LEAVE")

    Text("LOOKING DISAPPOINTED THE HEADMAN SAYS, 'GOOD LUCK IN YOUR TRAVELS.'")

StayAndFight:

    Text("'YOU MAY WANDER AS YOU WILL OR YOU MAY STAY HERE AND REST AND STUDY. FORGIVE ME, I MUST SEE TO MY PEOPLE.' HE LEAVES.")
    goto Done

    if Cell lacks Inside { goto RaidCries }
    PartyY -= 1
    Facing = East
    Text("THE SOUND OF WAR CRIES DRAWS YOU OUT OF THE HUT. ")
    FaceTheSound()
    goto RaidByStage

    RaidCries:
    HearWarCries()

    RaidByStage:
    if MarkFlags4 has FoughtTribe { goto HostileRaid }
    if Cell == 4 { goto HostileRaid }
    goto [KoboldHorde, KoboldsMelee, KoboldsMelee, KoboldLeader][KoboldRaidStage]

KoboldHorde:
    SetupMonster(Kobold, distance=2, closeup=Kobold)
    KoboldRaidStage = 1
    Text("A HORDE OF KOBOLDS IS ATTACKING THE CAMP. A LONE SENTRY CHARGES BUT IS QUICKLY CUT DOWN. A FAMILY EMERGES FROM A HUT AND CHARGES THE KOBOLDS.")

KoboldEncounter:
    PressToContinue()
    KoboldEncounterMenu()
    goto [Done, KoboldFightAllied][Scratch10]
    if RetreatOrFight() { goto KoboldRaid }

KoboldFightAllied:
    RollNomadAllies()

    Surprise = NoSurprise
    Morale = 80
    ClearMonsters()
    if NomadAllies == 0 { goto LoadKobolds }
    LoadMonster(Nomad, count=NomadAllies, icon=Nomad)

    LoadKobolds:
    LoadMonster(Kobold0, count=Scratch9, icon=Kobold)
    LoadMonster(Kobold123, count=Scratch9, icon=Kobold)

    Combat()
    CheckAttackedAlly()
    if CombatEnd != PartyFled { goto KoboldWaveDone }

    Text("AHEAD YOU SEE A LARGE GROUP OF KOBOLDS ENGAGED IN MELEE WITH SOME NOMADS. ")

    SetupMonster(Kobold, distance=2, closeup=Kobold)
