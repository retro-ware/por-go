	Gosub GetCampState
	Save 0 RestEvery
	Save 10 RestChance

	Return

	SetupMonster Swordsman11 distance=2 body=30
	PrintClear "JUST AS YOU THINK YOURSELVES FREE OF DANGER, THE HEADMAN, HIS SHAMAN AND GUARDS ATTACK WITH CRIES OF 'FOUL TRAITORS!' AT THE SAME TIME KOBOLDS ATTACK."
	Gosub PressToContinue
	EncounterMenu 11 MaxDist 30 Scratch10 0 0 0 0 0 "" "" "" 12 12
	Save 255 Portrait
	OnGoto Scratch10 2 Done BetrayalFight
	Gosub RetreatOrFight
	Goto KoboldRaid

BetrayalFight:
	Save 0 Surprise
	Save 100 Morale
	ClearMonsters
	LoadMonster ThLvlFighter41 count=5 icon=ThLvlFighter
	LoadMonster Hassad count=1 icon=Hassad
	LoadMonster Shaman count=1 icon=Shaman
	LoadMonster Kobold123 count=49 icon=Kobold
	Save 255 MonsterHit
	Or MarkFlags4 8 MarkFlags4
	Treasure 0 0 0 #1F40 0 0 0 255
	Goto TribeCombat
