	if 0 == Scratch10: Goto CampNotHostile
	Scratch1 = 0

	if 0 != Scratch10: Scratch1 = 1
	Return

SetRestRules:
	Gosub GetCampState
	RestEvery = 0
	RestChance = 10
	if Scratch1 == 1 && Cell == 4: Scratch1 = 0
	Scratch10 = RestChanceTable[Scratch1]
	if Scratch10 == 0: Return
	RestEvery = 1
	RestChance = 50
	if Scratch10 == 1: Return
	Scratch10 -= Hours
	RestEvery = Scratch10 * 12
	RestChance = 100
	Return

MakeAlliesNomads:
	Scratch1 = PartyCount
	Scratch4 = NomadAllies + PartyCount
	if NomadAllies == 0: Return

MakeAlliesLoop:
	LoadCharacter Scratch1
	SelSide = 128
	Scratch2 = Scratch1 + 128
	LoadCharacter Scratch2
	Scratch1 = Scratch1 + 1
	if Scratch1 < Scratch4: Goto MakeAlliesLoop
	Return

CheckAttackedAlly:
	if AttackedAlly == 0: Return

	MarkFlags4 = MarkFlags4 & 254
	Return

RollNomadAllies:
	Call CallLookAhead
	NomadAllies = 0
	Scratch10 = MarkFlags4 & 4
	if 0 != Scratch10: Goto SizeKoboldBand
	if Cell == 4: Goto SizeKoboldBand
	Scratch10 = Random 6
	NomadAllies = 1 + Scratch10
	Scratch10 = Random 6
	Scratch10 += 1
	NomadAllies += Scratch10

SizeKoboldBand:
	Call CallLookAhead
	PartyStrength Scratch10
	Scratch9 = 56 - NomadAllies
	if Scratch10 > Scratch9: Scratch10 = Scratch9
	Scratch9 = Scratch9 / 2
	Return

FindRetreatZone:

RetreatZoneLoop:
	alias ZoneYMax = Scratch6
	alias ZoneXMax = Scratch7
	alias ZoneYMin = Scratch8
	alias ZoneXMin = Scratch9
	ZoneXMin = ZoneXMinTable[Scratch10]
	ZoneYMin = ZoneYMinTable[Scratch10]
	ZoneXMax = ZoneXMaxTable[Scratch10]
	ZoneYMax = ZoneYMaxTable[Scratch10]
	if PartyX < ZoneXMin: Goto NextRetreatZone
	if PartyY < ZoneYMin: Goto NextRetreatZone
	if PartyX > ZoneXMax: Goto NextRetreatZone
	if PartyY <= ZoneYMax: Goto RetreatToZone

NextRetreatZone:
	Scratch10 += 1
	if Scratch10 < 12: Goto RetreatZoneLoop
	Goto ExitToWilderness
