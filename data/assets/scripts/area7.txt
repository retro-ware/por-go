def YesNoMenu() {
    Scratch10 = Menu("YES", "NO")
    Text(" ")
    return Scratch10
}

def PressToContinue() {
    Scratch9 = Menu("PRESS BUTTON OR RETURN TO CONTINUE")
    Text(" ")
    return
}

def SetSky() {
    SkyInside = 9
    Sky = 8
    return
}

on move {
    SetSky()
    Scratch2 = Cell & ~Inside
    for Scratch1 = 0 while Scratch1 < 8 {
        Scratch3 = Table99D8[Scratch1]
        Scratch4 = Table99E0[Scratch1]
        if Scratch2 == Scratch3 and Scratch4 == Facing { goto StairsDispatch }
    }
    exit

    StairsDispatch:
    Scratch3 = Table99E8[Scratch1]
    goto [StairsDown, StairsUp, HoldAtSecretDoor][Scratch3]
    exit # the original runs on into data at 99D8
}

on step {
    Scratch1 = Cell & ~Inside
    goto [RedrawExit, SecretDoorCheck, RedrawExit, RedrawExit, TrapDoorRoom, StatueRoom, BenchRoom,
        MerchantRoom, AudienceHall, RedrawExit, GenheerisLeaves][Scratch1]
    goto event2
}

on camp {
    RestEvery = 0
    RestChance = 10
    goto event3
}

on rest {
    exit
}

on load {
    SetSky()
    NoCoords = 1
    if TyranthraxusDefeated == 1 { goto LeaveCastle }
    if AreaKind == ValjevoCastle7 { exit }
    LoadMap(ValjevoCastle7)
    exit
}

StairsDown:
    Text("DO YOU WANT TO GO DOWN THESE STAIRS?")
    goto TakeStairsMenu

StairsUp:
    Text("DO YOU WANT TO GO UP THESE STAIRS?")

TakeStairsMenu:
    HoldMove = SkipMove
    if YesNoMenu() != "YES" { exit }
    PartyX = DestX[Scratch1]
    PartyY = DestY[Scratch1]
    if Scratch1 == 0 { goto DescendToArea5 }

RedrawExit:
    RedrawView()
    CallRedraw()
    exit

DescendToArea5:
    LoadMap(ValjevoCastle3)
    NewECL(ValjevoGrounds)

HoldAtSecretDoor:
    if SearchTowerFlags lacks SecretDoor { HoldMove = SkipMove }
    exit

SecretDoorCheck:
    if SearchTowerFlags has SecretDoor { goto SecretDoorFound }
    Scratch9 = 0

SecretDoorRoll:
    SelectMember(Scratch10)
    if SelStatus == Died { goto SecretDoorNextMember }
    Scratch9 = 1
    if Search == NotSearching { Scratch9 -= 1 }
    if SelRace == Elf { Scratch9 += 1 }
    if SelRace == HalfElf { Scratch9 += 1 }
    Scratch8 = Random(5)
    if Scratch8 <= Scratch9 { goto SecretDoorFound }

    SecretDoorNextMember:
    Scratch10 = Scratch10 + 1
    if Scratch10 < 8 { goto SecretDoorRoll }
    exit

SecretDoorFound:
    SearchTowerFlags |= SecretDoor
    Text("A SECRET DOOR IS FOUND ")
    gosub [DirFront, DirLeft, DirBehind, DirRight][Facing]
    PressToContinue()
    exit

def DirFront() {
    Print("IN FRONT OF YOU.")
    return
}

def DirLeft() {
    Print("TO YOUR LEFT.")
    return
}

def DirBehind() {
    Print("BEHIND YOU.")
    return
}

def DirRight() {
    Print("TO YOUR RIGHT.")
    return
}

TrapDoorRoom:
    Text("YOU ARE IN AN EMPTY ROOM. THE ONLY FEATURE IS A TRAP DOOR IN THE CENTER. DO YOU OPEN IT?")
    if YesNoMenu() != "YES" { exit }

    ChooseOpener:
    Who("WHO WILL OPEN IT?")
    if SelStatus != Okay { goto ChooseOpener }
    if SearchTowerFlags has MedusaKilled { goto SawStatues }
    Scratch10 = Random(19)
    Scratch10 += 1
    if Scratch10 >= SelSaveVsPetrify { goto OpenerPetrified }

    SawStatues:
    Text("QUICKLY LOOKING AWAY, ")
    Print($SelName)
    Print(" SAYS, 'I SAW BROKEN REMAINS OF STATUES... HEADS WITH FACES FULL OF TERROR.")
    goto JumpTrapDoor

    OpenerPetrified:
    SelStatus = Stoned
    Scratch1 = SelPartyIndex + 128
    Text("WITH A SCREAM OF HORROR ")
    Print($SelName)
    Print(" TURNS TO STONE.")
    SelectMember(Scratch1)

    JumpTrapDoor:
    Print(" DO YOU WANT TO JUMP THROUGH THE TRAP DOOR?")
    if YesNoMenu() != "YES" { exit }
    Text("THE PARTY JUMPS DOWN THROUGH THE TRAP DOOR...")
    PressToContinue()
    Damage(224, 3, 6, 0, 0)
    PartyX = 11
    PartyY = 5
    CallRedraw()

StatueRoom:
    if StatueRoomSeen != 0 { goto MedusaMenu }
    Text("YOU ARE IN A ROOM FILLED WITH BROKEN STATUES. A HEAD BROKEN LOOSE FROM THE TORSO SHOWS A FACE FILLED WITH TERROR.")
    PressToContinue()
    StatueRoomSeen = 1

    MedusaMenu:
    if SearchTowerFlags has MedusaKilled { exit }
    EncounterMenu(Blank, distance=1, closeup=Blank, result=Scratch10, combat=Attack, wait=CloseIn, flee=Attack,
        advance=CloseIn, parlay=Attack, near="A MEDUSA COMES RUNNING TOWARD YOU.", escape=9,
        strength=9)
    goto [RedrawExit, MedusaFight, RetreatFromMedusa][Scratch10]

MedusaFight:
    Surprise = NoSurprise
    Morale = 100
    ClearMonsters()
    LoadMonster(Medusa, count=1, icon=Medusa)
    Combat()
    if CombatEnd == PartyFled { goto RetreatFromMedusa }
    SearchTowerFlags |= MedusaKilled
    exit

RetreatFromMedusa:
    PartyX = 5
    PartyY = 7
    Facing = West
    goto RedrawExit

BenchRoom:
    if BenchRoomSeen != 0 { goto MessengerThief }
    Text("YOU ARE IN A ROOM FURNISHED WITH CHAIRS AND BENCHES.")
    BenchRoomSeen = 1

    MessengerThief:
    if GenheerisFlags has DragonDone { exit }
    if SearchTowerFlags has MessengerKilled|MessengerLeft { exit }
    Print(" WAITING HERE IS A MAN PLAYING WITH A DAGGER.")
    if BaneTempleFlags has OfferingBoxStolen|CrossedSwordsStolen { goto ThiefAttacks }
    Print(" AS YOU COME IN HE NODS AT YOU.")
    PressToContinue()
    Portrait = ShortBeard
    EncounterMenu(Swordsman11, distance=2, body=RogueDagger, result=Scratch10, combat=Attack, wait=Mirror,
        flee=Mirror, advance=Mirror, parlay=Mirror, near="HE STARES MOODILY AT HIS FEET.", escape=12,
        strength=12)
    Portrait = None
    goto [RedrawExit, ThiefFight, RetreatFromThief][Scratch10]
    Scratch10 = Parlay(1, 1, 1, 1, 0)
    goto [ThiefFight][Scratch10]
    Text("'HEY LOOK, COULD YOU KEEP QUIET? I JUST GOT BACK FROM THE NORTH AND I'M TIRED. OKAY?' A MOMENT LATER A VOICE FROM CALLS FROM THE OTHER DOORWAY, 'MESSENGER!'")
    PressToContinue()
    SearchTowerFlags |= MessengerLeft
    Text("THE MAN GETS UP WITH A SIGH AND GOES INTO THE OFFICE. A MINUTE LATER HE COMES OUT PUTTING PAPERS INTO A POUCH AND GOES OUT THE DOOR. THE VOICE CALLS, 'MESSENGER!'")
    goto RedrawExit

RetreatFromThief:
    PartyX = 3
    PartyY = 6
    Facing = East
    goto RedrawExit

ThiefAttacks:
    Text("THE THIEF SHOUTS 'BLASPHEMERS!' AND ATTACKS.")
    Delay()

ThiefFight:
    Morale = 100
    Surprise = NoSurprise
    ClearMonsters()
    LoadMonster(StLvlThief, count=1, icon=Rogue)
    Combat()
    if CombatEnd == PartyFled { goto RetreatFromThief }
    SearchTowerFlags |= MessengerKilled
    exit

MerchantRoom:
    if MerchantRoomSeen != 0 { goto GenheerisAtDesk }
    Text("YOU ENTER A ROOM FURNISHED AS THE APARTMENT OF A WEALTHY IF ECCENTRIC MERCHANT.")
    MerchantRoomSeen = 1

    GenheerisAtDesk:
    if GenheerisFlags has DragonDone { exit }
    if GenheerisFlags has GenheerisJoined|GenheerisLeft { exit }
    Print(" A MIDDLE AGED MAN IN ROBES LOOKS UP FROM A DESK.")
    PressToContinue()
    if SearchTowerFlags has MessengerKilled { goto GenheerisOffer }
    Portrait = GreyBeard
    EncounterMenu(Mage, distance=1, body=MerchantCoat, result=Scratch10, combat=Attack, wait=Talk, flee=Mirror,
        advance=Talk, parlay=Talk,
        near="'SO WHAT MESSAGE FROM TYRANTHRAXUS REQUIRES ALL OF YOU TO DELIVER IT?'", escape=0,
        strength=0)
    Portrait = None
    goto [RedrawExit, GenheerisOffer, RetreatFromThief][Scratch10]
    Scratch10 = Parlay(0, 0, 1, 1, 0)
    goto [GenheerisOffer][Scratch10]
    Text("'TELL AL-HYAM DHAZID TO EXECUTE PHASE 2 OF PLAN A. DISMISSED.'")
    Scratch10 = Menu("LEAVE", "STAY")
    goto [RetreatFromGenheeris][Scratch10]

GenheerisOffer:
    Portrait = GreyBeard
    SetupMonster(Mage, distance=0, body=MerchantCoat)
    Text("A SUDDEN LOOK OF FEAR COMES OVER HIS FACE. 'IF YOU'VE COME  FROM TYRANTHRAXUS I CAN OFFER A BETTER DEAL. IF YOU KILL HIM INSTEAD OF ME I WILL MAKE YOU ALL GENERALS!'")
    PressToContinue()
    if PartyCount == 8 { goto GenheerisDeal }
    Text("'IN FACT I WILL JOIN YOU. IS IT A DEAL?'")
    YesNoMenu()
    Portrait = None
    if (!=) { goto GenheerisFight } # YesNoMenu() != "YES"
    Text("GOOD.")
    AddNPC(Genheeris, 100)
    GenheerisFlags |= GenheerisJoined
    goto RedrawExit

    GenheerisDeal:
    Text("'IS IT A DEAL?")
    YesNoMenu()
    Portrait = None
    if (!=) { goto GenheerisFight } # YesNoMenu() != "YES"
    GenheerisFlags |= GenheerisLeft
    Text("'GOOD. I WILL NOTIFY SOME OTHERS. YOU GO KILL HIM.' HE QUICKLY GETS UP AND LEAVES THE ROOM.")
    goto RedrawExit

    GenheerisFight:
    Surprise = NoSurprise
    Morale = 100
    ClearMonsters()
    LoadMonster(Genheeris, count=1, icon=Nomad)
    Combat()
    GenheerisFlags |= GenheerisLeft
    if CombatEnd != PartyFled { exit }

RetreatFromGenheeris:
    PartyX = 2
    PartyY = 6
    Facing = South
    goto RedrawExit

GenheerisLeaves:
    if GenheerisFlags lacks GenheerisJoined { exit }
    GenheerisFlags &= ~(GenheerisJoined|GenheerisLeft)
    GenheerisFlags |= GenheerisLeft
    AuxCounter = 0

FindGenheerisLoop:
    SelectMember(AuxCounter)
    if SelStatus != Okay { goto FindGenheerisNext }
    if "GENHEERIS" != $SelName { goto FindGenheerisNext }
    SelStatus = Died
    SelName = 0
    AuxCounter += 128
    SelectMember(AuxCounter)
    exit

    FindGenheerisNext:
    AuxCounter += 1
    if AuxCounter < PartyCount { goto FindGenheerisLoop }
    exit

AudienceHall:
    if GenheerisFlags has AudienceHallSeen { goto HallGuards }
    Text("YOU ARE IN THE AUDIENCE HALL OF TYRANTHRAXUS, CONQUEROR OF PHLAN. THE CEILING HAS BEEN TORN AWAY LEAVING THE ENTIRE SECTION OF THE TOWER OPEN TO THE SKY.")
    PressToContinue()
    Text("IN THE CENTER OF THE ROOM IS A GLOWING POOL. AN ANCIENT BRONZE DRAGON, SURROUNDED BY A FIERY AURA, RISES FROM WHERE IT WAS COILED NEARBY.")
    PressToContinue()
    CallRedraw()
    GenheerisFlags |= AudienceHallSeen

    HallGuards:
    if GenheerisFlags has GuardsBeaten { goto DragonOffer }
    Portrait = GoldCapBeard
    EncounterMenu(Fighter16, distance=0, body=PlateArmor, result=Scratch10, combat=Attack, wait=Attack,
        flee=Attack, advance=Attack, parlay=Attack,
        near="TWO GUARDS MOVE TO BAR YOUR PATH. THE DRAGON LOOKS AT YOU WITH MALEVOLENCE. 'GUARDS, KILL THESE PESTS.'",
        escape=9, strength=9)
    Portrait = None
    goto [RedrawExit, GuardsFight][Scratch10]

RetreatFromHall:
    PartyX = 3
    PartyY = 6
    Facing = North
    goto RedrawExit

GuardsFight:
    Surprise = NoSurprise
    Morale = 100
    ClearMonsters()
    LoadMonster(ThLvlFighter85, count=12, icon=ThLvlFighter)
    Combat()
    if CombatEnd == PartyFled { goto RetreatFromHall }
    GenheerisFlags |= GuardsBeaten

DragonOffer:
    if GenheerisFlags has DragonDone { goto RedrawExit }
    Picture(BrownDragon)
    Text("THE DRAGON LOOKS AT YOU WITH INTEREST. 'YOU ARE INDEED MIGHTY WARRIORS. EVEN SO, THERE IS NO HOPE OF YOUR SUCCESS. BUT YOU INTEREST ME.'")
    PressToContinue()
    Text("'IF YOU JOIN ME I WILL MAKE YOU THE COMMANDERS OF THE ARMIES THAT WILL SWEEP THE MOONSEA. I CAN NOT BE DEFEATED, YOU KNOW. WILL YOU JOIN ME?'")
    PressToContinue()
    AuxCounter = 0

VoteLoop:
    SelectMember(AuxCounter)
    if SelStatus != Okay { goto VoteNext }
    if SelControl >= 128 { goto NpcVote }
    Text($SelName)
    Print(" HOW DO YOU VOTE?")
    Scratch10 = Menu("ATTACK", "JOIN TYRANTHRAXUS")
    goto [SaveVote][Scratch10]
    SelSide = MonsterSide
    SelControl = NPCMember
    goto SaveVote

    NpcVote:
    if "GENHEERIS" == $SelName { goto VoteNext }
    if SelAlignment == LawfulEvil { SelSide = MonsterSide }
    if SelAlignment == NeutralEvil { SelSide = MonsterSide }
    if SelAlignment == ChaoticEvil { SelSide = MonsterSide }

SaveVote:
    Scratch1 = 128 + AuxCounter
    SelectMember(Scratch1)

VoteNext:
    AuxCounter += 1
    if AuxCounter < PartyCount { goto VoteLoop }
    EncounterParamA = 3
    EncounterParamB = 9
    MaxDist = 1
    Surprise = NoSurprise
    Morale = 100
    ClearMonsters()
    LoadMonster(Tyranithraxus66, count=1, icon=Tyranithraxus)
    Combat()
    if CombatEnd == PartyFled { goto RetreatFromHall }
    if QuestTyranthraxus != Rewarded { QuestTyranthraxus = RewardDue }
    TowerEndFlags = GenheerisFlags | DragonDone
    TyranthraxusDefeated = 1
    Program(8)

LeaveCastle:
    Text("KNOWING THAT TYRANTHRAXUS HAS FINALLY BEEN DEFEATED, YOU LEAVE THE CASTLE. MOVING THROUGH THE STREETS YOU CATCH OCCASIONAL GLIMPSES OF MONSTERS IN HIDING. NONE ATTACK.")
    PressToContinue()
    Text("FINALLY YOU ENTER THE CIVILIZED AREA OF PHLAN.")
    Delay()
    WallSplitA = 0
    WallSplitB = 0
    PartyX = 0
    PartyY = 4
    Facing = East
    SymbolSet = 3
    TyranthraxusDefeated = 0
    NewECL(NewPhlan)
