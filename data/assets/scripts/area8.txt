on move {
    SkyColor8 = 11
    if Hours >= 14 { SkyColor8 = 8 }
    Sky = SkyColor8
    Scratch10 = ~Inside & Cell
    CheckQuestOver()
    if Scratch10 == 29 and Facing == South { goto CheckSouthDoor }
    if Scratch10 == 29 and Facing == East { goto CheckEastDoor }
    exit
}

WayBlocked:
    if HallGuardsFought != 0 { exit }
    Text("THE WAY IS BLOCKED.")
    HoldMove = SkipMove
    goto CouncilGuard

CheckSouthDoor:
    if ZhentilMission < Summoned { goto WayBlocked }
    exit

CheckEastDoor:
    if CadornaCommission < Offered { goto WayBlocked }
    exit

on step {
    Text("")
    ClearPicture()
    Scratch10 = ~Inside & Cell
    CheckQuestOver()
    CellEventIndex = 26
    goto FindCellEvent

    FindCellEvent:
    if CellEventIndex == Scratch10 { goto CellEvents }
    if CellEventIndex > 34 { exit }
    CellEventIndex += 1
    goto FindCellEvent
}

on camp {
    CheckQuestOver()
    if FoughtWatch != 0 { goto NoRestHere }
    RestEvery = 1
    RestChance = 101
    exit
}

NoRestHere:
    RestEvery = 0
    RestChance = 0
    exit

on rest {
    RestChance = 0
    goto CouncilGuard
}

on load {
    LoadMap(NewPhlan, icons=Overland)
    LoadWalls(Reset, Reset, Reset)
    if RestChance == 255 { exit }
    goto NoRestHere
}

CellEvents:
    CellEventIndex -= 26
    gosub [ExitToArea0, EnterCityHall, OutsideClerkOffice, ClerkOffice, CadornaChambers,
        CadornaZhentilMission, UrslingenRoom, CouncilChambers, GuardPost][CellEventIndex]
    ClearPicture()
    exit

ExitToArea0:
    NewECL(NewPhlan)

def EnterCityHall() {
    CheckQuestOver()
    if HallEntered > 0 { exit }
    HallEntered += 1
    Delay()
    Facing = South
    CallRedraw()
    if IronBoxState == Opened { goto CadornaConfronts }
    Text("YOU ENTER THE CITY HALL.  AS YOU FACE DOWN THE HALLWAY, YOU SEE A GUARDED DOOR.")
    return

    CadornaConfronts:
    QuestCadornaPayment = Rewarded
    IronBoxState = HandedIn
    if QuestCadornaTraitor >= RewardDue { exit }
    ShowPortrait(GoldCirclet, MerchantCoat)
    Text("COUNCILMAN CADORNA CONFRONTS YOU. 'YOU HAVE TAKEN MY FAMILY TREASURE. I DON'T FORGET SUCH THINGS.' HE STRIDES OFF.")
    ClearPicture()
    exit
}

def ClerkOffice() {
    CheckQuestOver()
    if ClerkTalkedThisVisit != 0 { exit }
    ShowClerk()
    if ClerkMet > 0 { goto BackSoSoon }
    ClerkMet = 1
    ClerkTalkedThisVisit = 1
    Text("AT YOUR ENTRY, THE COUNCIL CLERK BEGINS LOOKING THROUGH A STACK OF PAPERS.  'BEFORE I CAN OFFER ANY COMMISSIONS, I MUST SEE IF YOU ARE DUE A CURRENT REWARD.")
    PressToContinue()
    Scratch1 = 0
    RewardKind = NoReward
    Scratch7 = 255
    Scratch4 = 0
    Scratch5 = 0
    Scratch6 = 0
    goto UndeadRewardLoop

    UndeadRewardLoop:
    Scratch2 = UndeadKillTable[Scratch1]
    Scratch8 = UndeadPaidTable[Scratch1]
    Scratch9 = Scratch2 - Scratch8
    if Scratch9 > 0 { RewardKind = UndeadBounty }
    gosub [SetPlatinum, AddGems, AddJewelry, AddItems, AddGems, AddJewelry, AddItems][Scratch1]
    UndeadPaidTable[Scratch1] = Scratch2
    Scratch1 += 1
    if Scratch1 < 7 { goto UndeadRewardLoop }
    if RewardKind != UndeadBounty { goto BivantPenalty }
    Text("'WE HAVE BEEN INFORMED YOU HAVE ELIMINATED SOME UNDEAD FROM THE GRAVEYARD.")
    goto GiveReward
}

BivantPenalty:
    CheckBivantBungled()
    goto ClearingRewards

ScanQuests:
    Scratch1 = 0
    RewardKind = NoReward
    goto ScanQuestLoop

ScanQuestLoop:
    Scratch7 = 255
    Scratch2 = QuestBandits[Scratch1]
    if Scratch2 == 254 { goto ClerkSpeaks }
    goto NextQuest

NextQuest:
    Scratch1 += 1
    CheckQuestOver()
    if Scratch1 > 25 { goto GraveyardNotice }
    goto ScanQuestLoop

ClerkSpeaks:
    Text("THE CLERK SPEAKS. '")
    gosub [NorrisBonus, SokalKeepBonus, TempleAreaCleared, BivantRescued, DiscoursesValuable,
        DescriptionsAmuse, MapsHelp, HistoriesUseful, RecordsInsight, SmallValue, PodalPlazaSuccess,
        GraveyardPrize, KovelThievesGone, StojonowCleared, SwiftReturn, LizardmenGone, KoboldsGone,
        NomadCampNews, CadornaPaid, GateTaken, QuestOver, SlumsCleared, NoRewardText,
        TempleOfBaneLoot, CadornaTraitorNotice, CadornaDeadNotice][Scratch1]
    if Scratch1 > 22 { goto MarkQuestPaid }
    goto LoadReward

ClearingRewards:
    Scratch1 = LibraryEventFlags & SpectreBeaten|ReptileBeaten
    if Scratch1 != 96 { goto CheckPodalPlaza }
    if LibraryRewardGiven == Yes { goto CheckPodalPlaza }
    $AuxText = "LIBRARY.'"
    LibraryRewardGiven = Yes
    goto ClearingReward

    CheckPodalPlaza:
    if PodalPlazaCleared != RewardDue { goto CheckTextileHouse }
    $AuxText = "PODAL PLAZA.'"
    PodalPlazaCleared = Rewarded
    goto ClearingReward

    CheckTextileHouse:
    if TextileHouseCleared != RewardDue { goto ToScanQuests }
    $AuxText = "CADORNA TEXTILE HOUSE.'"
    TextileHouseCleared = Rewarded
    goto ClearingReward

    ClearingReward:
    Text("'THERE IS A REWARD FOR CLEARING THE ")
    Print($AuxText)
    Scratch1 = 10
    RewardKind = ClearedAreaReward
    goto LoadReward

    ToScanQuests:
    goto ScanQuests

def CheckBivantBungled() {
    if QuestBivant != Bungled { return }
    Text("'BY THE WAY, YOUR BUNGLING THE BIVANT RESCUE HAS COST US.'")
    PressToContinue()
    QuestBivant = Rewarded
    return
}

LoadReward:
    Scratch3 = RewardTable1[Scratch1]
    Scratch4 = RewardTable2[Scratch1]
    Scratch5 = RewardTable3[Scratch1]
    Scratch6 = RewardTable4[Scratch1]
    goto GiveReward

GiveReward:
    Print(" HERE IS YOUR REWARD.'")
    AuxCounter = Menu("PRESS <RETURN> OR BUTTON TO CONTINUE")
    FoughtWatch = 0
    ClearMonsters()
    Treasure(gold=Scratch3, platinum=Scratch4, gems=Scratch5, jewelry=Scratch6, items=Scratch7)
    Combat()
    if Scratch1 != 20 { ShowClerk() }
    if RewardKind == UndeadBounty {
        goto BivantPenalty
    } else if RewardKind > UndeadBounty {
        goto ClearingRewards
    }
    goto MarkQuestPaid

MarkQuestPaid:
    QuestBandits[Scratch1] = 255
    if Scratch1 != 20 { goto NextQuest }
    ClearPicture()
    CheckQuestOver()
    exit

    if QuestGraveyard == Rewarded { goto OfferCommissions }
    Scratch1 = PartyStrength()
    if Scratch1 < 19 { goto OfferCommissions }
    if GraveyardNoticeGiven == Yes { goto OfferCommissions }
    Text("PUTTING DOWN THE PAPERS, SHE SAYS, 'I MUST BRING TO YOUR ATTENTION THE FOLLOWING CONCERNING VALHINGEN GRAVEYARD: '")
    PressToContinue()
    if Scratch1 > 36 { goto GraveyardDanger }
    Text("THE UNDEAD ARE CROSSING THE RIVER IN HORDES AND KILLING OR CARRYING OFF OUR SETTLERS.  THE COUNCIL DEMANDS THESE ATTACKS BE STOPPED. A GENEROUS REWARD IS OFFERED.'")
    goto GraveyardContinue

        OfferBivant, OfferZhentilMission, OfferUrslingen, OfferCouncilMeeting][Scratch1]
    OffersListed += 1
    PressToContinue()
    goto NextOffer

NextOffer:
    Scratch1 += 1
    goto OfferLoop
