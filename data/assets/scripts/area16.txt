def SetSky() {
    SkyInside = 8
    Scratch10 = 11
    if Hours > 13 { Scratch10 = 8 }
    Sky = Scratch10
    return
}

def FindPoolCell() {
    Scratch10 = 0
    Scratch7 = Cell & ~Inside

    FindPoolCellLoop:
    Scratch9 = PoolCellTable[Scratch10]
    if Scratch9 == Scratch7 { return }
    Scratch10 += 1
    if Scratch10 < 13 { goto FindPoolCellLoop }
    Compare(1, 0)
    return
}

def CapLizardCounts() {
    Scratch8 = 0
    if LizardMenKilled < 50 { Scratch8 = 50 - LizardMenKilled }
    Scratch5 = Table9A09[Scratch7]
    if Scratch5 > Scratch8 { Scratch5 = Scratch8 }
    Scratch8 = 0
    if GiantLizardsKilled < 10 { Scratch8 = 10 - PoolTreasureTaken }
    Scratch4 = Table9A0E[Scratch7]
    if Scratch4 > Scratch8 { Scratch4 = Scratch8 }
    return
}

def CountLizardKills() {
    Scratch1 = Table9A09[Scratch7]
    Scratch6 = LizardMenKilled
    AddKills()
    LizardMenKilled = Scratch6
    Scratch1 = Table9A0E[Scratch7]
    Scratch6 = GiantLizardsKilled
    AddKills()
    GiantLizardsKilled = Scratch6
    return
}

def AddKills() {
    if Scratch1 < Beaten { goto AddKillsCapped }
    Scratch1 = Beaten

    AddKillsCapped:
    Beaten -= Scratch1
    Scratch6 += Scratch1
    return
}

    Scratch9 = Menu("PRESS <RETURN> OR BUTTON TO CONTINUE")

on move {
    SetSky()
    Scratch10 = 0
    Scratch7 = Cell & ~Inside

    BlockScanLoop:
    Scratch9 = Table9B00[Scratch10]
    Scratch8 = Table9B0C[Scratch10]
    if Scratch9 == Scratch7 and Scratch8 == Facing { goto BlockDispatch }
    Scratch10 = Scratch10 + 1
    if Scratch10 < 12 { goto BlockScanLoop }
    exit

    BlockDispatch:
    Scratch2 = Table9B18[Scratch10]
    Scratch1 = Table9B24[Scratch10]
    goto [MoatBlocks, LeaveRuins, TunnelSlopesUp, StairsUp][Scratch2]
    exit # the original runs on into data at 9B00
}

on step {
    Scratch10 = 0
    Scratch7 = Cell & ~Inside

    EventScanLoop:
    Scratch9 = Table9B96[Scratch10]
    if Scratch9 == Scratch7 { goto EventDispatch }
    Scratch10 = Scratch10 + 1
    if Scratch10 < 22 { goto EventScanLoop }
    exit

    EventDispatch:
    Scratch2 = Table9BAC[Scratch10]
    Scratch1 = Table9BC2[Scratch10]
    goto [RuinedBuilding, SearchForTunnel, CentralKeep, LShapedBuilding, FindTreasure, PoolLizardmen][Scratch2]
    exit # the original runs on into data at 9B96
}
