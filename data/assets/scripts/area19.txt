on move {
    exit
}

on step {
    Picture(Cave)
    Scratch2 = DragonTalkFlags & DragonAttacked
    if Scratch2 != 1 { goto CheckCaveState }
} # falls into EmptyCaveLeave

EmptyCaveLeave:
    Text("YOU ENTER THE ENORMOUS CAVE, SEARCH IT AND FIND NOTHING OF INTREST.  YOU THEN LEAVE.")
    NewECL(Mountains)

CheckCaveState:
    if Scratch2 = DragonTalkFlags & EfreetTold; Scratch2 != 0 { goto EmptyCaveLeave }
    Scratch2 = DragonTalkFlags & DragonSpoke
    if Scratch2 != 1 { goto MagicMouthAsks }
    if Carried(EfreetiBottle) { goto BottleRecovered }

    MagicMouthAsks:
    Text("AS YOU ARE ENTERING THE CAVE, A MAGIC MOUTH APPEARS ON ONE OF THE WALLS.  IN A BOOMING VOICE IT SAYS, 'WHO ENTERS THE HOME OF DIOGENES?  SPEAK NOW AND DECLARE YOUR INTENT!'")
    Scratch1 = Menu("STATE NAME AND INTENT", "LEAVE")
    goto [StateIntentMenu, LeaveCave][Scratch1]

StateIntentMenu:
    Text("WHAT WILL YOU SAY IS YOUR REASON FOR BEING HERE?")
    WaitReturnPressed()
    Scratch1 = VMenu("DRAGON'S CAVE", "ASK DRAGON FOR ADVICE", "CHALLENGE EVIL LEADER OF OLD PHLAN",
        "PLEDGE PARTY TO DRAGON'S SERVICE", "OFFER A GIFT")
    goto [AskAdvice, ChallengeDragon, PledgeService, OfferGift][Scratch1]

PledgeService:
    Picture(GreyDragon)
    Text("YOU HEAR A GREAT FLAPPING SOUND.  A STRONG WHOOSH OF AIR COMES DOWN UPON THE ADVENTURERS.  A LARGE SILVER DRAGON STANDS BEFORE YOU.  IN A BOOMING VOICE IT SAYS...")
    WaitReturnPressed()
    Text("'I ACCEPT YOUR PLEDGE.  FOR YOUR FIRST DUTY, I COMMAND YOU TO RECOVER A POWERFUL MAGICAL BOTTLE FROM A KOBOLD SETTLEMENT IN THE HILLS FAR TO THE EAST.  THEN RETURN TO ME AND I SHALL GIVE YOU ANOTHER TASK.'")

DragonFliesOff:
    WaitReturnPressed()
    Text("AFTER SPEAKING, THE DRAGON FLYS THROUGH AN OPENNING IN THE CEILING OF THE CAVE.  YOU EXIT THE CAVE TO CONTINUE YOUR QUEST.")
    DragonTalkFlags |= DragonSpoke
    goto LeaveCave

ChallengeDragon:
    Picture(GreyDragon)
    Text("YOU HEAR A ROAR THEN A GREAT FLAPPING SOUND.  A STRONG RUSH OF AIR BATTERS THE PARTY AS A LARGE SILVER DRAGON LANDS BEFORE THE ADVENTURERS.  IN A BOOMING VOICE IT SAYS...")
    WaitReturnPressed()
    Text("'YOU FOOLS.  I AM A GOOD SILVER DRAGON NOT THE EVIL LEADER OF PHLAN.  DON'T YOU KNOW THAT WE METALLIC DRAGONS ARE GOOD CREATURES?'  WHAT WILL YOU DO?")
    Scratch1 = Menu("APOLOGIZE", "SLAY DRAGON")
    goto [ApologizeToDragon, DragonFleesUp][Scratch1]

ApologizeToDragon:
    Text("AFTER HEARING YOUR APOLOGY THE DRAGON SAYS, 'THOUGH YOU ARE FOOLISH I BELIEVE YOUR ACTS ARE NOBLE.  I WILL HELP YOU WITH YOUR QUEST TO CLEANSE THE CITY OF PHLAN FROM EVIL.'")
    WaitReturnPressed()
    Text("FAR TO THE EAST THERE IS A LARGE KOBOLD COMMUNITY.  THEY HAVE POSSESION OF A POWERFUL SILVER BOTTLE.  THIS BOTTLE WILL HELP YOU RID THE CITY OF ITS UNDEAD PROBLEM.  GET THE BOTTLE THEN RETURN TO ME.")
    goto DragonFliesOff

DragonFleesUp:
    Text("AS YOU READY YOUR WEAPONS TO ATTACK.  THE DRAGON TAKES FLIGHT AND ROARS, 'EVIL CREATURES, BEGONE FROM MY LAIR.'  HE THEN FLIES UP THROUGH A HOLE IN THE CEILING.")
    WaitReturnPressed()
    Text("YOU MAY EASILY FOLLOW THE DRAGON BY CLIMBING UP A WALL AND THEN THROUGH THE HOLE IN THE CEILING.  WHAT WILL YOU DO?")
    Scratch1 = Menu("FOLLOW DRAGON", "LEAVE")
    goto [FollowDragonUp, LeaveCave][Scratch1]

FollowDragonUp:
    Text("YOU CLIMB TO THE SECOND LEVEL OF THE DRAGON'S DEN.  THERE IS THE DRAGON WAITING FOR YOU.  HE LOOKS EXTREMELY MAD!  WHAT WILL YOU DO?")
    Scratch1 = Menu("ATTACK", "LEAVE")
    goto [AttackDragonFireball, LeaveCave][Scratch1]

AttackDragonFireball:
    Text("YOU TRY TO ATTACK THE DRAGON BUT SEEM TO BE FROZEN IN TIME.  THE DRAGON THROWS A FIREBALL IN THE MIDDLE OF THE PARTY AND THEN DISAPPEARS.  YOU HAVE A STRANGE FEELING THAT YOU JUST MADE A BIG MISTAKE.")
    Damage(192, 11, 6, 0, 4)
    Text("AFTER RECOVERING FROM THE FIREBALL YOU SEARCH THE REST OF THE LAIR AND FIND NOTHING OF INTREST.")
    DragonTalkFlags |= DragonAttacked
    goto LeaveCave

AskAdvice:
    Picture(GreyDragon)
    Text("THE CHARACTERS HEAR A LOUD FLAPPING NOISE AND THEN A WARM RUSH OF AIR HITS THEIR FACES.  STANDING BEFORE YOU NOW IS A GREAT SILVER DRAGON.  HE SAYS, 'YOU ARE WISE TO SEEK MY COUNCIL AND I CAN HELP YOU...'")
    WaitReturnPressed()
    Text("'FAR TO THE EAST THERE IS A LARGE KOBOLD COMPLEX.  THERE IS A SILVER BOTTLE OF GREAT POWER THERE THAT WILL HELP YOU ON YOUR QUEST.  RETRIEVE THAT BOTTLE THEN RETURN AND I WILL SEND YOU ON ANOTHER QUEST.'")
    goto DragonFliesOff

OfferGift:
    Picture(GreyDragon)
    Text("THE CHARACTERS HEAR A LOUD FLAPPING NOISE AND THEN FEEL A WARM JET OF AIR HIT THEIR FACES.  A HUGE SILVER DRAGON FLIES DOWN TO MEET THEM.")
    WaitReturnPressed()
    Text("THE DRAGON LANDS, LOOKS THE PARTY OVER THEN SAYS 'YOU HAVE NOTHING THAT INTRESTS ME.  IF YOU WISH TO APEASE ME THEN GO EAST AND DESTROY THE KOBOLD COMMUNITY THERE.'")
    WaitReturnPressed()
    Text("'THERE YOU SHALL FIND A MAGICAL BOTTLE WHICH WILL HELP YOU IN CLEANSING OLD PHLAN OF ITS EVIL!'")
    goto DragonFliesOff

BottleRecovered:
    Text("AS YOU ENTER THE CAVE YOU ARE GREETED BY DIOGENES, THE SILVER DRAGON.  HE SAYS 'I SEE YOU HAVE RECOVERED THE SILVER BOTTLE FROM THE KOBOLD CAVES.  I CONGRADULATE YOU.'")
    WaitReturnPressed()
    Text("'THAT BOTTLE CONTAINS A POWERFUL EFREET WHO WAS SUMMONED TO KILL VAMPIRES.  HE WILL BE A VALUABLE ALLY IN OLD PHLAN.  IN VALHINGEN GRAVEYARD THERE IS A VAMPIRE.'")
    WaitReturnPressed()
    Text("'WHEN YOU MEET THE VAMPIRE USE THE BOTTLE AND YOU MAY HAVE A CHANCE IN DEFEATING HIM.  I WISH YOU LUCK.'  AFTER SPEAKING DIOGENES FLIES THROUGH A HOLE IN THE CEILING AND DISSAPPEARS.")
    DragonTalkFlags |= EfreetTold
    goto LeaveCave

LeaveCave:
    NewECL(Mountains)

def WaitReturnPressed() {
    Scratch1 = Menu("PRESS <RETURN> OR BUTTON TO CONTINUE")
    return
}

on camp {
    RestEvery = 0
    SelMemorized13 = 1
    exit
}

on rest {
    exit
}

on load {
    In3D = Dungeon3D
    exit
}
