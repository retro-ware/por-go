event0:
	Exit

event1:
	Picture Cave
	Scratch2 = DragonCaveFlags & 128
	if Scratch2 != 1: Goto CheckCaveState

EmptyCaveLeave:
	PrintClear "YOU ENTER THE ENORMOUS CAVE, SEARCH IT AND FIND NOTHING OF INTREST.  YOU THEN LEAVE."
	NewECL Mountains

CheckCaveState:
	Scratch2 = DragonCaveFlags & 2
	if 0 != Scratch2: Goto EmptyCaveLeave
	Scratch2 = DragonCaveFlags & 1
	if Scratch2 != 1: Goto MagicMouthAsks
	if Carried 82: Goto BottleRecovered

MagicMouthAsks:
	alias Choice = Scratch1
	PrintClear "AS YOU ARE ENTERING THE CAVE, A MAGIC MOUTH APPEARS ON ONE OF THE WALLS.  IN A BOOMING VOICE IT SAYS, 'WHO ENTERS THE HOME OF DIOGENES?  SPEAK NOW AND DECLARE YOUR INTENT!'"
	Choice = Menu "STATE NAME AND INTENT" -> StateIntentMenu "LEAVE" -> LeaveCave

StateIntentMenu:
	alias Choice = Scratch1
	PrintClear "WHAT WILL YOU SAY IS YOUR REASON FOR BEING HERE?"
	Gosub WaitReturnPressed
	Choice = VMenu "DRAGON'S CAVE" "ASK DRAGON FOR ADVICE" "CHALLENGE EVIL LEADER OF OLD PHLAN" "PLEDGE PARTY TO DRAGON'S SERVICE" "OFFER A GIFT"
	OnGoto Choice AskAdvice ChallengeDragon PledgeService OfferGift

PledgeService:
	Picture GreyDragon
	PrintClear "YOU HEAR A GREAT FLAPPING SOUND.  A STRONG WHOOSH OF AIR COMES DOWN UPON THE ADVENTURERS.  A LARGE SILVER DRAGON STANDS BEFORE YOU.  IN A BOOMING VOICE IT SAYS..."
	Gosub WaitReturnPressed
	PrintClear "'I ACCEPT YOUR PLEDGE.  FOR YOUR FIRST DUTY, I COMMAND YOU TO RECOVER A POWERFUL MAGICAL BOTTLE FROM A KOBOLD SETTLEMENT IN THE HILLS FAR TO THE EAST.  THEN RETURN TO ME AND I SHALL GIVE YOU ANOTHER TASK.'"

DragonFliesOff:
	Gosub WaitReturnPressed
	PrintClear "AFTER SPEAKING, THE DRAGON FLYS THROUGH AN OPENNING IN THE CEILING OF THE CAVE.  YOU EXIT THE CAVE TO CONTINUE YOUR QUEST."
	DragonCaveFlags |= 1
	Goto LeaveCave

ChallengeDragon:
	alias Choice = Scratch1
	Picture GreyDragon
	PrintClear "YOU HEAR A ROAR THEN A GREAT FLAPPING SOUND.  A STRONG RUSH OF AIR BATTERS THE PARTY AS A LARGE SILVER DRAGON LANDS BEFORE THE ADVENTURERS.  IN A BOOMING VOICE IT SAYS..."
	Gosub WaitReturnPressed
	PrintClear "'YOU FOOLS.  I AM A GOOD SILVER DRAGON NOT THE EVIL LEADER OF PHLAN.  DON'T YOU KNOW THAT WE METALLIC DRAGONS ARE GOOD CREATURES?'  WHAT WILL YOU DO?"
	Choice = Menu "APOLOGIZE" -> ApologizeToDragon "SLAY DRAGON" -> DragonFleesUp

ApologizeToDragon:
	PrintClear "AFTER HEARING YOUR APOLOGY THE DRAGON SAYS, 'THOUGH YOU ARE FOOLISH I BELIEVE YOUR ACTS ARE NOBLE.  I WILL HELP YOU WITH YOUR QUEST TO CLEANSE THE CITY OF PHLAN FROM EVIL.'"
	Gosub WaitReturnPressed
	PrintClear "FAR TO THE EAST THERE IS A LARGE KOBOLD COMMUNITY.  THEY HAVE POSSESION OF A POWERFUL SILVER BOTTLE.  THIS BOTTLE WILL HELP YOU RID THE CITY OF ITS UNDEAD PROBLEM.  GET THE BOTTLE THEN RETURN TO ME."
	Goto DragonFliesOff

DragonFleesUp:
	alias Choice = Scratch1
	PrintClear "AS YOU READY YOUR WEAPONS TO ATTACK.  THE DRAGON TAKES FLIGHT AND ROARS, 'EVIL CREATURES, BEGONE FROM MY LAIR.'  HE THEN FLIES UP THROUGH A HOLE IN THE CEILING."
	Gosub WaitReturnPressed
	PrintClear "YOU MAY EASILY FOLLOW THE DRAGON BY CLIMBING UP A WALL AND THEN THROUGH THE HOLE IN THE CEILING.  WHAT WILL YOU DO?"
	Choice = Menu "FOLLOW DRAGON" -> FollowDragonUp "LEAVE" -> LeaveCave

FollowDragonUp:
	alias Choice = Scratch1
	PrintClear "YOU CLIMB TO THE SECOND LEVEL OF THE DRAGON'S DEN.  THERE IS THE DRAGON WAITING FOR YOU.  HE LOOKS EXTREMELY MAD!  WHAT WILL YOU DO?"
	Choice = Menu "ATTACK" -> AttackDragonFireball "LEAVE" -> LeaveCave

AttackDragonFireball:
	PrintClear "YOU TRY TO ATTACK THE DRAGON BUT SEEM TO BE FROZEN IN TIME.  THE DRAGON THROWS A FIREBALL IN THE MIDDLE OF THE PARTY AND THEN DISAPPEARS.  YOU HAVE A STRANGE FEELING THAT YOU JUST MADE A BIG MISTAKE."
	Damage 192 11 6 0 4
	PrintClear "AFTER RECOVERING FROM THE FIREBALL YOU SEARCH THE REST OF THE LAIR AND FIND NOTHING OF INTREST."
	DragonCaveFlags |= 128
	Goto LeaveCave

AskAdvice:
	Picture GreyDragon
	PrintClear "THE CHARACTERS HEAR A LOUD FLAPPING NOISE AND THEN A WARM RUSH OF AIR HITS THEIR FACES.  STANDING BEFORE YOU NOW IS A GREAT SILVER DRAGON.  HE SAYS, 'YOU ARE WISE TO SEEK MY COUNCIL AND I CAN HELP YOU...'"
	Gosub WaitReturnPressed
	PrintClear "'FAR TO THE EAST THERE IS A LARGE KOBOLD COMPLEX.  THERE IS A SILVER BOTTLE OF GREAT POWER THERE THAT WILL HELP YOU ON YOUR QUEST.  RETRIEVE THAT BOTTLE THEN RETURN AND I WILL SEND YOU ON ANOTHER QUEST.'"
	Goto DragonFliesOff

OfferGift:
	Picture GreyDragon
	PrintClear "THE CHARACTERS HEAR A LOUD FLAPPING NOISE AND THEN FEEL A WARM JET OF AIR HIT THEIR FACES.  A HUGE SILVER DRAGON FLIES DOWN TO MEET THEM."
	Gosub WaitReturnPressed
	PrintClear "THE DRAGON LANDS, LOOKS THE PARTY OVER THEN SAYS 'YOU HAVE NOTHING THAT INTRESTS ME.  IF YOU WISH TO APEASE ME THEN GO EAST AND DESTROY THE KOBOLD COMMUNITY THERE.'"
	Gosub WaitReturnPressed
	PrintClear "'THERE YOU SHALL FIND A MAGICAL BOTTLE WHICH WILL HELP YOU IN CLEANSING OLD PHLAN OF ITS EVIL!'"
	Goto DragonFliesOff

BottleRecovered:
	PrintClear "AS YOU ENTER THE CAVE YOU ARE GREETED BY DIOGENES, THE SILVER DRAGON.  HE SAYS 'I SEE YOU HAVE RECOVERED THE SILVER BOTTLE FROM THE KOBOLD CAVES.  I CONGRADULATE YOU.'"
	Gosub WaitReturnPressed
	PrintClear "'THAT BOTTLE CONTAINS A POWERFUL EFREET WHO WAS SUMMONED TO KILL VAMPIRES.  HE WILL BE A VALUABLE ALLY IN OLD PHLAN.  IN VALHINGEN GRAVEYARD THERE IS A VAMPIRE.'"
	Gosub WaitReturnPressed
	PrintClear "'WHEN YOU MEET THE VAMPIRE USE THE BOTTLE AND YOU MAY HAVE A CHANCE IN DEFEATING HIM.  I WISH YOU LUCK.'  AFTER SPEAKING DIOGENES FLIES THROUGH A HOLE IN THE CEILING AND DISSAPPEARS."
	DragonCaveFlags |= 2
	Goto LeaveCave

LeaveCave:
	NewECL Mountains

WaitReturnPressed:
	alias Choice = Scratch1
	Choice = Menu "PRESS <RETURN> OR BUTTON TO CONTINUE"
	Return

event2:
	RestEvery = 0
	SelMemorized13 = 1
	Exit

event3:
	Exit

event4:
	In3D = Dungeon3D
	Exit
