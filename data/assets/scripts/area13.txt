def CheckTrap() {
    CheckParty(0, 19, Scratch1, Scratch1, Scratch1, Scratch1)
    if Scratch1 == 1 { goto TrapFound }
    if Search != Searching { goto TrapNotDisarmed }
    AuxMember = 0
} # falls into MemberFindsTrap

MemberFindsTrap:
    SelectMember(AuxMember)
    if SelStatus != Okay { goto NextTrapFinder }
    AuxRoll = Random(99)
    if SelFindTraps > AuxRoll { goto TrapFound }

    NextTrapFinder:
    AuxMember += 1
    if AuxMember < 8 { goto MemberFindsTrap }
    goto TrapNotDisarmed

TrapFound:
    Text("YOU FIND A TRAP, BUT THE TUNNEL IS TOO NARROW TO AVOID IT.")
    WhatDoYouDo()
    AuxMember = Menu("DISARM IT", "GO ON", "LEAVE")
    goto [DisarmTrap, TrapNotDisarmed, TrapBackOff][AuxMember]

DisarmTrap:
    Who("WHO WILL DISARM THE TRAP?")
    if SelStatus != Okay { goto TrapNotDisarmed }
    Scratch4 = 128
    AuxRoll = Random(99)
    if SelFindTraps < AuxRoll { goto TrapNotDisarmed }
    Scratch10 = 1
    return

TrapNotDisarmed:
    Scratch10 = 0
    return

TrapBackOff:
    PartyX = PrevX
    PartyY = PrevY
    CallRedraw()
    exit

    gosub [GuideReturn, KoboldTapsWall, KoboldTapsWall, BoulderLever, ClamberDown][Scratch1]
    if Scratch1 == 4 { goto Done }
    Scratch6 += 1
    goto GuideStep

def WalkSteps() {
    Scratch2 = 1
    if Scratch1 == 0 {
        CallRedraw()
        Delay()
        return
    }

    WalkStepLoop:
    if Scratch2 > Scratch1 { return }
    gosub [StepNorthY, StepEastX, StepSouthY, StepWestX][Facing]
    CallRedraw()
    Delay()
    Scratch2 += 1
    goto WalkStepLoop
}
