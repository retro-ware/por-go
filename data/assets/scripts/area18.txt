on camp {
    if PatrolsBeaten >= 64 { RestSafe() }
    RestEvery = 2
    RestChance = 16
    if Cell == 132 { RestEvery = 0 }
    if Cell == 138 { RestEvery = 0 }
    if PodalPlazaCleared > 253 { RestEvery = 0 }
    exit
}

def RestSafe() {
    RestEvery = 1
    RestChance = 99
    return
}

on rest {
    RollPatrol()
    SetupMonster(EncMonster, distance=2, closeup=EncMonster)
    goto PatrolFight
}

on load {
    LoadMap(PodolPlaza)
    LoadWalls(Stone, Overgrown, RedBrick)
    NoCoords = 0
    if AreaKind == PodolPlaza { exit }
    if QuestPodalAuction != AuctionOn { goto ResetPatrols }
    PatrolsBeaten = 64
    DisguiseState = NotChosen
    exit

    ResetPatrols:
    if PodalPlazaCleared < RewardDue { PatrolsBeaten = 0 }
    DisguiseState = CoverBroken
    exit
}

on move {
    if LeftMap == 0 { goto PlazaApproach }
    CallStepForward()
    HoldMove = SkipMove
    NoCoords = 0
    goto [ExitNorth, ExitEast, ExitSouth, ExitWest][Facing]
    goto ExitNorth
}

ExitNorth:
    EnterArea(StojanowGate)

ExitEast:
    EnterArea(KutosWell)

ExitSouth:
    EnterArea(HobgoblinBarracks)

ExitWest:
    MapX = 11
    MapY = 28
    EnterArea(Wilderness)

PlazaApproach:
    if Hours < 14 {
        Sky = 11
    } else {
        Sky = 8
    }
    if DisguiseState != NotChosen { exit }
    Scratch1 = VMenu("THE PLAZA AHEAD IS CROWDED WITH MONSTERS. HOW WILL YOU PROCEDE?", "STRIDE BOLDLY FORWARD!",
        "DISGUISE PARTY AS MONSTERS.", "SNEAK, REMAINING UNSEEN")
    DisguiseState = DisguiseChoiceTable[Scratch1]
    exit

on step {
    Scratch1 = Cell & ~Inside
    goto [StreetSquare, AuctionStart, TavernEnter, ThePittSign, IlmaterTemple, BaneShrine, RobedOrc,
        AbandonedShop, DecayedBuilding, RememberBuilding, Done, Done][Scratch1]
    goto StreetSquare
}

StreetSquare:
    Scratch1 = Random(25)
    if PatrolsBeaten < 64 { goto StreetSquareEarly }
    if Scratch1 < 3 { goto OverhearRumor }
    if Scratch1 > 21 { goto PatrolMeet }
    if Search == Searching { Compare(Scratch1, 16) }
    if (>) { goto PatrolMeet } # Scratch1 > 16 if that ran, else Search > Searching
    goto Done

    StreetSquareEarly:
    if PatrolsBeaten >= 9 { goto SlumsCleared9 }
    if Scratch1 == 0 { goto PatrolMeet }
    if Search == Searching { Compare(Scratch1, 18) }
    if (>) { goto PatrolMeet } # Scratch1 > 18 if that ran, else Search > Searching
    goto Done

    PatrolMeet:
    RollPatrol()
    if DisguiseState == CoverBroken { goto PatrolEncounterMenu }
    if DisguiseState == NotChosen { goto PatrolEncounterMenu }
    SetupMonster(EncMonster, distance=2, closeup=EncMonster)
    Delay()
    SpotPatrol()
    Approach()
    Delay()
    if DisguiseState == Disguised { goto SneakPast }
    Scratch1 = Random(1)
    goto [PatrolGlare, PatrolCheckOut][Scratch1]
    goto PatrolGlare

PatrolGlare:
    Print("THEY GLARE AT YOUR PARTY, ")
    goto DisguiseCheck

PatrolCheckOut:
    Print("THEY CHECK OUT YOUR PARTY, ")
    goto DisguiseCheck

DisguiseCheck:
    Delay()
    Approach()
    Scratch10 = 5
    Scratch1 = 0
    goto DisguiseCheckMember

DisguiseCheckMember:
    SelectMember(Scratch1)
    if SelRace == Human { goto DisguiseCheckNext }
    Scratch10 += 5
    goto DisguiseCheckNext

    DisguiseCheckNext:
    Scratch1 += 1
    if Scratch1 < 8 { goto DisguiseCheckMember }
    Scratch9 = Random(100)
    if Scratch9 > Scratch10 { goto PatrolWalkOn }
    goto PatrolDemandWho

PatrolDemandWho:
    Print("AND DEMAND, 'WHO ARE YOU?'")
    Approach()
    Scratch1 = Parlay(0, 1, 2, 3, 4)
    goto [PatrolGrumble, HereForBoss, SettlersKillThem, SettlersKillThem, PatrolGrumble][Scratch1]
    goto SettlersKillThem

SettlersKillThem:
    Text("THEY YELL 'SETTLERS! KILL THEM!'")
    goto PatrolFight

HereForBoss:
    Scratch1 = Random(30)
    if SelReaction < Scratch1 { goto SettlersKillThem }
    Text("'OH, YOU'RE HERE FOR THE BOSS.'")
    goto Done

PatrolGrumble:
    Scratch1 = Random(50)
    Scratch2 = PartyStrength()
    if Scratch2 < Scratch1 { goto SettlersKillThem }
    Text("THEY GRUMBLE AND MOVE ON.")
    goto Done

PatrolWalkOn:
    Scratch1 = Random(1)
    goto [WalkOn, LaughWalkOn][Scratch1]
    goto WalkOn

WalkOn:
    Print("AND WALK ON.")
    goto Done

LaughWalkOn:
    Print("LAUGH TO THEMSELVES AND WALK ON.")
    goto Done

SneakPast:
    Scratch1 = Random(5)
    PressToContinue()
    goto [PatrolDemandWho, StopTheyMoveOn, StopTheyMoveOn, SlipPast, SlipPast, SlipPast][Scratch1]
    goto SlipPast

SlipPast:
    Text("YOUR PARTY SLIPS PAST THEM.")
    goto Done

StopTheyMoveOn:
    Text("YOU STOP, AND THEY MOVE ON.")
    goto Done

PatrolEncounterMenu:
    EncMenuMode = 2
    if PatrolsBeaten >= 64 { EncMenuMode = 0 }
    SpotPatrol()
    EncounterMenu(EncMonster, distance=EncMenuMode, closeup=EncMonster, result=Scratch1, combat=Flee,
        wait=CloseIn, flee=Attack, advance=Mirror, parlay=Talk, near="THE MONSTERS STAND READY",
        mid="THE MONSTERS PREPARE", escape=EncDistance, strength=EncDistance)
    Morale = 100
    goto [Done, PatrolFight, PartyFlees, PatrolParlay][Scratch1]
    goto PatrolFight

PatrolFight:
    LoadPatrol()
    Morale = EncMorale
    Combat()
    CountPatrolBeaten()
    if Cell < 127 { goto CoverBlown }
    if CombatEnd > PartyDied { goto PartyFlees }
    goto Done

def CountPatrolBeaten() {
    if PatrolsBeaten >= 64 { return }
    PatrolsBeaten += 1
    return
}

Done:
    RedrawView()
    Picture(None)
    exit

PartyFlees:
    Text("THE PARTY RUNS AND ESCAPES.")
    Facing = Random(3)
    Scratch1 = Random(7)
    Portrait = None
    PartyX = RandomExitX[Scratch1]
    PartyY = Scratch1
    CallRedraw()
    NoCoords = 1
    goto CoverBlown

CoverBlown:
    if DisguiseState == NotChosen { exit }
    if DisguiseState == CoverBroken { exit }
    DisguiseState = CoverBroken
    PressToContinue()
    Text("YOUR COVER IS BLOWN.")
    goto Done

PatrolParlay:
    if DisguiseState == CoverBroken and PatrolsBeaten == 64 { goto PatrolFight }
    Scratch1 = Parlay(10, 20, 20, 20, 30)
    Scratch2 = Random(50)
    Scratch1 = Scratch1 + Scratch2
    Scratch2 = SelReaction * 2
    Scratch1 = Scratch1 + Scratch2
    Scratch2 = EncParlayTable[EncType]
    Scratch1 = Scratch1 + Scratch2
    if Scratch1 < 112 { goto ParlayMoveIt }
    Text("'WE'LL LET YOU OFF THIS TIME.' THE MONSTERS SCUTTLE AWAY.")
    goto Done

    ParlayMoveIt:
    if Scratch1 < 90 { goto ParlayTeachLesson }
    Text("'MOVE IT, TROLL-BAIT!' THEY SNARL AND MOVE OFF.")
    goto Done

    ParlayTeachLesson:
    Text("'WE'RE GONNA HAVE TA TEACH YOU A LESSON!'")
    goto PatrolFight

def RollPatrol() {
    Scratch1 = Random(8)
    EncExtraCount = 0
    EncType = EncTypeTable[Scratch1]
    EncParlayMod = EncParlayTable[EncType]
    EncMonster = EncMonsterTable[EncType]
    goto [PatrolGoblins, PatrolGoblins, PatrolGoblins, PatrolGoblins, PatrolOrcs, PatrolOrcs,
        PatrolOrcs, PatrolHobgoblins, PatrolOgres][Scratch1]
    goto PatrolGoblins
}

PatrolGoblins:
    $AuxCounter = " GOBLINS"
    goto RollPatrolCount

PatrolOrcs:
    $AuxCounter = " ORCS"
    goto RollPatrolCount

PatrolHobgoblins:
    $AuxCounter = " HOBGOBLINS"
    goto RollPatrolCount

PatrolOgres:
    $AuxCounter = " OGRES"
    goto RollPatrolCount

RollPatrolCount:
    Scratch1 = PartyStrength()
    Scratch2 = EncStrengthTable[EncType]
    Scratch1 = Scratch1 / Scratch2
    Scratch1 = Scratch1 / 3
    Scratch2 = Random(Scratch1)
    Scratch1 = Scratch1 + Scratch2
    if 64 == PatrolsBeaten and CoverBroken == DisguiseState { Scratch1 = 2 * Scratch1 }
    if Scratch1 < 2 { Scratch1 = 2 }
    Scratch2 = EncMinCountTable[EncType]
    if Scratch1 < Scratch2 { goto SetPatrolCount }
    EncExtraCount = EncExtraTable[EncType]
    if Scratch1 > 50 { Scratch1 = 50 }
    goto SetPatrolCount

    SetPatrolCount:
    EncCount = Scratch1
    EncDistance = EncDistanceTable[EncType]
    return

def LoadPatrol() {
    ClearMonsters()
    Scratch1 = EncCount + EncExtraCount
    Text(Scratch1)
    Print($AuxCounter)
    Print(" RUSH TO ATTACK.")
    Scratch1 = EncMonsterTable[EncType]
    LoadMonster(Scratch1, count=EncCount, icon=Scratch1)
    EncMorale = EncMoraleTable[EncType]
    if EncExtraCount == 0 { return }
    EncMorale = EncMorale + 10
    Scratch1 += 1
    LoadMonster(Scratch1, count=EncExtraCount, icon=Scratch1)
    return
}

def SpotPatrol() {
    Text("YOU SPOT A GROUP OF")
    Print($AuxCounter)
    Print(". ")
    return
}

    if OverheardIndex == 5 { OverheardIndex -= 5 }
    if DisguiseState >= AuctionEnded { goto Done }
    RollPatrol()
    SetupMonster(EncMonster, distance=2, closeup=EncMonster)
    Delay()
    Text("AS YOU MOVE YOU OVERHEAR SOME")
    Approach()
    Delay()
    Print($AuxCounter)
    Print(" SAY, ")
    Scratch1 = OverheardIndex
    OverheardIndex += AboutOgres
    Approach()
    goto [RumorBoss, RumorOgres, RumorBaneTemple, RumorLard, RumorBidder][Scratch1]
    goto RumorBoss

    Print("'DO YOU THINK HE'LL REALLY SHOW UP -- THE BOSS I MEAN?'")
    goto Done

RumorOgres:
    Print("'MORE OGRES ON THE STREET, SINCE THE GIANTS TOOK OVER THE CASTLE.'")
    goto Done

RumorLard:
    Print("'HE MELTED INTO A PUDDLE OF LARD WHEN HE TOUCHED THE THING!'")
    goto Done

RumorBidder:
    Print("'THE BOSS WILL BE THE GUY BIDDING UP THE PRICE.'")
    goto Done

RumorBaneTemple:
    Print("'THE PLAZA'S BANE TEMPLE IS RECRUITING FOR MACE AGAIN. GRISHNAK WON'T BE HAPPY.'")
    goto Done

AuctionStart:
    NoCoords = 0
    if DisguiseState > 253 { exit }
    Picture(OldMan)
    Text("THE AUCTIONEER CRIES 'CREATURES OF ALL AGES, WELCOME TO THIS AUCTION FOR AN ITEM BOTH MAGICAL AND POWERFUL!'")
    PressToContinue()
    Text("THE AUCTIONEER HAS EITHER A WAND OR STAFF.")
    PressToContinue()
    Scratch1 = VMenu("DO YOU WISH TO:", "STAND AND LISTEN", "MOVE IN CLOSER", "LISTEN TO COMMENTS", "LEAVE")
    goto [AuctionBidding, MoveCloser, AuctionComments, AuctionLeave][Scratch1]
    goto AuctionBidding

AuctionBidding:
    Text("THE BIDDING ESCALATES TO 5,000 GOLD PIECES -- HIGH BID FROM A MAN IN PLAIN CLOTHES, NEXT TO AN OGRE.")
    PressToContinue()
    Scratch1 = VMenu("DO YOU WISH TO:", "MAKE A BID", "WAIT FOR WINNER", "TRY TO LEAVE")
    goto [MakeBid, WaitForWinner, AuctionLeave][Scratch1]
    goto MakeBid

MakeBid:
    Text("YOU UP THE BID. MONSTERS NEARBY TURN, STARE,")
    RollSuspicion()
    if CoverBroken == DisguiseState { goto AuctionCancelled }
    MonstersTurnAway()
    Text("THE MAN THEN SPEAKS A WORD AND THE AUCTIONEER'S BLOCK IS ENVELOPED IN DARKNESS.")
    PressToContinue()
    Text("THERE IS MASS CONFUSION. SUDDENLY, THE OGRE SPRINTS FROM THE DARKNESS.")
    PressToContinue()
    Scratch1 = VMenu("DO YOU:", "FOLLOW THE OGRE", "MOVE QUIETLY AWAY", "LOOK FOR THE MAN IN PLAIN CLOTHES")
    goto [FollowOgre, AuctionLeave, LookForGarwin][Scratch1]
    goto FollowOgre

FollowOgre:
    Text("THE OGRE IS TACKLED AND SEARCHED. HE DOESN'T HAVE THE ITEM ON HIM.")
    PressToContinue()
    Text("THE MONSTERS LOOK AROUND FOR A SCAPEGOAT. THEIR EYES FALL UPON YOUR PARTY, THEN THEY ")
    RollSuspicion()
    if CoverBroken == DisguiseState { goto AuctionCancelled }
    MonstersTurnAway()
    goto AuctionOver

LookForGarwin:
    Text("'GARWIN, I'LL GET YOU!' CRIES THE AUCTIONEER. HOWEVER, THE MAN HAS ESCAPED IN THE CONFUSION.")
    goto AuctionOver

WaitForWinner:
    Text("'GOING... GOING... GONE!' THE AUCTIONEER CRIES.")
    PressToContinue()
    Text("A MAN AND OGRE EXCHANGE THE WAND AND A LARGE BAG, THEN DISAPPEAR INTO THE CROWD.")
    goto AuctionOver

AuctionLeave:
    Text("AS YOU MOVE AWAY THE MONSTERS GLANCE YOUR WAY, ")
    RollSuspicion()
    if CoverBroken == DisguiseState { goto AuctionCancelled }
    MonstersTurnAway()
    Text("THE AUCTION ENDS. YOU DON'T SEE WHO BOUGHT THE ITEM.")
    goto AuctionOver

MoveCloser:
    Text("THE MONSTERS SHOOT YOU ANGRY LOOKS, ")
    RollSuspicion()
    if CoverBroken == DisguiseState { goto AuctionCancelled }
    MonstersTurnAway()
    Text("YOU NOW SEE THE WAND ISN'T A MAJOR ARTIFACT, IT'S ONLY A WAND OF FEAR!")
    goto AuctionBidding

AuctionComments:
    Scratch1 = Random(1)
    goto [CommentIllumination, CommentCommand][Scratch1]
    goto CommentIllumination

CommentIllumination:
    Text("'IT'S A WAND OF ILLUMINATION. WHO CARES, I SEE IN THE DARK ANYWAY!'")
    PressToContinue()
    goto AuctionBidding

CommentCommand:
    Text("'IT'S A STAFF OF COMMAND. I COULD HAVE MY OWN ARMY!'")
    PressToContinue()
    goto AuctionBidding

AuctionCancelled:
    Print(" AND YELL 'SETTLERS! KILL THEM!'")
    PressToContinue()
    Text("'THE AUCTION IS CANCELLED!' YELLS THE AUCTIONEER AS HE FLEES.")
    PressToContinue()
    DisguiseState = CoverBroken
    QuestPodalAuction = RewardDue
    goto PatrolEncounterMenu

AuctionOver:
    PressToContinue()
    Text("THE AUCTION OVER, THE MONSTERS GRUMBLE AND TURN AWAY.'")
    DisguiseState = AuctionEnded
    QuestPodalAuction = RewardDue
    goto Done

def RollSuspicion() {
    if DisguiseState == Disguised { DisguiseState = CoverBroken }
    Scratch2 = Random(5)
    if Scratch2 == 0 { DisguiseState = CoverBroken }
    return
}

def MonstersTurnAway() {
    Print("GRUMBLE, AND TURN AWAY.")
    PressToContinue()
    return
}

TavernEnter:
    if TavernState != FirstVisit { goto TavernCrowd }
    Text("YOU HAVE ENTERED A CROWDED TAVERN.")
    Portrait = LongHaired
    SetupMonster(ShieldBearer, distance=0, body=BareChest)
    Print("YOU OPENED THE DOOR INTO A DRUNK BUCCANEER.")
    PressToContinue()
    Text("HE GLARES AT YOU,'ONE OF YOU SHALL PAY FOR THIS INSULT. WHO WILL IT BE?'WHAT DO YOU DO?")
    goto BuccaneerMenu

BuccaneerMenu:
    Scratch1 = Menu("CHALLENGE", "ATTACK", "FLEE", "BRIBE")
    goto [ChallengeBuccaneer, PatronsAttack, PartyFlees, BribeBuccaneer][Scratch1]
    goto ChallengeBuccaneer

ChallengeBuccaneer:
    Who("WHO WILL CHALLENGE THE BUCCANEER?")
    if SelStatus > Died { goto DuelBuccaneer }
    Text("THAT ONE IS UNACCEPTABLE. CHOOSE ANOTHER.")
    goto ChallengeBuccaneer

    DuelBuccaneer:
    ClearMonsters()
    CallDuelB()
    LoadMonster(ThLvlFighter41, count=1, icon=ThLvlFighter)
    Combat()
    TavernState = BrawlBegun
    if CombatEnd > PartyDied { goto BuccaneerPaid }
    Text("THE BATTLE IS OVER, THE PATRONS RETURN TO DRINKING.")
    PressToContinue()
    Portrait = None
    goto TavernCrowd

BuccaneerPaid:
    Text("THE BUCCANEER SMILES.'THAT IS SUFFICIENT PAYMENT.' HE LEAVES.")
    PressToContinue()
    Picture(None)
    Portrait = None
    goto TavernCrowd

PatronsAttack:
    Text("YOUR ACTIONS ANGER THE PATRONS AND THEY ATTACK.")
    Morale = 30
    ClearMonsters()
    LoadMonster(Hobgoblin, count=4, icon=Hobgoblin)
    LoadMonster(GoblinGuard, count=12, icon=GoblinGuard)
    LoadMonster(ThLvlFighter41, count=1, icon=ThLvlFighter)
    LoadMonster(Orc4, count=8, icon=Orc)
    Combat()
    Portrait = None
    if CombatEnd > PartyDied { goto PartyFlees }
    TavernState = BrawlWon
    goto BarDestroyed

BribeBuccaneer:
    Text("'A TENTH OF YOUR MONEY THEN.'DO YOU ACCEPT?")
    Scratch1 = Menu("YES", "NO")
    goto [PayBribe, BuccaneerMenu][Scratch1]
    goto PayBribe

PayBribe:

    goto BuccaneerPaid

TavernCrowd:
    if TavernState > BrawlBegun { goto BarDestroyed }
    SetupMonster(Goblin, distance=0, closeup=Goblin)
    Text("THE CREATURES IGNORE YOU FOR THE MOMENT.")
    PressToContinue()
    Scratch1 = VMenu("WHAT DO YOU DO?", "MIX WITH MONSTERS", "DRINK HEAVILY", "ATTACK", "LEAVE")
    goto [MixWithMonsters, DrinkHeavily, PatronsAttack, StepBack][Scratch1]
    goto MixWithMonsters

MixWithMonsters:
    if DisguiseState == Disguised { goto PatronsAttack }
    if DisguiseState == CoverBroken { goto PatronsAttack }
    Text("MONSTERS INSULT YOUR ABILITIES AND PARENTAGE.")
    PressToContinue()
    Scratch1 = VMenu("WHAT DO YOU DO?", "ATTACK", "TRY TO LEAVE", "INSULT BACK", "JOIN IN HEAVY DRINKING")
    goto [FreeForAll, PatronsAttack, FreeForAll, DrinkHeavily][Scratch1]
    goto DrinkHeavily

DrinkHeavily:
    PartyX = 2
    PartyY = 10
    Facing = South
    CallRedraw()
    Rob(1, 100, 50)
    NoCoords = 1
    Text("YOU HAD A GREAT TIME UNTIL YOU ALL PASSED OUT. YOUR PURSE IS MISSING.")
    if PatrolsBeaten >= 64 { DisguiseState = CoverBroken }
    exit

StepBack:
    PartyX = PrevX
    PartyY = PrevY
    CallRedraw()
    exit

FreeForAll:
    Text("YOUR ACTIONS TRIGGER A FREE FOR ALL.")
    PressToContinue()
    Morale = 50
    ClearMonsters()
    LoadMonster(GoblinGuard, count=25, icon=GoblinGuard)
    LoadMonster(Hobgoblin, count=20, icon=Hobgoblin)
    LoadMonster(Ogre, count=5, icon=Ogre)
    Scratch2 = 0
    goto FreeForAllMember

    FreeForAllMember:
    Scratch1 = Random(44)
    Scratch1 += 8
    Scratch2 += 1
    SelectMember(Scratch1)
    SelSide = PartyControlled
    SelControl = NPCMember
    Scratch1 = 128 | Scratch1
    SelectMember(Scratch1)
    if Scratch2 < 21 { goto FreeForAllMember }
    Combat()
    if CombatEnd > PartyDied { goto PartyFlees }
    TavernState = BrawlWon
    goto BarDestroyed

    if TavernState > BrawlWon { exit }
    Text("THE BAR LOOKS LIKE IT WAS DESTROYED BY EXPERTS.")

    exit

ThePittSign:
    NoCoords = 0
    Text("OVER THIS DOOR IS A CRUDE SIGN SAYING, 'THE PITT.'")
    exit

IlmaterTemple:
    if LastBuildingCell != 4 { goto IlmaterHealer }
    Text("THIS IS A HIDDEN TEMPLE OF ILMATER. YOU FEEL SAFE HERE.")
    goto IlmaterHealer

    Scratch1 = Menu("YES", "NO")

    goto Heal

    ClearMonsters()
    Combat()
    goto RestHere

RestHere:
    Text("'YOU CAN REST HERE AS LONG AS YOU WISH.' SHE THEN LEAVES.")

    LastBuildingCell = 4
    exit

BaneShrine:
    NoCoords = 0
    Text("THIS IS A SMALL SHRINE. SYMBOLS OF BANE HAVE REPLACED THOSE OF ILMATER.")

RobedOrc:
    if BaneRecruiterState != NotMet { goto BaneMeeting }
    SetupMonster(Orc, distance=1, closeup=Orc)
    Text("A ROBED ORC IS SITTING HERE. WHAT DO YOU DO?")
    Scratch1 = Menu("LEAVE", "GREET", "ATTACK")
    if DisguiseState == CoverBroken { goto AttackOrc }
    if DisguiseState == Disguised { goto AttackOrc }
    goto [StepBack, GreetOrc, AttackOrc][Scratch1]
    goto GreetOrc

GreetOrc:
    Text("THE ORC SMILES CROOKEDLY.'GREETINGS FROM BANE.")
    if QuestTempleOfBane != 0 { goto RecruitRecover }
    Print(" WE ARE RECRUITING CREATURES FOR OUR TEMPLE TO THE NORTHEAST. ARE YOU INTERESTED?'")
    goto RecruitMenu

RecruitMenu:
    Scratch1 = Menu("YES", "NO")
    goto [AcceptRecruit, DeclineRecruit][Scratch1]
    goto RecruitRecover

RecruitRecover:
    Print(" WE ARE HIRING CREATURES TO RECOVER OUR TEMPLE. ARE YOU INTERESTED?'")
    goto RecruitMenu

AcceptRecruit:
    Text("'MEET HERE IN TWO HOURS. YOU MAY MEET HIGH PRIEST MACE.' HE ESCORTS YOU OUT.")
    goto SetMeetHour

SetMeetHour:
    BaneRecruiterState = MeetingSet
    BaneMeetHour = Hours
    BaneMeetHour += 2
    if BaneMeetHour > 23 { BaneMeetHour -= 24 }
    goto StepBack

DeclineRecruit:
    Text("'IF YOU CHANGE YOUR MIND, RETURN IN TWO HOURS.' HE ESCORTS YOU OUT.")
    goto SetMeetHour

BaneMeeting:
    if BaneRecruiterState != MeetingSet { exit }
    if BaneMeetHour >= Hours { exit }
    SetupMonster(Orc, distance=1, closeup=Orc)
    Text("MANY CREATURES HAVE GATHERED. THE ROBED ORC STANDS AND CALLS OUT,'FIRST, TO SHOW YOUR LOYALTIES, KILL THE UNBELIEVERS.' HE POINTS AT YOUR GROUP.")
    PressToContinue()
    Morale = 50
    ClearMonsters()
    LoadMonster(Orc4, count=8, icon=Orc)
    LoadMonster(OrcLeader5, count=1, icon=OrcLeader)
    LoadMonster(Ogre, count=3, icon=Ogre)
    Treasure(items=[LeatherHolySymbol*6])
    Combat()
    BaneRecruiterState = AmbushFought
    if CombatEnd > PartyDied { goto PartyFlees }
    exit

    Morale = 90
    ClearMonsters()
    LoadMonster(OrcLeader5, count=1, icon=OrcLeader)
    Treasure(items=[LeatherHolySymbol*6])
    Combat()
    BaneRecruiterState = AmbushFought
    if CombatEnd > PartyDied { goto PartyFlees }
    exit

RememberBuilding:
    LastBuildingCell = Cell
    exit

AbandonedShop:
    if 137 == LastBuildingCell { exit }
    Text("AN ABANDONED SHOP. BROKEN SHELVES ARE ALL THAT REMAIN.")
    goto RememberBuilding

DecayedBuilding:
    if 137 == LastBuildingCell { exit }
    Text("A DECAYED BUILDING LEFT TO THE RATS.")
    goto RememberBuilding

def PressToContinue() {
    Scratch2 = Menu("PRESS <RETURN> OR BUTTON TO CONTINUE")
    return
}

SlumsCleared9:
    if PatrolsBeaten == 9 { PodalPlazaCleared = RewardDue }
    PatrolsBeaten = 10
    exit
