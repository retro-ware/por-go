RollPatrolCount:
	Scratch1 = PartyStrength
	Scratch2 = EncStrengthTable[EncType]
	Scratch1 = Scratch1 / Scratch2
	Scratch1 = Scratch1 / 3
	Scratch2 = Random Scratch1
	Scratch1 = Scratch1 + Scratch2
	if 64 == PatrolsBeaten && CoverBroken == DisguiseState: Scratch1 = 2 * Scratch1
	if Scratch1 < 2: Scratch1 = 2
	Scratch2 = EncMinCountTable[EncType]
	if Scratch1 < Scratch2: Goto SetPatrolCount
	EncExtraCount = EncExtraTable[EncType]
	if Scratch1 > 50: Scratch1 = 50

	EncCount = Scratch1
	EncDistance = EncDistanceTable[EncType]
	Return

LoadPatrol:
	ClearMonsters
	Scratch1 = EncCount + EncExtraCount
	PrintClear Scratch1
	Print $AuxCounter
	Print " RUSH TO ATTACK."
	Scratch1 = EncMonsterTable[EncType]
	LoadMonster Scratch1 count=EncCount icon=Scratch1
	EncMorale = EncMoraleTable[EncType]
	if EncExtraCount == 0: Return
	EncMorale = EncMorale + 10
	Scratch1 += 1
	LoadMonster Scratch1 count=EncExtraCount icon=Scratch1
	Return
