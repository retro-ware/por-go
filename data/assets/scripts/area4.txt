RollSurprise:
	PartySurprise Scratch8 Scratch9
	And CastleFlags 1 Scratch10
	IfNe
	Add 1 Scratch9 Scratch9
	Surprise Scratch8 Scratch9 0 0
	if AlarmEndHour != 0: And Surprise 1 Surprise
	if Surprise == 3: Save 0 Surprise
	Compare Surprise 1
	Return

CheckAlarm:
	if AlarmEndHour < 1: Return
	Save Hours AuxRoll
	if Hours == 0 && AlarmHour == 23: Save 24 AuxRoll
	if AuxRoll < AlarmEndHour: Goto AlarmStillOn
	if TenMinutes < AlarmMinute: Goto AlarmStillOn
	Save 0 AlarmEndHour
	Compare 0 1
	Return

AlarmStillOn:
	if Search == 1: Goto AlarmSounding
	Random 9 AuxRoll
	if AuxRoll != 0: Goto AlarmOnReturn

AlarmSounding:
	PrintClear "AN ALARM IS SOUNDING."
	Delay

AlarmOnReturn:
	Compare 1 0
	Return

DelayRaiseAlarm:
	Delay

RaiseAlarm:
	if AlarmEndHour == 0: PrintClear "AN ALARM STARTS RINGING!"
	Delay
	Save Hours AlarmHour
	Save TenMinutes AlarmMinute
	Add 1 AlarmHour AlarmEndHour
	Return
