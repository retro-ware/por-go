RollSurprise:
	PartySurprise Scratch8 Scratch9
	if CastleFlags has Disguised: Scratch9 += 1
	Surprise Scratch8 Scratch9 0 0
	if AlarmEndHour != 0: Surprise &= PartyIsSurprised
	if Surprise == 3: Surprise = NoSurprise
	Compare Surprise 1
	Return

CheckAlarm:
	if AlarmEndHour < 1: Return
	AuxRoll = Hours
	if Hours == 0 && AlarmHour == 23: AuxRoll = 24
	if AuxRoll < AlarmEndHour: Goto AlarmStillOn
	if TenMinutes < AlarmMinute: Goto AlarmStillOn
	AlarmEndHour = 0
	Compare 0 1
	Return

AlarmStillOn:
	if Search == 1: Goto AlarmSounding
	AuxRoll = Random 9
	if AuxRoll != 0: Goto AlarmOnReturn

AlarmSounding:
	PrintClear "AN ALARM IS SOUNDING."
	Delay

AlarmOnReturn:
	Compare 1 0
	Return

DelayRaiseAlarm:
	Delay

RaiseAlarm:
	if AlarmEndHour == 0: PrintClear "AN ALARM STARTS RINGING!"
	Delay
	AlarmHour = Hours
	AlarmMinute = TenMinutes
	AlarmEndHour = 1 + AlarmHour
	Return
