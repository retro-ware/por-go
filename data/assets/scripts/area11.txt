event0:
	Scratch5 = 11
	if Hours >= 14: Scratch5 = 8
	Sky = Scratch5
	SkyInside = 10
	Scratch10 = 127 & Cell
	Exit

event1:
	PrintClear ""
	Picture None
	Portrait = None
	Scratch10 = 127 & Cell
	if Scratch10 == 0: Goto LeaveArea
	AuxCounter = 10

FindCellLoop:
	if AuxCounter == Scratch10: Goto CellDispatch
	if AuxCounter > 18: Exit
	AuxCounter += 1
	Goto FindCellLoop

event2:
	if FoughtWatch != 0: Goto NoRestHere
	RestEvery = 1
	RestChance = 101
	Exit

NoRestHere:
	RestEvery = 0
	RestChance = 0
	Exit

event3:
	alias Choice = Scratch1
	RestChance = 0
	Portrait = NoseguardHelmet
	SetupMonster ArmoredFighter distance=2 body=PlateArmor
	Delay
	Approach
	Delay
	Approach
	Delay
	PrintClear "YOU ARE ROUSTED BY THE CITY WATCH AND TOLD TO MOVE ALONG. WHAT DO YOU DO?"
	Choice = Menu "GO" -> ClearAndExit "FIGHT" -> FightWatch

event4:
	LoadMap 0 unused=0 icons=0
	LoadWalls Reset Reset Reset
	if RestChance == 255: Exit
	Goto NoRestHere

CellDispatch:
	AuxCounter -= 10
	if TrainerCooldown != 0: TrainerCooldown -= 1
	OnGoto AuxCounter ReadNotices ReadDoorSign ClericTrainer MagicUserTrainer EnterArena FightersSign FighterTrainer ThiefTrainer RoguesSign

ReadNotices:
	if NoticesRead > 0: Exit
	PrintClear "NOTICES FOR TEACHERS AND STUDENTS ARE POSTED ON THE WALLS.  A NOTE DIRECTS MAGIC USERS AND CLERICS TO GO NORTH AND FIGHTERS AND 'ROGUES' TO GO EAST."
	NoticesRead += 1
	Exit

ReadDoorSign:
	if DoorSignRead > 0: Exit
	PrintClear "A SIGN ON THE WALL DIRECTS MAGIC USERS TO GO THROUGH THE EAST DOOR AND CLERICS TO GO THROUGH THE WEST DOOR."
	DoorSignRead += 1
	Exit

ClericTrainer:
	if TrainerCooldown > 0: Exit
	TrainerCooldown = 2
	ShowPortrait DarkBeard ClericMace
	Scratch4 = 114
	$AuxMonsterName = "CLERICS "
	Goto AskTrain

MagicUserTrainer:
	if TrainerCooldown > 0: Exit
	TrainerCooldown = 2
	ShowPortrait DarkBeard NobleRobe
	Scratch4 = 113
	$AuxMonsterName = "MAGIC USERS "
	Goto AskTrain

EnterArena:
	if ArenaEntered > 0: Goto ClearAndExit
	PrintClear "THE ROOM IS FILLED WITH DUELING PAIRS."
	Gosub PressReturn
	ArenaEntered = 1
	Portrait = Bearded
	SetupMonster Fighter16 distance=2 body=PlateArmor
	Delay
	Approach
	Delay
	Approach
	Delay
	if DuelOffered > 0: Goto OfferPartner
	PrintClear "THE ARENA MASTER ASKS IF YOU WISH TO DUEL.  'THE DUELS  ARE EVENLY MATCHED AND NORMALLY NOT TO THE DEATH. DO YOU DUEL?'"
	Ask YesNoMenu "YES" -> ChooseDuelist "NO" -> OfferPartner

ChooseDuelist:
	Who "WHO WILL DUEL?"
	if SelControl < 128: Goto StartDuel
	PrintClear "I CANNOT ARRANGE SUCH A DUEL."
	Exit

StartDuel:
	ClearMonsters
	Call CallDuelA
	Combat
	Exit

OfferPartner:
	DuelOffered = 1
	if PartnerDeclined != 0: Exit
	PrintClear "'DO YOU SEEK A PARTNER FOR YOUR ADVENTURING?'"
	Ask YesNoMenu "YES" -> RollHireling "NO" -> DeclinePartner

RollHireling:
	if PartnerTries > 7: Goto NoHirelingLeft
	PartnerTries += 1
	Scratch1 = Random 250
	if Scratch1 < 20: Goto HireAcolyte
	if Scratch1 < 70: Goto HireWarrior
	if Scratch1 < 110: Goto HireEvoker
	if Scratch1 < 140: Goto HireSwordsman
	if Scratch1 < 170: Goto HireRobber
	if Scratch1 < 195: Goto HireCurate
	if Scratch1 < 220: Goto HireHero
	if Scratch1 < 240: Goto HireTheurgist

NoHirelingLeft:
	PrintClear "'THERE IS NO ONE WHO WILL GO WITH YOU.'"
	PartnerTries = 8
	Picture None
	Portrait = None
	Exit

HireAcolyte:
	Scratch2 = 0
	$AuxMonsterName = "ACOLYTE"
	Goto OfferHireling

HireWarrior:
	Scratch2 = 1
	$AuxMonsterName = "WARRIOR"
	Goto OfferHireling

HireEvoker:
	Scratch2 = 2
	$AuxMonsterName = "EVOKER"
	Goto OfferHireling

HireSwordsman:
	Scratch2 = 3
	$AuxMonsterName = "SWORDSMAN"
	Goto OfferHireling

HireRobber:
	Scratch2 = 4
	$AuxMonsterName = "ROBBER"
	Goto OfferHireling

HireCurate:
	Scratch2 = 5
	$AuxMonsterName = "CURATE"
	Goto OfferHireling

HireHero:
	Scratch2 = 6
	$AuxMonsterName = "HERO"
	Goto OfferHireling

HireTheurgist:
	Scratch2 = 7
	$AuxMonsterName = "THEURGIST"

OfferHireling:
	if Scratch2 <= 1: $AuxText = "1 SHARE"
	if Scratch2 == 2: $AuxText = "2 SHARES"
	if Scratch2 >= 3: $AuxText = "3 SHARES "
	if Scratch2 >= 4: $AuxTreasureText = " PLUS A PICK OF SPECIAL ITEMS"
	if Scratch2 > 5: $AuxText = "4 SHARES "
	PrintClear "'THERE IS A FINE "
	Print $AuxMonsterName
	Print " WHO WILL JOIN YOU FOR "
	Print $AuxText
	Print ". IS THIS ACCEPTABLE?'"
	Ask YesNoMenu "YES" -> AcceptHireling "NO" -> ConsiderAnother

AcceptHireling:
	alias Hireling = Scratch1
	Gosub CheckPartyFull
	Hireling = HirelingTable[Scratch2]
	Scratch2 = Scratch2 * 5
	Scratch2 += 50
	AddNPC Hireling Scratch2

ConsiderAnother:
	PrintClear "'WOULD YOU LIKE TO CONSIDER SOMEONE ELSE?'"
	Ask YesNoMenu "YES" -> RollHireling "NO" -> DeclinePartner

DeclinePartner:
	PartnerDeclined = 1
	Picture None
	Portrait = None
	Exit

FightersSign:
	PrintClear "A SIGN ON THE NORTH DOOR READS 'FIGHTERS'."
	Exit

FighterTrainer:
	if TrainerCooldown > 0: Exit
	TrainerCooldown = 2
	ShowPortrait DarkBeard PlateArmor
	Scratch4 = 120
	$AuxMonsterName = "FIGHTERS "
	Goto AskTrain

ThiefTrainer:
	if TrainerCooldown > 0: Exit
	TrainerCooldown = 2
	ShowPortrait DarkBeard RogueDagger
	Scratch4 = 116
	$AuxMonsterName = "THIEVES "
	Goto AskTrain

RoguesSign:
	PrintClear "A SIGN ON THE NORTH DOOR READS 'ROGUES'."
	Exit

AskTrain:
	PartyExp = Scratch4
	PrintClear "'WE TRAIN ONLY "
	Print $AuxMonsterName
	Print "HERE. DO YOU WANT TO TRAIN?'"
	Ask YesNoMenu "YES" -> OpenTraining "NO" -> ClearAndExit

OpenTraining:
	Program 0
	Picture None
	Exit

FightWatch:
	WatchCombatDone11 = 0
	Morale = 50
	ClearMonsters
	LoadMonster Level3Mu count=2 icon=Level3Mu
	LoadMonster ThLvlFighter84 count=12 icon=11
	LoadMonster Aides count=12 icon=ThLvlFighter
	LoadMonster Nomad count=12 icon=Nomad
	Combat
	WatchCombatDone11 = 1
	Gosub AfterWatchFight
	FoughtWatch = 1
	Exit

LeaveArea:
	NewECL NewPhlan

AfterWatchFight:
	if CombatEnd > 128: Goto EjectParty
	if WatchCombatDone11 > 0: Return

ClearAndExit:
	Temp1 = 0
	Temp2 = 0
	Temp3 = 0
	Picture None
	Portrait = None
	Exit

YesNoMenu:
	alias Choice = Scratch1
	Choice = Menu "YES" "NO"
	Return

CheckPartyFull:
	if PartyCount < 8: Return
	PrintClear "'YOUR PARTY IS TOO LARGE TO JOIN AT THIS TIME.  RETURN WHEN IT IS SMALLER.'"
	Gosub PressReturn
	Goto ClearAndExit

EjectParty:
	Facing = Random 3
	AuxMember = Random 4
	PartyX = EjectXTable[AuxMember]
	PartyY = EjectYTable[AuxMember]
	PrintClear ""
	Call CallRedraw
	NewECL NewPhlan

PressReturn:
	AuxMenuChoice = Menu "PRESS <RETURN> OR BUTTON TO CONTINUE"
	Return

EjectXTable:
	Data 3 10 2 12

EjectYTable:
	Data 11 10 0 3

HirelingTable:
	Data 108 103 122 36 111 112 109 110 0
