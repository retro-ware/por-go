event0:
	Save 11 Scratch5
	if Hours >= 14: Save 8 Scratch5
	Save Scratch5 Sky
	Save 10 SkyInside
	And 127 Cell Scratch10
	Exit

event1:
	PrintClear ""
	Picture None
	Save 255 Portrait
	And 127 Cell Scratch10
	if Scratch10 == 0: Goto LeaveArea
	Save 10 AuxCounter

FindCellLoop:
	if AuxCounter == Scratch10: Goto CellDispatch
	if AuxCounter > 18: Exit
	Add 1 AuxCounter AuxCounter
	Goto FindCellLoop

event2:
	if FoughtWatch != 0: Goto NoRestHere
	Save 1 RestEvery
	Save 101 RestChance
	Exit

NoRestHere:
	Save 0 RestEvery
	Save 0 RestChance
	Exit

event3:
	Save 0 RestChance
	Save 57 Portrait
	SetupMonster ArmoredFighter distance=2 body=1
	Delay
	Approach
	Delay
	Approach
	Delay
	PrintClear "YOU ARE ROUSTED BY THE CITY WATCH AND TOLD TO MOVE ALONG. WHAT DO YOU DO?"
	HMenu Scratch1 2 "GO" "FIGHT"
	OnGoto Scratch1 2 ClearAndExit FightWatch

event4:
	LoadFiles 0 0 0
	LoadPieces 127 127 127
	if RestChance == 255: Exit
	Goto NoRestHere

CellDispatch:
	Sub 10 AuxCounter AuxCounter
	if TrainerCooldown != 0: Sub 1 TrainerCooldown TrainerCooldown
	OnGoto AuxCounter 9 ReadNotices ReadDoorSign ClericTrainer MagicUserTrainer EnterArena FightersSign FighterTrainer ThiefTrainer RoguesSign

ReadNotices:
	if NoticesRead > 0: Exit
	PrintClear "NOTICES FOR TEACHERS AND STUDENTS ARE POSTED ON THE WALLS.  A NOTE DIRECTS MAGIC USERS AND CLERICS TO GO NORTH AND FIGHTERS AND 'ROGUES' TO GO EAST."
	Add 1 NoticesRead NoticesRead
	Exit

ReadDoorSign:
	if DoorSignRead > 0: Exit
	PrintClear "A SIGN ON THE WALL DIRECTS MAGIC USERS TO GO THROUGH THE EAST DOOR AND CLERICS TO GO THROUGH THE WEST DOOR."
	Add 1 DoorSignRead DoorSignRead
	Exit

ClericTrainer:
	if TrainerCooldown > 0: Exit
	Save 2 TrainerCooldown
	ShowPortrait 16 4
	Save 114 Scratch4
	Save "CLERICS " $AuxMonsterName
	Goto AskTrain

MagicUserTrainer:
	if TrainerCooldown > 0: Exit
	Save 2 TrainerCooldown
	ShowPortrait 16 2
	Save 113 Scratch4
	Save "MAGIC USERS " $AuxMonsterName
	Goto AskTrain

EnterArena:
	if ArenaEntered > 0: Goto ClearAndExit
	PrintClear "THE ROOM IS FILLED WITH DUELING PAIRS."
	Gosub PressReturn
	Save 1 ArenaEntered
	Save 9 Portrait
	SetupMonster Fighter16 distance=2 body=1
	Delay
	Approach
	Delay
	Approach
	Delay
	if DuelOffered > 0: Goto OfferPartner
	PrintClear "THE ARENA MASTER ASKS IF YOU WISH TO DUEL.  'THE DUELS  ARE EVENLY MATCHED AND NORMALLY NOT TO THE DEATH. DO YOU DUEL?'"
	Gosub YesNoMenu
	OnGoto Scratch1 2 ChooseDuelist OfferPartner

ChooseDuelist:
	Who "WHO WILL DUEL?"
	if SelControl < 128: Goto StartDuel
	PrintClear "I CANNOT ARRANGE SUCH A DUEL."
	Exit

StartDuel:
	ClearMonsters
	Call CallDuelA
	Combat
	Exit

OfferPartner:
	Save 1 DuelOffered
	if PartnerDeclined != 0: Exit
	PrintClear "'DO YOU SEEK A PARTNER FOR YOUR ADVENTURING?'"
	Gosub YesNoMenu
	OnGoto Scratch1 2 RollHireling DeclinePartner

RollHireling:
	if PartnerTries > 7: Goto NoHirelingLeft
	Add 1 PartnerTries PartnerTries
	Random 250 Scratch1
	if Scratch1 < 20: Goto HireAcolyte
	if Scratch1 < 70: Goto HireWarrior
	if Scratch1 < 110: Goto HireEvoker
	if Scratch1 < 140: Goto HireSwordsman
	if Scratch1 < 170: Goto HireRobber
	if Scratch1 < 195: Goto HireCurate
	if Scratch1 < 220: Goto HireHero
	if Scratch1 < 240: Goto HireTheurgist

NoHirelingLeft:
	PrintClear "'THERE IS NO ONE WHO WILL GO WITH YOU.'"
	Save 8 PartnerTries
	Picture None
	Save 255 Portrait
	Exit

HireAcolyte:
	Save 0 Scratch2
	Save "ACOLYTE" $AuxMonsterName
	Goto OfferHireling

HireWarrior:
	Save 1 Scratch2
	Save "WARRIOR" $AuxMonsterName
	Goto OfferHireling

HireEvoker:
	Save 2 Scratch2
	Save "EVOKER" $AuxMonsterName
	Goto OfferHireling

HireSwordsman:
	Save 3 Scratch2
	Save "SWORDSMAN" $AuxMonsterName
	Goto OfferHireling

HireRobber:
	Save 4 Scratch2
	Save "ROBBER" $AuxMonsterName
	Goto OfferHireling

HireCurate:
	Save 5 Scratch2
	Save "CURATE" $AuxMonsterName
	Goto OfferHireling

HireHero:
	Save 6 Scratch2
	Save "HERO" $AuxMonsterName
	Goto OfferHireling

HireTheurgist:
	Save 7 Scratch2
	Save "THEURGIST" $AuxMonsterName

OfferHireling:
	if Scratch2 <= 1: Save "1 SHARE" $AuxText
	if Scratch2 == 2: Save "2 SHARES" $AuxText
	if Scratch2 >= 3: Save "3 SHARES " $AuxText
	if Scratch2 >= 4: Save " PLUS A PICK OF SPECIAL ITEMS" $AuxTreasureText
	if Scratch2 > 5: Save "4 SHARES " $AuxText
	PrintClear "'THERE IS A FINE "
	Print $AuxMonsterName
	Print " WHO WILL JOIN YOU FOR "
	Print $AuxText
	Print ". IS THIS ACCEPTABLE?'"
	Gosub YesNoMenu
	OnGoto Scratch1 2 AcceptHireling ConsiderAnother

AcceptHireling:
	Gosub CheckPartyFull
	GetTable HirelingTable Scratch2 Scratch1
	Mul Scratch2 5 Scratch2
	Add 50 Scratch2 Scratch2
	AddNPC Scratch1 Scratch2

ConsiderAnother:
	PrintClear "'WOULD YOU LIKE TO CONSIDER SOMEONE ELSE?'"
	Gosub YesNoMenu
	OnGoto Scratch1 2 RollHireling DeclinePartner

DeclinePartner:
	Save 1 PartnerDeclined
	Picture None
	Save 255 Portrait
	Exit

FightersSign:
	PrintClear "A SIGN ON THE NORTH DOOR READS 'FIGHTERS'."
	Exit

FighterTrainer:
	if TrainerCooldown > 0: Exit
	Save 2 TrainerCooldown
	ShowPortrait 16 1
	Save 120 Scratch4
	Save "FIGHTERS " $AuxMonsterName
	Goto AskTrain

ThiefTrainer:
	if TrainerCooldown > 0: Exit
	Save 2 TrainerCooldown
	ShowPortrait 16 8
	Save 116 Scratch4
	Save "THIEVES " $AuxMonsterName
	Goto AskTrain

RoguesSign:
	PrintClear "A SIGN ON THE NORTH DOOR READS 'ROGUES'."
	Exit

AskTrain:
	Save Scratch4 PartyExp
	PrintClear "'WE TRAIN ONLY "
	Print $AuxMonsterName
	Print "HERE. DO YOU WANT TO TRAIN?'"
	Gosub YesNoMenu
	OnGoto Scratch1 2 OpenTraining ClearAndExit

OpenTraining:
	Program 0
	Picture None
	Exit

FightWatch:
	Save 0 WatchCombatDone11
	Save 50 Morale
	ClearMonsters
	LoadMonster Level3Mu count=2 icon=Level3Mu
	LoadMonster ThLvlFighter84 count=12 icon=11
	LoadMonster Aides count=12 icon=ThLvlFighter
	LoadMonster Nomad count=12 icon=Nomad
	Combat
	Save 1 WatchCombatDone11
	Gosub AfterWatchFight
	Save 1 FoughtWatch
	Exit

LeaveArea:
	NewECL NewPhlan

AfterWatchFight:
	if CombatEnd > 128: Goto EjectParty
	if WatchCombatDone11 > 0: Return

ClearAndExit:
	Save 0 Temp1
	Save 0 Temp2
	Save 0 Temp3
	Picture None
	Save 255 Portrait
	Exit

YesNoMenu:
	HMenu Scratch1 2 "YES" "NO"
	Return

CheckPartyFull:
	if PartyCount < 8: Return
	PrintClear "'YOUR PARTY IS TOO LARGE TO JOIN AT THIS TIME.  RETURN WHEN IT IS SMALLER.'"
	Gosub PressReturn
	Goto ClearAndExit

EjectParty:
	Random 3 Facing
	Random 4 AuxMember
	GetTable EjectXTable AuxMember PartyX
	GetTable EjectYTable AuxMember PartyY
	PrintClear ""
	Call CallRedraw
	NewECL NewPhlan

PressReturn:
	HMenu AuxMenuChoice 1 "PRESS <RETURN> OR BUTTON TO CONTINUE"
	Return

EjectXTable:
	Data 3 10 2 12

EjectYTable:
	Data 11 10 0 3

HirelingTable:
	Data 108 103 122 36 111 112 109 110 0
