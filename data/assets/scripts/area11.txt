on move {
    Scratch5 = 11
    if Hours >= 14 { Scratch5 = 8 }
    Sky = Scratch5
    SkyInside = 10
    Scratch10 = ~Inside & Cell
    exit
}

on step {
    Text("")
    Picture(None)
    Portrait = None
    Scratch10 = ~Inside & Cell
    if Scratch10 == 0 { goto LeaveArea }
    AuxCounter = 10
    goto FindCellLoop

    FindCellLoop:
    if AuxCounter == Scratch10 { goto CellDispatch }
    if AuxCounter > 18 { exit }
    AuxCounter += 1
    goto FindCellLoop
}

on camp {
    if FoughtWatch != 0 { goto NoRestHere }
    RestEvery = 1
    RestChance = 101
    exit
}

NoRestHere:
    RestEvery = 0
    RestChance = 0
    exit

on rest {
    RestChance = 0
    Portrait = NoseguardHelmet
    SetupMonster(ArmoredFighter, distance=2, body=PlateArmor)
    Delay()
    Approach()
    Delay()
    Approach()
    Delay()
    Text("YOU ARE ROUSTED BY THE CITY WATCH AND TOLD TO MOVE ALONG. WHAT DO YOU DO?")
    Scratch1 = Menu("GO", "FIGHT")
    goto [ClearAndExit, FightWatch][Scratch1]
    goto event4
}

on load {
    LoadMap(NewPhlan, icons=Overland)
    LoadWalls(Reset, Reset, Reset)
    if RestChance == 255 { exit }
    goto NoRestHere
}

CellDispatch:
    AuxCounter -= 10
    if TrainerCooldown != 0 { TrainerCooldown -= 1 }
    goto [ReadNotices, ReadDoorSign, ClericTrainer, MagicUserTrainer, EnterArena, FightersSign,
        FighterTrainer, ThiefTrainer, RoguesSign][AuxCounter]
    goto ReadNotices

ReadNotices:
    if NoticesRead > 0 { exit }
    Text("NOTICES FOR TEACHERS AND STUDENTS ARE POSTED ON THE WALLS.  A NOTE DIRECTS MAGIC USERS AND CLERICS TO GO NORTH AND FIGHTERS AND 'ROGUES' TO GO EAST.")
    NoticesRead += 1
    exit

ReadDoorSign:
    if DoorSignRead > 0 { exit }
    Text("A SIGN ON THE WALL DIRECTS MAGIC USERS TO GO THROUGH THE EAST DOOR AND CLERICS TO GO THROUGH THE WEST DOOR.")
    DoorSignRead += 1
    exit

ClericTrainer:
    if TrainerCooldown > 0 { exit }
    TrainerCooldown = 2
    ShowPortrait(DarkBeard, ClericMace)
    Scratch4 = 114
    $AuxMonsterName = "CLERICS "
    goto AskTrain

MagicUserTrainer:
    if TrainerCooldown > 0 { exit }
    TrainerCooldown = 2
    ShowPortrait(DarkBeard, NobleRobe)
    Scratch4 = 113
    $AuxMonsterName = "MAGIC USERS "
    goto AskTrain

EnterArena:
    if ArenaEntered > 0 { goto ClearAndExit }
    Text("THE ROOM IS FILLED WITH DUELING PAIRS.")
    WaitReturnPressed()
    ArenaEntered = 1
    Portrait = Bearded
    SetupMonster(Fighter16, distance=2, body=PlateArmor)
    Delay()
    Approach()
    Delay()
    Approach()
    Delay()
    if DuelOffered > 0 { goto OfferPartner }
    Text("THE ARENA MASTER ASKS IF YOU WISH TO DUEL.  'THE DUELS  ARE EVENLY MATCHED AND NORMALLY NOT TO THE DEATH. DO YOU DUEL?'")
    YesNoMenu()
    goto [ChooseDuelist, OfferPartner][Scratch1]
    goto ChooseDuelist

ChooseDuelist:
    Who("WHO WILL DUEL?")
    if SelControl < 128 { goto StartDuel }
    Text("I CANNOT ARRANGE SUCH A DUEL.")
    exit

    StartDuel:
    ClearMonsters()
    CallDuelA()
    Combat()
    exit

OfferPartner:
    DuelOffered = 1
    if PartnerDeclined != 0 { exit }
    Text("'DO YOU SEEK A PARTNER FOR YOUR ADVENTURING?'")
    YesNoMenu()
    goto [RollHireling, DeclinePartner][Scratch1]
    goto RollHireling

RollHireling:
    if PartnerTries > 7 { goto NoHirelingLeft }
    PartnerTries += 1
    Scratch1 = Random(250)
    if Scratch1 < 20 { goto HireAcolyte }
    if Scratch1 < 70 { goto HireWarrior }
    if Scratch1 < 110 { goto HireEvoker }
    if Scratch1 < 140 { goto HireSwordsman }
    if Scratch1 < 170 { goto HireRobber }
    if Scratch1 < 195 { goto HireCurate }
    if Scratch1 < 220 { goto HireHero }
    if Scratch1 < 240 { goto HireTheurgist }
    goto NoHirelingLeft

    NoHirelingLeft:
    Text("'THERE IS NO ONE WHO WILL GO WITH YOU.'")
    PartnerTries = 8
    Picture(None)
    Portrait = None
    exit

    HireAcolyte:
    Scratch2 = 0
    $AuxMonsterName = "ACOLYTE"
    goto OfferHireling

    HireWarrior:
    Scratch2 = 1
    $AuxMonsterName = "WARRIOR"
    goto OfferHireling

    HireEvoker:
    Scratch2 = 2
    $AuxMonsterName = "EVOKER"
    goto OfferHireling

    HireSwordsman:
    Scratch2 = 3
    $AuxMonsterName = "SWORDSMAN"
    goto OfferHireling

    HireRobber:
    Scratch2 = 4
    $AuxMonsterName = "ROBBER"
    goto OfferHireling

    HireCurate:
    Scratch2 = 5
    $AuxMonsterName = "CURATE"
    goto OfferHireling

    HireHero:
    Scratch2 = 6
    $AuxMonsterName = "HERO"
    goto OfferHireling

    HireTheurgist:
    Scratch2 = 7
    $AuxMonsterName = "THEURGIST"
    goto OfferHireling

    OfferHireling:
    if Scratch2 <= 1 { $AuxText = "1 SHARE" }
    if Scratch2 == 2 { $AuxText = "2 SHARES" }
    if Scratch2 >= 3 { $AuxText = "3 SHARES " }
    if Scratch2 >= 4 { $AuxTreasureText = " PLUS A PICK OF SPECIAL ITEMS" }
    if Scratch2 > 5 { $AuxText = "4 SHARES " }
    Text("'THERE IS A FINE ")
    Print($AuxMonsterName)
    Print(" WHO WILL JOIN YOU FOR ")
    Print($AuxText)
    Print(". IS THIS ACCEPTABLE?'")
    YesNoMenu()
    goto [AcceptHireling, ConsiderAnother][Scratch1]
    goto AcceptHireling

AcceptHireling:
    CheckPartyFull()
    Scratch1 = HirelingTable[Scratch2]
    Scratch2 = Scratch2 * 5
    Scratch2 += 50
    AddNPC(Scratch1, Scratch2)
    goto ConsiderAnother

ConsiderAnother:
    Text("'WOULD YOU LIKE TO CONSIDER SOMEONE ELSE?'")
    YesNoMenu()
    goto [RollHireling, DeclinePartner][Scratch1]
    goto DeclinePartner

DeclinePartner:
    PartnerDeclined = 1
    Picture(None)
    Portrait = None
    exit

FightersSign:
    Text("A SIGN ON THE NORTH DOOR READS 'FIGHTERS'.")
    exit

FighterTrainer:
    if TrainerCooldown > 0 { exit }
    TrainerCooldown = 2
    ShowPortrait(DarkBeard, PlateArmor)
    Scratch4 = 120
    $AuxMonsterName = "FIGHTERS "
    goto AskTrain

ThiefTrainer:
    if TrainerCooldown > 0 { exit }
    TrainerCooldown = 2
    ShowPortrait(DarkBeard, RogueDagger)
    Scratch4 = 116
    $AuxMonsterName = "THIEVES "
    goto AskTrain

RoguesSign:
    Text("A SIGN ON THE NORTH DOOR READS 'ROGUES'.")
    exit

AskTrain:
    PartyExp = Scratch4
    Text("'WE TRAIN ONLY ")
    Print($AuxMonsterName)
    Print("HERE. DO YOU WANT TO TRAIN?'")
    YesNoMenu()
    goto [OpenTraining, ClearAndExit][Scratch1]
    goto OpenTraining

OpenTraining:
    Program(0)
    Picture(None)
    exit

FightWatch:
    WatchCombatDone11 = 0
    Morale = 50
    ClearMonsters()
    LoadMonster(Level3Mu, count=2, icon=Level3Mu)
    LoadMonster(ThLvlFighter84, count=12, icon=Soldier)
    LoadMonster(Aides, count=12, icon=ThLvlFighter)
    LoadMonster(Nomad, count=12, icon=Nomad)
    Combat()
    WatchCombatDone11 = 1
    AfterWatchFight()
    FoughtWatch = 1
    exit

LeaveArea:
    NewECL(NewPhlan)

def AfterWatchFight() {
    if CombatEnd > PartyDied { goto EjectParty }
    if WatchCombatDone11 > 0 { return }
    goto ClearAndExit
}

ClearAndExit:
    Temp1 = 0
    Temp2 = 0
    Temp3 = 0
    Picture(None)
    Portrait = None
    exit

def YesNoMenu() {
    Scratch1 = Menu("YES", "NO")
    return
}

def CheckPartyFull() {
    if PartyCount < 8 { return }
    Text("'YOUR PARTY IS TOO LARGE TO JOIN AT THIS TIME.  RETURN WHEN IT IS SMALLER.'")
    WaitReturnPressed()
    goto ClearAndExit
}

EjectParty:
    Facing = Random(3)
    AuxMember = Random(4)
    PartyX = EjectXTable[AuxMember]
    PartyY = EjectYTable[AuxMember]
    Text("")
    CallRedraw()
    NewECL(NewPhlan)

def WaitReturnPressed() {
    AuxMenuChoice = Menu("PRESS <RETURN> OR BUTTON TO CONTINUE")
    return
}

# tables
EjectXTable = [3, 10, 2, 12]
EjectYTable = [11, 10, 0, 3]
HirelingTable = [108, 103, 122, 36, 111, 112, 109, 110, 0]
