on move {
    Scratch5 = 11
    if Hours >= 14 { Scratch5 = 8 }
    Sky = Scratch5
    SkyInside = 9
    if LeftMap == 0 { goto StoreCellCode }
    NoCoords = 0
    CallStepForward()
    HoldMove = SkipMove
    if Facing == East {
        EnterArea(NewPhlan)
    } else {
        EnterArea(KutosWell)
    }

    StoreCellCode:
    Scratch10 = ~Inside & Cell
    exit
}

AfterPatrolFight:
    if CombatEnd > PartyDied { goto RunAway }
    if CombatEnd < 1 {
        PatrolsBeaten += 1
        CountSlumsCleared()
    }
    if (>) { goto Done }
    if (==) { Compare(Beaten, 0) }
    if Beaten > 0 {
        PatrolsBeaten += 1
        CountSlumsCleared()
    }
    goto Done

FightPatrol:
    if Scratch8 == 4 { goto CheckOrcStrength }
    if AuxPartyStrength < 8 { goto FightPlainPatrol }
    goto AddLeaders

    CheckOrcStrength:
    if AuxPartyStrength < 14 { goto FightPlainPatrol }

    AddLeaders:
    Scratch3 = 1 + Scratch8
    if Scratch8 < 1 {
        Scratch2 = 3
    } else {
        Scratch2 = 4
    }
    if AuxPartyStrength > 18 { goto FightPatrolWithBugbear }
    goto FightPatrolWithLeaders

    FightPlainPatrol:
    Morale = PatrolMorale
    ClearMonsters()
    LoadMonster(Scratch8, count=AuxPartyStrength, icon=Scratch8)
    Combat()
    goto AfterPatrolFight

    FightPatrolWithLeaders:
    PatrolMorale += 5
    Morale = PatrolMorale
    ClearMonsters()
    LoadMonster(Scratch3, count=Scratch2, icon=Scratch3)
    LoadMonster(Scratch8, count=AuxPartyStrength, icon=Scratch8)
    Combat()
    goto AfterPatrolFight

    FightPatrolWithBugbear:
    PatrolMorale += 15
    Morale = PatrolMorale
    ClearMonsters()
    LoadMonster(Scratch3, count=Scratch2, icon=Scratch3)
    LoadMonster(Bugbear, count=1, icon=Bugbear)
    LoadMonster(Scratch8, count=AuxPartyStrength, icon=Scratch8)
    Combat()
    goto AfterPatrolFight

YouHide:
    Text("YOU HIDE.")
    exit

RunAway:
    Facing = Random(3)
    AuxTemp9 = Random(4)
    PartyX = HideXTable[AuxTemp9]
    PartyY = HideYTable[AuxTemp9]
    Text("")

RedrawAndRecheck:
    CallRedraw()
    goto L_9975

Done:
    Picture(None)
    exit

ConversePatrol:
    Surprise = NoSurprise
    Text("YOU ARE CONVERSING WITH ")
    if AuxPartyStrength > 18 { Print("A BUGBEAR LEADING") }
    Print("A PARTY OF ")
    Print($AuxText)
    if AuxPartyStrength > 8 { goto ParlayMidStrength }
    AuxParlay = Parlay(10, 20, 20, 20, 30)
    goto ParlayResult

    ParlayMidStrength:
    if AuxPartyStrength > 15 { goto ParlayHighStrength }
    AuxParlay = Parlay(5, 15, 15, 10, 25)
    goto ParlayResult

    ParlayHighStrength:
    AuxParlay = Parlay(0, 10, 10, 0, 15)

    ParlayResult:
    AuxRoll2 = Random(30)
    AuxParlay = AuxParlay + AuxRoll2
    AuxParlay = AuxParlay + SelReaction
    if AuxParlay < 75 { goto PatrolAngered }
    if AuxParlay > 85 { goto PatrolWarnsLeave }
    Text("THE MONSTERS RESPOND, 'DA BOSS DON'T LIKE YOUSE GUYS. GIVE US SUMPTIN AN MAYBE WE'LL FERGET WE SEEN YAS.' (WHAT DO YOU DO?)")
    Scratch9 = Menu("PAY", "SURRENDER", "RUN", "FIGHT")
    goto [PayPatrol, SurrenderToPatrol, TryRunning, FightPatrol][Scratch9]

PayPatrol:
    Text("TYPE IN THE NUMBER OF GP YOU OFFER.")
    Scratch9 = InputNumber(2)
    if Scratch9 >= AuxPartyStrength { goto PatrolTakesCoins }
    Text("'WHAT A BUNCH OF CHEAPSKATES,' THE LEADER COMPLAINS. 'LET 'EM HAVE IT BOYS!'")
    goto FightPatrol

    PatrolTakesCoins:
    Text("THEY PAW THROUGH YOUR COINS AND LEAVE.")
    Rob(1, 50, 0)
    Picture(None)
    exit

SurrenderToPatrol:
    Text("THE MONSTERS ROB AND BLINDFOLD YOU. YOU ARE KICKED SOUNDLY AND SENT ON YOUR WAY.")
    Rob(1, 100, 100)
    WaitReturnPressed()
    goto RunAway

TryRunning:
    CheckParty(SelMove, 0, Scratch9, Scratch1, Scratch1, Scratch1)
    if AuxEncResult < Scratch1 { goto RunAway }
    if AuxEncResult > Scratch1 { goto FightPatrol }
    if AuxEncResult == Scratch1 { Scratch1 = Random(1) }
    if Scratch1 == 1 { goto RunAway }
    goto FightPatrol

PatrolWarnsLeave:
    if AuxParlay > 95 { goto PatrolAdvice }
    Text("THE LEADER SPEAKS, 'THE BOSS DON'T ALLOW YOUR KIND IN HIS TERRITORY.  LEAVE NOW OR WE'LL HAVE TO RUN YOU IN.' (WHAT DO YOU DO?)")
    Scratch9 = Menu("FIGHT", "STAY", "LEAVE", "RUN")
    goto [FightPatrol, FightPatrol, Done, RunAway][Scratch9]

    PatrolAdvice:
    Text("'LET ME GIVE YOU A PIECE OF ADVICE,' THEIR LEADER SAYS.  'STAY AWAY FROM THE TEXTILE HOUSE. THE UNDEAD ARE GOING CRAZY UP THERE.'  THE MONSTERS TURN TO LEAVE. (WHAT DO YOU DO?)")
    Scratch9 = Menu("LET THEM GO", "STOP THEM")
    goto [Done, StopPatrol][Scratch9]

StopPatrol:
    Text("'HOW DARE YOU INTERFERE WITH THE BOSS'S BUSINESS?' THE LEADER ROARS.  'KILL THEM,' HE ORDERS.")
    WaitReturnPressed()
    goto FightPatrol

PatrolAngered:
    Text("YOU HAVE ANGERED THE")
    Print($AuxText)
    Print("THEY ATTACK.")
    if AuxParlay < 65 { PatrolMorale = PatrolMorale + 20 }
    WaitReturnPressed()
    goto FightPatrol

def ClearPortrait() {
    Portrait = None
    return
}

def CountSlumsCleared() {
    if QuestSlums >= RewardDue { return }
    QuestSlums += 1
    if QuestSlums < 25 { return }
    QuestSlums = RewardDue
    return
}

def WaitReturnPressed() {
    AuxMenuChoice = Menu("PRESS <RETURN> OR BUTTON TO CONTINUE")
    Text("")
    return
}
