ParlayResult:
	alias Choice = Scratch9
	AuxRoll2 = Random 30
	AuxParlay = AuxParlay + AuxRoll2
	AuxParlay = AuxParlay + SelReaction
	if AuxParlay < 75: Goto PatrolAngered
	if AuxParlay > 85: Goto PatrolWarnsLeave
	PrintClear "THE MONSTERS RESPOND, 'DA BOSS DON'T LIKE YOUSE GUYS. GIVE US SUMPTIN AN MAYBE WE'LL FERGET WE SEEN YAS.' (WHAT DO YOU DO?)"
	Choice = Menu "PAY" -> PayPatrol "SURRENDER" -> SurrenderToPatrol "RUN" -> TryRunning "FIGHT" -> FightPatrol

PayPatrol:
	PrintClear "TYPE IN THE NUMBER OF GP YOU OFFER."
	Scratch9 = InputNumber 2
	if Scratch9 >= AuxPartyStrength: Goto PatrolTakesCoins
	PrintClear "'WHAT A BUNCH OF CHEAPSKATES,' THE LEADER COMPLAINS. 'LET 'EM HAVE IT BOYS!'"
	Goto FightPatrol

PatrolTakesCoins:
	PrintClear "THEY PAW THROUGH YOUR COINS AND LEAVE."
	Rob 1 50 0
	Picture None
	Exit

SurrenderToPatrol:
	PrintClear "THE MONSTERS ROB AND BLINDFOLD YOU. YOU ARE KICKED SOUNDLY AND SENT ON YOUR WAY."
	Rob 1 100 100
	Gosub WaitReturnPressed
	Goto RunAway

TryRunning:
	CheckParty SelMove 0 Scratch9 Scratch1 Scratch1 Scratch1
	if AuxEncResult < Scratch1: Goto RunAway
	also if AuxEncResult > Scratch1: Goto FightPatrol
	also if AuxEncResult == Scratch1: Scratch1 = Random 1
	if Scratch1 == 1: Goto RunAway
	Goto FightPatrol

PatrolWarnsLeave:
	alias Choice = Scratch9
	if AuxParlay > 95: Goto PatrolAdvice
	PrintClear "THE LEADER SPEAKS, 'THE BOSS DON'T ALLOW YOUR KIND IN HIS TERRITORY.  LEAVE NOW OR WE'LL HAVE TO RUN YOU IN.' (WHAT DO YOU DO?)"
	Choice = Menu "FIGHT" -> FightPatrol "STAY" -> FightPatrol "LEAVE" -> Done "RUN" -> RunAway

PatrolAdvice:
	alias Choice = Scratch9
	PrintClear "'LET ME GIVE YOU A PIECE OF ADVICE,' THEIR LEADER SAYS.  'STAY AWAY FROM THE TEXTILE HOUSE. THE UNDEAD ARE GOING CRAZY UP THERE.'  THE MONSTERS TURN TO LEAVE. (WHAT DO YOU DO?)"
	Choice = Menu "LET THEM GO" -> Done "STOP THEM" -> StopPatrol

StopPatrol:
	PrintClear "'HOW DARE YOU INTERFERE WITH THE BOSS'S BUSINESS?' THE LEADER ROARS.  'KILL THEM,' HE ORDERS."
	Gosub WaitReturnPressed
	Goto FightPatrol

PatrolAngered:
	PrintClear "YOU HAVE ANGERED THE"
	Print $AuxText
	Print "THEY ATTACK."
	if AuxParlay < 65: PatrolMorale = PatrolMorale + 20
	Gosub WaitReturnPressed
	Goto FightPatrol

ClearPortrait:
	Portrait = None
	Return

CountSlumsCleared:
	if QuestSlums >= RewardDue: Return
	QuestSlums += 1
	if QuestSlums < 25: Return
	QuestSlums = RewardDue
	Return

WaitReturnPressed:
	alias Choice = AuxMenuChoice
	Choice = Menu "PRESS <RETURN> OR BUTTON TO CONTINUE"
	PrintClear ""
	Return
