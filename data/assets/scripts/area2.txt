on load {
    Sky = 11
    SkyInside = 8
    LoadMap(2, unused=2, icons=255)
    LoadWalls(1, 3, 20)
    if AreaKind == HobgoblinBarracks { exit }
    NoCoords = 0
    exit
}

on camp {
    RestEvery = 12
    RestChance = 6
    if TextileHouseCleared > 253 { RestEvery = 0 }
    exit
}

on rest {
    goto RandomEncounter
}

on move {
    if LeftMap == 0 { exit }
    CallStepForward()
    HoldMove = SkipMove
    goto [ExitNorth, ExitEast, Done, ExitWest][Facing]
} # falls into ExitNorth

ExitNorth:
    EnterArea(PodolPlaza)

ExitEast:
    EnterArea(MendorsLibrary)

ExitWest:
    SymbolSet = 7
    MapX = 11
    MapY = 28
    NewECL(Wilderness)

on step {
    Scratch10 = ~Inside & Cell
    Text(" ")
    goto [RandomEncounterRoll, CourtyardAmbush, Barracks, StenchRoom, DeterioratedRoom, TapestryRoom,
        Nursery, CarvedWall, GraffitiRoom, Done, Done, Done, DarkObject, DeadHobgoblin, Kitchen,
        SmokingFire, Larder, SmallDarkRoom, GrishnakTemple, BloodRoom, SecretDoorHint, OgreChief,
        RandomEncounterRoll, DiaryPage, ThievesWell][Scratch10]
} # falls into Done

Done:
    exit

RandomEncounterRoll:
    Scratch1 = 5
    if Search != Searching { Scratch1 = 30 }
    Scratch1 = Random(Scratch1)
    if 1 != Scratch1 { exit }

RandomEncounter:
    Scratch1 = 18
    if Scratch10 == 22 and IronBoxFound == 0 { goto PickEncounter }
    if TextileHouseCleared > 253 { exit }
    Scratch1 = Random(10)
    Scratch1 += EncounterCount
    if EncounterCount == 9 and IronBoxFound == 0 { exit }
    EncounterCount += 1
    if EncounterCount != 10 or IronBoxFound != Yes { goto PickEncounter }
    TextileHouseCleared = RewardDue
    exit

    PickEncounter:
    if Scratch1 > 17 { goto HobgoblinsCharge }
    if Scratch1 > 12 { goto SkeletalForce }
    if Scratch1 > 7 { goto GhoulsAttack }
    if Scratch1 > 3 { goto ScorpionsApproach }
    Morale = 60
    SetupMonster(GiantFrog, distance=0, closeup=GiantFrog)
    Text("EVIL LOOKING FROGS BOIL UP FROM BELOW.")
    PressToContinue()
    ClearMonsters()
    LoadMonster(PoisonousFrog, count=8, icon=PoisonousFrog)
    Combat()
    if CombatEnd > PartyDied { goto EjectToStreet }
    exit

    HobgoblinsCharge:
    Morale = 60
    SetupMonster(Hobgoblin, distance=2, closeup=Hobgoblin)
    Text("HOBGOBLINS CHARGE SCREAMING.")
    PressToContinue()
    ClearMonsters()
    LoadMonster(Hobgoblin, count=8, icon=Hobgoblin)
    LoadMonster(HobgoblinLdr, count=1, icon=HobgoblinLdr)
    Combat()
    if CombatEnd > PartyDied { goto EjectToStreet }
    exit

    SkeletalForce:
    Morale = 100
    SetupMonster(Skeleton, distance=2, closeup=Skeleton)
    Text("A SKELETAL FORCE SPOTS YOU AND ATTACKS.")
    PressToContinue()
    ClearMonsters()
    LoadMonster(Wight, count=1, icon=Wight)
    LoadMonster(Skeleton, count=10, icon=Skeleton)
    LoadMonster(Zombie, count=8, icon=35)
    Combat()
    if CombatEnd > PartyDied { goto EjectToStreet }
    exit

    GhoulsAttack:
    Morale = 100
    SetupMonster(Ghoul, distance=2, closeup=Ghoul)
    Text("SOME GHOULS DECIDE TO ADD YOU TO THEIR MENU.")
    PressToContinue()
    ClearMonsters()
    LoadMonster(Wight, count=1, icon=Wight)
    LoadMonster(Ghoul, count=5, icon=Ghoul)
    Combat()
    if CombatEnd > PartyDied { goto EjectToStreet }
    exit

    ScorpionsApproach:
    SetupMonster(Scorpion, distance=1, closeup=Scorpion)
    Morale = 50
    Text("SCORPIONS APPROACH HUNGRILY.")
    PressToContinue()
    ClearMonsters()
    LoadMonster(LargeScorpion, count=3, icon=18)
    LoadMonster(HugeScorpion, count=1, icon=18)
    Combat()
    if CombatEnd > PartyDied { goto EjectToStreet }
    exit

CourtyardAmbush:
    if HobgoblinTempleState != DrumFightPending { exit }
    SetupMonster(Hobgoblin, distance=0, closeup=Hobgoblin)
    Text("YOU ARE ENTERING A WEEDY, GARBAGE STREWN COURTYARD. A DRUM BEGINS TO BEAT AND HOBGOBLINS ATTACK.")
    Morale = 60
    PressToContinue()
    ClearMonsters()
    LoadMonster(Hobgoblin, count=12, icon=Hobgoblin)
    LoadMonster(HobgoblinLdr, count=2, icon=HobgoblinLdr)
    Combat()
    if CombatEnd > PartyDied { goto ChasedOut }
    HobgoblinTempleState = CourtyardWon
    Text("THINGS SEEM STRANGELY QUIET. THE DRUM HAS FADED TO SILENCE.")
    exit

    ChasedOut:
    Text("YOU ARE CHASED OUT INTO THE STREETS.")

EjectToStreet:
    Scratch2 = Random(1)
    goto [EjectWest, EjectEast][Scratch2]

EjectWest:
    PartyX = 4
    PartyY = 2
    exit

EjectEast:
    PartyX = 12
    PartyY = 2
    exit

    AuxMember = 3

MarkBarracksSeen:
    BarracksSeen = Yes
    exit

MarkTapestrySeen:

    AuxParlayChoice = Menu("SEARCH", "DESTROY", "LEAVE")

Kitchen:

SmallDarkRoom:
    Text("YOU ARE IN A VERY SMALL, DARK ROOM.")
    if PrisonerState == DealtWith { exit }

PrisonerMenu:

LeavePrisoner:

    AuxParlayChoice = Menu("UNLOCK", "PICK LOCK", "BREAK CHAINS", "LEAVE")

    KeyOpensLocks:

    goto PrisonerContinue

def FindSkullcrusher() {
    Scratch2 = 0
    Scratch3 = 0

    FindSkullcrusherLoop:
    SelectMember(Scratch2)
    if $SelName == "SKULLCRUSHER" { goto SkullcrusherFound }
    Scratch2 += 1
    if Scratch2 != 9 { goto FindSkullcrusherLoop }
    goto FindSkullcrusherDone

    SkullcrusherFound:
    if SelStatus != Okay { goto FindSkullcrusherDone }
    Scratch3 = 255

    FindSkullcrusherDone:
    return
}

def PressToContinue() {
    Scratch1 = Menu("PRESS <RETURN> OR BUTTON TO CONTINUE.")
    return
}

ThievesWell:
    Text("ON THE WELL IS A NOTE,'THIEVES ONLY.' THE WELL HAS NO RUNGS OR ROPE AND SEEMS VERY DEEP. ONLY A SKILLFUL CLIMBER COULD SURVIVE THE DESCENT. WHAT DO YOU DO?")
    Scratch1 = Menu("LEAVE", "CLIMB DOWN")
    goto [Done, ChooseClimber][Scratch1]

ChooseClimber:
    Who("WHO WILL CLIMB DOWN?")
    if SelStatus == Okay { goto ClimbDown }
    Text("THAT ONE IS NOT WELL ENOUGH TO CLIMB.")
    PressToContinue()
    goto ChooseClimber

    ClimbDown:
    PartyX = 0
    Facing = East
    CallRedraw()
    PressToContinue()
    Scratch1 = Random(99)
    if Scratch1 < SelThiefSkill6 { goto MeetRestal }
    Text("YOU BEGIN A LONG FALL.")
    PressToContinue()
    Scratch1 = 0

FeatherFallScan:
    Scratch2 = EffectTable[Scratch1]
    if Scratch2 == 27 { goto FeatherFallCheck }

FeatherFallNext:
    Scratch1 += 1
    if Scratch1 == 64 { goto HitStoneFloor }
    goto FeatherFallScan

FeatherFallCheck:
    Scratch2 = 64 + Scratch1
    Scratch3 = EffectTable[Scratch2]
    if Scratch3 != SelPartyIndex2 { goto FeatherFallNext }
    Text("YOU FLOAT DOWN THE WELL.")
    PressToContinue()
    Text("FROM AN ARCHWAY AT THE BOTTOM COME ARROWS.")
    PressToContinue()
    Damage(8, 1, 6, 0, 50)
    goto LaughingThief

HitStoneFloor:
    Text("YOU HIT AN UNYIELDING STONE FLOOR.")
    PressToContinue()
    Damage(160, 15, 6, 0, 128)
    if SelStatus == Okay { goto LaughingThief }
    PartyX = 1
    Facing = West
    CallRedraw()
    Text("YOUR FRIEND FLOATS OUT OF THE WELL.")
    exit

LaughingThief:
    ShowThief()
    Text("HERE IS A LAUGHING THIEF.")
    RingLiftsYou()
    LeaveWell()
    exit

MeetRestal:
    ShowThief()
    Text("AN OLD THIEF STANDS HERE.'WELCOME TO MY GUILD. I AM RESTAL.' HE BOWS LOW.")
    PressToContinue()
    if CadornaCommission != CommissionGiven { goto RestalFarewell }
    if IronBoxState == 0 { goto TextileHouseOffer }
    if QuestCadornaPayment == Rewarded { goto RestalFarewell }
    if RestalDealDone == 1 { goto RestalFarewell }
    Text("'WE CAN DUPLICATE THE SEAL ON THAT CHEST -- FOR HALF THE TREASURE -- NO ONE WILL EVER KNOW. INTERESTED?'")
    Scratch1 = Menu("YES", "NO")
    goto [DuplicateSeal, RestalFarewell][Scratch1]

    IronBoxState = CarriedUnopened
