    Scratch2 = 1
    ScaleCost()

    Scratch2 = 9
    ScaleCost()
    Scratch10 = 189
    if TakeCoinKind() { return }
    Scratch2 = 9

def ScaleCost() {
    &Scratch6 = &Scratch4
    goto ScaleCostLoop

    ScaleCostLoop:
    &Scratch4 += &Scratch6
    Scratch2 -= 1
    if Scratch2 > 0 { goto ScaleCostLoop }
    return
}

def TakeCoinKind() {
    Scratch1 = 0
    goto TakeFromMember
}

TakeFromMember:
    SelectMember(Scratch1)
    if CheckMemberOk() { goto NextCoinMember }
    Scratch6 = SelName[Scratch10]
    Scratch7 = SelName2[Scratch10]
    &Scratch8 = &Scratch6
    if &Scratch8 < &Scratch4 { goto DeductCoins }
    &Scratch8 = &Scratch4
    goto DeductCoins

    DeductCoins:
    &Scratch6 -= &Scratch8
    SelName[Scratch10] = Scratch6
    SelName2[Scratch10] = Scratch7
    &Scratch4 -= &Scratch8
    Scratch2 = Scratch1 | 128
    SelectMember(Scratch2)
    if Scratch4 == 0 { return }
    goto NextCoinMember

    NextCoinMember:
    Scratch1 = Scratch1 + 1
    if Scratch1 < 8 { goto TakeFromMember }
    return Scratch4 == 0

def CheckMemberOk() {
    if SelStatus == Died { goto MemberNotOk }
    if SelStatus != Okay { goto MemberOk }
    if SelControl >= 128 { goto MemberNotOk }
    goto MemberOk

    MemberOk:
    return 0 != 0

    MemberNotOk:
    return 0 != 1
}

def CheckAlarmSounding() {
    if AlarmEndHour < 1 { return }
    AuxRoll = Hours
    if Hours == 0 and AlarmHour == 23 { AuxRoll = 24 }
    if AuxRoll < AlarmEndHour { goto AlarmStillOn }
    if TenMinutes < AlarmMinute { goto AlarmStillOn }
    AlarmEndHour = 0
    return 0 < 1

    AlarmStillOn:
    if Search == Searching { goto SayAlarmSounding }
    AuxRoll = Random(9)
    if 0 != AuxRoll { goto AlarmActive }
    goto SayAlarmSounding

    SayAlarmSounding:
    Text("AN ALARM IS SOUNDING.")
    Delay()
    goto AlarmActive

    AlarmActive:
    return 1 < 0
}

def StartAlarm() {
    if AlarmEndHour == 0 { Text("AN ALARM STARTS RINGING!") }
    AlarmHour = Hours
    AlarmMinute = TenMinutes
    AlarmEndHour = 1 + AuxCounter
    Delay()
    return
}

    Surprise = NoSurprise
    Morale = 80
    StartAlarm()
    ClearMonsters()
    LoadMonster(Level6Mu24, count=2, icon=Level6Mu)
    LoadMonster(Aides, count=2, icon=Aides)
    LoadMonster(Ettin, count=6, icon=Ettin)
    Combat()
    if CombatEnd == PartyFled { goto PushedBack }
    goto GarrisonSlain

MarkWestTowerUsed:
    GateFlags |= SouthDoorCleared
    exit

MarkEastTowerUsed:
    GateFlags |= NorthDoorCleared
    exit
