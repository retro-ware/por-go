	alias BlockedTerrainEntry = Scratch2
	if Scratch1 == 8: Return
	BlockedTerrainEntry = BlockedTerrain[Scratch1]
	if ProbeWall == BlockedTerrainEntry {
		HoldMove = 255
		Return
	}
	Scratch1 += 1
	Goto CheckTerrainLoop

	Return

	Return

PickEncounter:
	Scratch1 = Random 20
	if Scratch1 == 20 && ForceEncounter25 == 0: Return
	Scratch1 = Random 19
	OnGosub Scratch1 PickBoars PickBoars PickBoars PickMantis PickTigers PickTigers PickTigers PickThriKreens PickThriKreens PickAnkheg PickCentaurs PickCentaurs PickSoldiers PickSoldiers PickThieves BuccaneerBaseSite PickQuicklings PickQuicklings PickSnakes PickSnakes
	AuxTemp9 = EncData9[EncounterType25]
	AuxMember = EncData2[EncounterType25]
	AuxRoll = EncData3[EncounterType25]
	AuxMonster = EncMonster[EncounterType25]
	AuxEncResult = EncData5[EncounterType25]
	AuxEncResult2 = EncMorale[EncounterType25]
	AuxEncTemp = EncData7[EncounterType25]
	AuxPartyStrength = EncMonster2[EncounterType25]
	AuxCounter = EncStrength[EncounterType25]
	PartyStrength Scratch1
	Scratch1 = Scratch1 / AuxCounter
	Scratch2 = Random 5
	OnGoto Scratch2 GroupSizeLarge GroupSizeMedium GroupSizeMedium GroupSizeMedium SplitGroups

GroupSizeLarge:
	Scratch1 = Scratch1 / 3
	if Scratch1 <= 0: Scratch1 = 1
	Goto SplitGroups

GroupSizeMedium:
	Scratch1 = Scratch1 / 2
	if Scratch1 <= 0: Scratch1 = 1

SplitGroups:
	EncounterExtraCount25 = 0
	Scratch3 = Scratch1
	AuxMenuChoice = EncGroupSize[EncounterType25]
	if AuxMenuChoice == 0: Goto CapGroupSize

SplitGroupsLoop:
	if Scratch3 < AuxMenuChoice: Goto CapGroupSize
	EncounterExtraCount25 += 1
	Scratch3 -= AuxMenuChoice
	Goto SplitGroupsLoop

CapGroupSize:
	if Scratch1 >= 56: Scratch1 = 56 - Scratch2
	EncounterCount25 = Scratch1
	Return
