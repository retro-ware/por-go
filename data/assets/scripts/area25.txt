on move {
    if InLair == InsideLair { exit }
    if MapY <= 2 { EdgeNorth() }
    if MapX <= 2 { EdgeWest() }
    if MapX == 15 { EdgeEast() }
    CheckTerrainAhead()
    exit
}

def EdgeNorth() {
    goto [BlockMove, BlockMove, EdgeOk, EdgeOk, EdgeOk, EdgeOk, EdgeOk, BlockMove][FacingRaw]
} # falls into EdgeWest

EdgeWest:
    goto [EdgeOk, EdgeOk, EdgeOk, EdgeOk, EdgeOk, BlockMove, BlockMove, BlockMove][FacingRaw]

BlockMove:
    HoldMove = SkipMove
    return

def EdgeEast() {
    goto [EdgeOk, ExitToCity, ExitToCity, EdgeEastCorner, EdgeOk, EdgeOk, EdgeOk, EdgeOk][FacingRaw]
} # falls into EdgeOk

EdgeOk:
    return

ExitToCity:
    MapX = 3
    EnterArea(Wilderness)

EdgeEastCorner:
    if MapY != 31 { goto ExitToCity }
    HoldMove = SkipMove
    return

def CheckTerrainAhead() {
    ProbeX = MapX
    ProbeY = MapY
    gosub [ProbeN, ProbeNE, ProbeE, ProbeSE, ProbeS, ProbeSW, ProbeW, ProbeNW][FacingRaw]
    CallUnknownC01B()
    Scratch1 = 0

    CheckTerrainLoop:
    if Scratch1 == 8 { return }
    Scratch2 = BlockedTerrain[Scratch1]
    if ProbeWall == Scratch2 {
        HoldMove = SkipMove
        return
    }
    Scratch1 += 1
    goto CheckTerrainLoop
}

def ProbeNW() {
    ProbeX -= 1
} # falls into ProbeN

ProbeN:
    ProbeY -= 1
    return

def ProbeNE() {
    ProbeY -= 1
} # falls into ProbeE

ProbeE:
    ProbeX += 1
    return

def ProbeSE() {
    ProbeX += 1
} # falls into ProbeS

ProbeS:
    ProbeY += 1
    return

def ProbeSW() {
    ProbeY += 1
} # falls into ProbeW

ProbeW:
    ProbeX -= 1
    return

on step {
    if InLair == InsideLair { goto LairCell }
    Scratch1 = 0
    Scratch4 = 0

    FindSiteRow:
    if Scratch1 >= 8 { goto RollEncounter }
    Scratch3 = SiteRowEnd[Scratch1]
    Scratch2 = SiteRowY[Scratch1]
    if Scratch2 == MapY { goto SiteRowFound }
    Scratch1 += 1
    Scratch4 += Scratch3
    goto FindSiteRow

    SiteRowFound:
    Scratch3 += Scratch4

    FindSiteX:
    if Scratch4 >= Scratch3 { goto RollEncounter }
    Scratch5 = SiteX[Scratch4]
    if Scratch5 == MapX { goto SiteFound }
    Scratch4 += 1
    goto FindSiteX

    SiteFound:
    Scratch1 = SiteKind[Scratch4]
    goto [SiteNone, DragonInDistance, DragonCaveEvent, AnkhegAmbush, SlaverCaravan, BuccaneerBaseSite,
        GuardsRideBy, OutpostRiders][Scratch1]
} # falls into SiteNone

SiteNone:
    exit

    goto [SpotGroup, PartySurprised, SurpriseGroup, FaceToFace][Surprise]

    Print($AuxMonsterName)

        flee=Attack, advance=Mirror, parlay=Talk, escape=AuxEncTemp, strength=AuxEncTemp)

    Print(" FALL FOR YOUR SPARKLING PERSONALITY AND WANDER AWAY WITHOUT A FIGHT.")

def PickEncounter() {
    Scratch1 = Random(20)
    if Scratch1 == 20 and ForceEncounter25 == 0 { return }
    Scratch1 = Random(19)
    gosub [PickBoars, PickBoars, PickBoars, PickMantis, PickTigers, PickTigers, PickTigers,
        PickThriKreens, PickThriKreens, PickAnkheg, PickCentaurs, PickCentaurs, PickSoldiers,
        PickSoldiers, PickThieves, BuccaneerBaseSite, PickQuicklings, PickQuicklings, PickSnakes,
        PickSnakes][Scratch1]
    AuxTemp9 = EncData9[EncounterType25]
    AuxMember = EncData2[EncounterType25]
    AuxRoll = EncData3[EncounterType25]
    AuxMonster = EncMonster[EncounterType25]
    AuxEncResult = EncData5[EncounterType25]
    AuxEncResult2 = EncMorale[EncounterType25]
    AuxEncTemp = EncData7[EncounterType25]
    AuxPartyStrength = EncMonster2[EncounterType25]
    AuxCounter = EncStrength[EncounterType25]
    Scratch1 = PartyStrength()
    Scratch1 = Scratch1 / AuxCounter
    Scratch2 = Random(5)
    goto [GroupSizeLarge, GroupSizeMedium, GroupSizeMedium, GroupSizeMedium, SplitGroups][Scratch2]
} # falls into GroupSizeLarge

GroupSizeLarge:
    Scratch1 = Scratch1 / 3
    if Scratch1 <= 0 { Scratch1 = 1 }
    goto SplitGroups

GroupSizeMedium:
    Scratch1 = Scratch1 / 2
    if Scratch1 <= 0 { Scratch1 = 1 }

SplitGroups:
    EncounterExtraCount25 = 0
    Scratch3 = Scratch1
    AuxMenuChoice = EncGroupSize[EncounterType25]
    if AuxMenuChoice == 0 { goto CapGroupSize }

    SplitGroupsLoop:
    if Scratch3 < AuxMenuChoice { goto CapGroupSize }
    EncounterExtraCount25 += 1
    Scratch3 -= AuxMenuChoice
    goto SplitGroupsLoop

    CapGroupSize:
    if Scratch1 >= 56 { Scratch1 = 56 - Scratch2 }
    EncounterCount25 = Scratch1
    return
