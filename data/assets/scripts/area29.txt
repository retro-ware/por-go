event0:

	Gosub UpdateSky
	if LeftMap == 0: Goto CheckEntry

	HoldMove = 255
	SymbolSet = ExitSymbolSetByFacing[Facing]

CheckEntry:
	Scratch10 = 127 & Cell
	if Scratch10 != 1: Goto CheckSecretDoorBlock
	GroundKind = 0
	Gosub UpdateSky

CheckSecretDoorBlock:

	GroundKind = 1

	Call CallLookAhead

	PartyY = Scratch4

	Scratch2 = 1

	if CombatEnd > 128: Goto DropOff
	if CombatEnd < 1: Goto CountEncounter
	if Beaten == 0: Exit

	if EncounterCount != 255: EncounterCount += 1
	Goto Done

EncounterFight:
	PartyStrength AuxPartyStrength
	AuxTemp9 = 0
	Morale = AuxEncTemp
	if Scratch7 < 1: AuxPartyStrength = AuxPartyStrength / 3
	also if Scratch7 > 1: AuxPartyStrength = AuxPartyStrength / 4
	also if Scratch7 != 1: Goto LoadEncounter
	AuxPartyStrength = AuxPartyStrength / 2
	AuxTemp9 = 3

	if AuxPartyStrength == 0: AuxPartyStrength = 1
	ClearMonsters
	if AuxTemp9 != 0: LoadMonster KoboldLeader1 count=3 icon=KoboldLeader
	LoadMonster Scratch8 count=AuxPartyStrength icon=Scratch8
	Combat
	Goto AfterCombat
