event0:
	Gosub UpdateSky
	if LeftMap == 0: Goto CheckEntry
	Call CallStepForward
	Save 255 HoldMove
	GetTable ExitSymbolSetByFacing Facing SymbolSet
	GetTable ExitAreaByFacing Facing Scratch10
	Save 0 NoCoords
	NewECL Scratch10

CheckEntry:
	And 127 Cell Scratch10
	if Scratch10 != 1: Goto CheckSecretDoorBlock
	Save 0 GroundKind
	Gosub UpdateSky

CheckSecretDoorBlock:
	if SecretDoorFound == 255: Goto CheckNailedDoor
	if Scratch10 == 6 && Facing == 0: Save 255 HoldMove
	if Scratch10 == 7 && Facing == 2: Save 255 HoldMove

CheckNailedDoor:
	if Scratch10 == 21 && Facing == 2: Goto NailedDoor
	GetTable ExitCellByFacing Facing Scratch9
	if Scratch9 != Scratch10: Goto CheckWellAmbush
	And WildFlags 1 AuxMember
	IfEq
	Goto WellPlaza
	Save 1 GroundKind
	Gosub UpdateSky

CheckWellAmbush:
	Gosub ExitIfWellCleared
	Save PartyX Scratch2
	Save PartyY Scratch4
	Call CallStepForward
	Call CallLookAhead
	And Cell 127 Scratch1
	Save Scratch2 PartyX
	Save Scratch4 PartyY
	Call CallLookAhead
	if Scratch1 != 3: Exit
	Save 1 Scratch4
	Save 1 Scratch2
	Goto WaitingKobolds

event1:
	Save 0 EncounterMode
	PrintClear ""
	And 127 Cell Scratch10
	Save 0 Scratch1
	Save 2 Scratch2
	GetTable CellEventTable Scratch10 AuxCounter
	OnGoto AuxCounter 16 Done RandomEncounterRoll WellPlaza KutosWell WellApproach PlazaWander SicklyKobolds LizardPatrol SeerWoman NorthWallDoor SouthWallDoor HiddenCatacombs CatacombSniper1 CatacombSniper2 BanditTreasure NorrisBandits

KutosWell:
	if InWell != 0: Goto WellSecretDoor
	if Scratch10 == 2: Save 0 AuxEncChoice
	if Scratch10 == 5: Save 2 AuxEncChoice
	if Scratch10 == 16: Save 3 AuxEncChoice
	if Scratch10 == 17: Save 1 AuxEncChoice
	if AuxEncChoice != Facing: Goto NearWell
	PrintClear "BEFORE YOU LIES KUTO'S WELL.  IT IS A SOURCE OF WATER IN ALL SEASONS."
	Goto ArrowRain

	Return

NearWell:

ArrowRain:
	Gosub ExitIfWellCleared
	PartySurprise AuxMember AuxRoll
	Surprise AuxMember AuxRoll 6 2
	if Search != 1: Exit
	Damage 9 1 6 0 41
	PrintClear "A RAIN OF ARROWS POURS OVER YOU."
	if Surprise != 1: Goto MonstersAttack
	Exit

RandomEncounterRoll:
	if InWell != 0: Exit
	if EncounterCount > 10: Exit
	if 0 != Scratch10: Exit
	Random 10 AuxSightText
	if Search == 1: Add 2 AuxSightText AuxSightText
	if AuxSightText > 9: Goto CheckEncounterCount
	Exit

event2:
	Call CallLookAhead
	Save 0 RestEvery
	Save 100 RestChance
	if InWell != 1: Goto PlazaRest
	if Cell == 16: Exit
	if BanditsDone != 255: Save 1 RestEvery
	Exit

PlazaRest:
	if EncounterCount >= 10: Exit
	if Cell == 1: Exit
	Save 12 RestEvery
	Save 12 RestChance
	Exit

event3:
	if InWell != 0: Goto NorrisBandits
	Save 200 EncounterMode
	Save 0 Surprise
	Goto RandomEncounter

event4:
	Gosub UpdateSky
	Save 0 NoCoords
	if 1 == InWell: Goto LoadWellFiles
	LoadFiles 29 2 255
	LoadPieces 3 20 1
	Exit

LoadWellFiles:
	LoadFiles 32 2 255
	LoadPieces 18 17 1
	Exit

SouthWallDoor:
	Save "SOUTH WALL." $AuxMonsterName
	Goto SearchSecretDoor

NorthWallDoor:
	Save "NORTH WALL." $AuxMonsterName

SearchSecretDoor:
	if SecretDoorFound == 255: Goto SecretDoorSpotted
	Save 0 AuxMember

SearchMemberLoop:
	LoadCharacter AuxMember
	if SelStatus != 1: Goto NextSearcher
	GetTable SecretDoorChanceByRace SelRace AuxSearchCount
	Add Search AuxSearchCount AuxSearchCount
	Random 5 AuxRoll
	if AuxSearchCount > AuxRoll: Goto SecretDoorSpotted

NextSearcher:
	Add 1 AuxMember AuxMember
	if AuxMember < 8: Goto SearchMemberLoop
	Exit

SecretDoorSpotted:
	PrintClear "YOU FIND A SECRET DOOR IN THE "
	Print $AuxMonsterName
	Save 255 SecretDoorFound
	Exit

NorrisBandits:
	if BanditsDone == 255: Exit
	Save 10 Portrait
	SetupMonster 16 2 6
	Delay
	Approach
	Delay
	Approach
	PrintClear "YOU ARE SURROUNDED BY THE BANDIT BAND OF THE INFAMOUS NORRIS THE  GRAY.  'YOU DARE TO INVADE MY HALLS?', HE SNARLS. 'SURRENDER OR DIE!'"
	Gosub PressToContinue
	PrintClear "WHAT DO YOU DO?"
	HMenu AuxMember 2 "SURRENDER" "FIGHT"
	Save 255 Portrait
	OnGoto AuxMember 1 SurrenderToBandits

	LoadMonster 32 1 7
	LoadMonster 57 5 57
	LoadMonster 1 9 1

	Gosub PressToContinue
	PrintClear "YOU HAVE DEFEATED THE DREADED BANDIT GANG AND RID PHLAN OF A GREAT EVIL.  THE HONEST SETTLERS WILL CHEER YOUR DEEDS!"
	Gosub PressToContinue
	PrintClear "YOU HAVE ALSO TAKEN POSSESSION OF A HIDEOUT IN WHICH YOU MAY REST UNDISTURBED FROM THE RIGORS OF YOUR QUESTING."
	Save 250 BanditTreasureState
	Save 0 AuxMember
	Gosub PauseLoop
	Exit

	RedrawView
	PrintClear "YOU ARE STRIPPED OF ALL VALUABLES AND BLINDFOLDED.  THE BANDITS RELEASE YOU SOMEWHERE ABOVE GROUND."
	Rob 1 100 100
	Save 255 BanditsDone
	Save 255 BanditTreasureState

	Save 0 GroundKind
	Gosub UpdateSky
	LoadFiles 29 2 255

	Gosub ShowKobolds
	PrintClear "MONSTERS FROM THE SURROUNDING BUILDINGS ATTACK YOU."

WellFight:
	Gosub ExitIfWellCleared
	Gosub PressToContinue
	Save Scratch2 MaxDist
	Save 67 Morale
	ClearMonsters
	LoadMonster 0 6 0
	LoadMonster 1 3 1
	Combat
	if CombatEnd >= 128: Goto AfterCombat
	if Scratch1 == 250: Goto WellCleared
	if Scratch4 < 1: PrintClear "A BAND OF KOBOLDS CLIMBS FROM THE WELL AND ATTACKS."
	also if Scratch4 == 1: PrintClear "A BAND OF KOBOLDS CLIMBS FROM THE WELL BEHIND YOU AND ATTACKS."
	SetupMonster 0 2 0
	Delay
	Approach
	Delay
	Approach
	Save 25 AuxMember
	Gosub PauseLoop
	Save 250 Scratch1
	Goto WellFight

WellCleared:

	Exit

CheckEncounterCount:

	Save 255 EncounterCount
	Exit

RandomEncounter:
	Random 2 Scratch7
	OnGoto Scratch7 3 NameGnolls NameKobolds NameLizardmen
	GetTable EncMonsterDistTable Scratch7 MonsterDist
	GetTable EncTable2 Scratch7 Scratch8
	GetTable EncTable3 Scratch7 AuxEncResult
	GetTable EncTable4 Scratch7 AuxEncResult2
	GetTable EncTable5 Scratch7 AuxEncMonster2
	GetTable EncTable6 Scratch7 AuxEncTemp
	if EncounterMode == 200: Goto FaceToFace
	PartySurprise AuxMember AuxRoll
	Surprise AuxMember AuxRoll 0 0
	OnGoto Surprise 4 BloodthirstyMonsters SurprisedBy SurprisedParty FaceToFace

	SetupMonster AuxEncMonster2 2 Scratch8
	Delay
	Approach
	Delay
	Approach
	PrintClear "YOU ARE SURPRISED BY "
	Print $AuxText
	Gosub PressToContinue
	Save 0 MaxDist
	Gosub EncounterFight
	Goto AfterCombat

	Random 1 AuxMember
	if AuxMember < 1: Goto AvoidedBy
	Save "YOU SEE SOME BLOODTHIRSTY " $AuxMonsterName
	Save $AuxText $AuxNameCopy2
	EncounterMenu AuxEncMonster2 MonsterDist Scratch8 AuxMember 0 0 0 1 0 $AuxMonsterName "" "" AuxEncResult AuxEncResult2
	Add 9 AuxEncTemp AuxEncTemp
	OnGoto AuxMember 4 Done EncounterFight DropOff EncounterFight

FaceToFace:
	Save "YOU ARE SUDDENLY FACE TO FACE WITH ARMED " $AuxMonsterName
	Save $AuxText $AuxNameCopy6
	EncounterMenu AuxEncMonster2 0 Scratch8 AuxMember 0 2 0 1 4 $AuxMonsterName "" "" AuxEncResult AuxEncResult2
	OnGoto AuxMember 4 Done EncounterFight DropOff ParlayMonsters

AvoidedBy:
	Save "YOU ARE BEING AVOIDED BY SOME " $AuxMonsterName
	Save $AuxText $AuxNameCopy3
	EncounterMenu AuxEncMonster2 MonsterDist Scratch8 AuxMember 0 2 2 1 4 $AuxMonsterName "" "" AuxEncResult AuxEncResult2
	Sub 10 AuxEncTemp AuxEncTemp
	OnGoto AuxMember 4 Done EncounterFight DropOff ParlayMonsters

SurprisedParty:

	Save $AuxText $AuxNameCopy3
	EncounterMenu AuxEncMonster2 MonsterDist Scratch8 AuxMember 0 2 2 1 4 $AuxMonsterName "" "" AuxEncResult AuxEncResult2
	Sub 20 AuxEncTemp AuxEncTemp
	OnGoto AuxMember 4 Done FightSurprised Done ParlayMonsters

	And WildFlags 4 AuxMember
	IfNe
	Exit
	if Search != 1: Exit
	PrintClear "BENEATH A RUG YOU FIND A COMPARTMENT CONTAINING ARMS AND ARMOR.  DO YOU TAKE THEM?"
	Gosub YesOrExit
	ClearMonsters
	Treasure 0 0 0 0 0 0 0 58

	Or WildFlags 4 WildFlags
	Exit

AfterCombat:
	if CombatEnd > 128: Goto DropOff
	if CombatEnd < 1: Goto CountEncounter
	if Beaten == 0: Exit

CountEncounter:
	if EncounterCount != 255: Add 1 EncounterCount EncounterCount
	Goto Done

FightSurprised:
	Save 2 Surprise

EncounterFight:
	PartyStrength AuxPartyStrength
	Save 0 AuxTemp9
	Save AuxEncTemp Morale
	if Scratch7 < 1: Div AuxPartyStrength 3 AuxPartyStrength
	also if Scratch7 > 1: Div AuxPartyStrength 4 AuxPartyStrength
	also if Scratch7 != 1: Goto LoadEncounter
	Div AuxPartyStrength 2 AuxPartyStrength
	Save 3 AuxTemp9

LoadEncounter:
	if AuxPartyStrength == 0: Save 1 AuxPartyStrength
	ClearMonsters
	if AuxTemp9 != 0: LoadMonster 1 3 1
	LoadMonster Scratch8 AuxPartyStrength Scratch8
	Combat
	Goto AfterCombat

DropOff:
	Random 3 Facing
	Random 7 AuxTemp9
	Save 1 NoCoords
	Save 1 NoCoords
	GetTable DropOffXTable AuxTemp9 PartyX
	GetTable DropOffYTable AuxTemp9 PartyY
	PrintClear ""
	Call CallRedraw
	Goto event1

Done:
	RedrawView
	Save 255 Portrait
	Call CallRedraw
	Exit

ParlayMonsters:
	Save 0 Surprise
	Parlay 10 20 20 20 30 AuxParlay
	Random 30 AuxRoll2
	Add AuxParlay AuxRoll2 AuxParlay
	Add AuxParlay SelReaction AuxParlay
	if AuxParlay < 95: Goto MonstersOffended
	PrintClear "'WE HAVE TO KEEP THIS AREA FREE OF INTRUDERS -- BOSS'S ORDERS.  WE'LL LET YOU GO THIS TIME.  BUT IF YOU COME BACK, YOU'LL FACE BIG TROUBLE.' THE MONSTERS LEAVE."
	Goto Done

MonstersOffended:
	PrintClear "YOU HAVE OFFENDED THE "
	Print $AuxText
	Print "THEY ATTACK."
	if AuxParlay < 65: Save 99 AuxEncTemp
	Goto EncounterFight

	if Hours >= 14: Save 8 Sky
	also if Hours < 14: Save 11 Sky
	if InWell != 1 {
		Save 10 SkyInside
		Save 65 SetSkyShade
	}
	also if InWell == 1 {
		Save 8 SetSkyShade
		Save 8 SkyInside
	}
	GetTable GroundShadeTable GroundKind SetGroundShade
	Return

	Exit
