on move {
    UpdateSky()
    if LeftMap == 0 { goto CheckEntry }
    CallStepForward()
    HoldMove = SkipMove
    SymbolSet = ExitSymbolSetByFacing[Facing]
    Scratch10 = ExitAreaByFacing[Facing]
    NoCoords = 0
    NewECL(Scratch10)

    CheckEntry:
    Scratch10 = ~Inside & Cell
    if Scratch10 != 1 { goto CheckSecretDoorBlock }
    GroundKind = Surface
    UpdateSky()

    CheckSecretDoorBlock:
    if SecretDoorFound == Yes { goto CheckNailedDoor }
    if Scratch10 == 6 and Facing == North { HoldMove = SkipMove }
    if Scratch10 == 7 and Facing == South { HoldMove = SkipMove }

    CheckNailedDoor:
    if Scratch10 == 21 and Facing == South { goto NailedDoor }
    Scratch9 = ExitCellByFacing[Facing]
    if Scratch9 != Scratch10 { goto CheckWellAmbush }
    if AuxMember = WildFlags & WellRungs; AuxMember == 0 { goto WellPlaza }
    GroundKind = ByTheWell
    UpdateSky()

    CheckWellAmbush:
    ExitIfWellCleared()
    Scratch2 = PartyX
    Scratch4 = PartyY
    CallStepForward()
    CallLookAhead()
    Scratch1 = Cell & ~Inside
    PartyX = Scratch2
    PartyY = Scratch4
    CallLookAhead()
    if Scratch1 != 3 { exit }
    Scratch4 = 1
    Scratch2 = 1
    goto WaitingKobolds
}

on step {
    L_9A23:
    EncounterMode = 0
    Text("")
    Scratch10 = ~Inside & Cell
    Scratch1 = 0
    Scratch2 = 2
    AuxCounter = CellEventTable[Scratch10]
    goto [Done, RandomEncounterRoll, WellPlaza, KutosWellEvent, WellApproach, PlazaWander,
        SicklyKobolds, LizardPatrol, SeerWoman, NorthWallDoor, SouthWallDoor, HiddenCatacombs,
        CatacombSniper1, CatacombSniper2, BanditTreasure, NorrisBandits][AuxCounter]
    goto KutosWellEvent
}

KutosWellEvent:
    if InWell != 0 { goto WellSecretDoor }
    if Scratch10 == 2 { AuxEncChoice = 0 }
    if Scratch10 == 5 { AuxEncChoice = 2 }
    if Scratch10 == 16 { AuxEncChoice = 3 }
    if Scratch10 == 17 { AuxEncChoice = 1 }
    if AuxEncChoice != Facing { goto NearWell }
    Text("BEFORE YOU LIES KUTO'S WELL.  IT IS A SOURCE OF WATER IN ALL SEASONS.")
    goto ArrowRain

    if AuxSightText > 9 { goto CheckEncounterCount }

    RestChance = 100

    NextSearcher:
    AuxMember += 1

    Combat()

    if Scratch1 == 250 { goto WellCleared }
    if Scratch4 < 1 {
        Text("A BAND OF KOBOLDS CLIMBS FROM THE WELL AND ATTACKS.")
    } else if Scratch4 == 1 {

    AuxEncResult2 = EncTable4[Scratch7]

    goto [Done, EncounterFight, DropOff, EncounterFight][AuxMember]

        strength=AuxEncResult2)

    goto [Done, EncounterFight, DropOff, ParlayMonsters][AuxMember]

SeerWoman:
    if AuxMember = WildFlags & SpiritWarning; AuxMember != 0 { goto HiddenArmor }
    Picture(WomanInGreen)
    Text("A WIDE-EYED WOMAN IS SEATED ON A RUG.  AT YOUR ENTRY, SHE STANDS. 'GREETINGS BOLD ONES, I HAVE LONG AWAITED YOUR COMING.  MY TIME HERE IS SHORT FOR THE WORLD ENTERS A NEW AGE.")

    WildFlags |= SpiritWarning

    if AuxMember = WildFlags & SearchedTreasure; AuxMember != 0 { exit }
    if Search != Searching { exit }
    Text("BENEATH A RUG YOU FIND A COMPARTMENT CONTAINING ARMS AND ARMOR.  DO YOU TAKE THEM?")
    YesOrExit()
    ClearMonsters()
    Treasure(items=[BracersAc4, QuarterStaff1, BandedMail1])
    Combat()
    WildFlags |= SearchedTreasure
    exit

AfterCombat:
    if CombatEnd > PartyDied { goto DropOff }
    if CombatEnd < 1 { goto CountEncounter }
    if Beaten == 0 { exit }

    CountEncounter:
    if EncounterCount != 255 { EncounterCount += 1 }
    goto Done

FightSurprised:
    Surprise = MonstersAreSurprised

EncounterFight:
    AuxPartyStrength = PartyStrength()
    AuxTemp9 = 0
    Morale = AuxEncTemp
    if Scratch7 < 1 {
        AuxPartyStrength = AuxPartyStrength / 3
    } else if Scratch7 > 1 {
        AuxPartyStrength = AuxPartyStrength / 4
    } else if Scratch7 != 1 {
        goto LoadEncounter
    }
    AuxPartyStrength = AuxPartyStrength / 2
    AuxTemp9 = 3

    LoadEncounter:
    if AuxPartyStrength == 0 { AuxPartyStrength = 1 }
    ClearMonsters()
    if AuxTemp9 != 0 { LoadMonster(KoboldLeader1, count=3, icon=KoboldLeader) }
    LoadMonster(Scratch8, count=AuxPartyStrength, icon=Scratch8)
    Combat()
    goto AfterCombat

DropOff:
    Facing = Random(3)
    AuxTemp9 = Random(7)
    NoCoords = 1
    NoCoords = 1
    PartyX = DropOffXTable[AuxTemp9]
    PartyY = DropOffYTable[AuxTemp9]
    Text("")
    CallRedraw()
    goto L_9A23

Done:
    RedrawView()
    Portrait = None
    CallRedraw()
    exit

ParlayMonsters:
    Surprise = NoSurprise
    AuxParlay = Parlay(10, 20, 20, 20, 30)
    AuxRoll2 = Random(30)
    AuxParlay = AuxParlay + AuxRoll2
    AuxParlay = AuxParlay + SelReaction
    if AuxParlay < 95 { goto MonstersOffended }
    Text("'WE HAVE TO KEEP THIS AREA FREE OF INTRUDERS -- BOSS'S ORDERS.  WE'LL LET YOU GO THIS TIME.  BUT IF YOU COME BACK, YOU'LL FACE BIG TROUBLE.' THE MONSTERS LEAVE.")
    goto Done

    MonstersOffended:
    Text("YOU HAVE OFFENDED THE ")
    Print($AuxText)
    Print("THEY ATTACK.")
    if AuxParlay < 65 { AuxEncTemp = 99 }
    goto EncounterFight

def ShortPause() {
    AuxMember = 50
    goto PauseLoop
}

PauseLoop:
    AuxMember += 1
    if AuxMember < 100 { goto PauseLoop }
    return

def PressToContinue() {
    AuxMenuChoice = Menu("PRESS <RETURN> OR BUTTON TO CONTINUE")
    return
}

def UpdateSky() {
    if Hours >= 14 {
        Sky = 8
    } else {
        Sky = 11
    }
    if InWell != DownTheWell {
        SkyInside = 10
        SetSkyShade = 65
    } else {
        SetSkyShade = 8
        SkyInside = 8
    }
    SetGroundShade = GroundShadeTable[GroundKind]
    return
}

def YesOrExit() {
    AuxMember = Menu("YES", "NO")
    if AuxMember == 0 { return }
    exit
}

    SetupMonster(Kobold, distance=Scratch2, closeup=Kobold)
    Delay()
    Approach()
    Delay()
    Approach()
    return
}

def ExitIfWellCleared() {
    if AuxMember = WildFlags & WellRungs; AuxMember != 0 { exit }
    return
}
