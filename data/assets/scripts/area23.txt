def SearchSecretDoor() {
    if Scratch1 has SecretDoorsFound { goto SecretDoorFound }
    Scratch10 = 0
    goto SearchMemberLoop
}

SearchMemberLoop:
    SelectMember(Scratch10)
    if SelStatus == Died { goto NextSearcher }
    Scratch9 = Search
    if SelRace == Elf { Scratch9 += 1 }
    if SelRace == HalfElf { Scratch9 += 1 }
    Scratch8 = Random(5)
    if Scratch8 > Scratch9 { goto NextSearcher }

SecretDoorFound:
    SecretDoorsFound = Scratch1 | SecretDoorsFound
    Text("A SECRET DOOR IS FOUND ")
    Scratch10 = ByFacing9B0F[Facing]
    Scratch2 += Scratch10
    Scratch2 = Table9B08[Scratch2]
    gosub [DoorInFront, DoorToRight, DoorBehind, DoorToLeft][Scratch2]
    goto WaitReturnPressed

NextSearcher:
    Scratch10 = Scratch10 + 1
    if Scratch10 < 8 { goto SearchMemberLoop }
    return
