    CheckFled()
    if Scratch1 == 2 { goto FleeRandomly }
    VampireState = Slain
    QuestGraveyard = RewardDue
    exit

ReloadVampire:
    ClearMonsters()
    LoadMonster(Vampire105, count=1, icon=Vampire)
    goto VampireFightLoop

PitStairs:
    if VampireFought > 0 { exit }
    Text("WOODEN STAIRS LEAD INTO A PIT.  DO YOU GO DOWN?")
    YesNoMenu()
    goto [DescendPit, Done][Scratch1]

DescendPit:
    PartyX = 14
    Facing = West
    CallRedraw()
    if VampireState > Slain { exit }

VampireLair:
    if VampireState >= FledAsVapor { goto VampireTreasure }
    Sky = 15
    CallRedraw()
    Text("THE CHAMBER LIGHTS UP TO REVEAL A TALL, PALE MAN WITH VERY WHITE TEETH.")
    SetupMonster(Vampire, distance=1, closeup=Vampire)
    Approach()
    WaitReturnPressed()
    if MysticManState == MysticJoined {
        Scratch1 = 1
    } else {
        Scratch1 = 0
    }
    ClearMonsters()
    LoadMonster(Vampire23, count=1, icon=Vampire)
    LoadMonster(Wolf, count=50, icon=Wolf)
    if Carried(EfreetiBottle) {
        LoadMonster(Efreeti70, count=1, icon=Efreeti)
        VampireGuardLoaded = Yes
    }
    Combat()
    CheckFled()
    if Scratch1 == 2 { goto FleeRandomly }
    VampireFought += 1
    MysticManState = Finished
    VampireState = FledAsVapor
    if UndeadKillTable > 249 { UndeadKillTable = 249 }
    UndeadKillTable += 5
    if ZombieStrength > 249 { ZombieStrength = 249 }
    ZombieStrength += 5
    if WightStrength > 249 { WightStrength = 249 }
    WightStrength += 5
    LairsCleared += 5
    Sky = 8
    Text("THE VAMPIRE TURNS INTO A VAPOR CLOUD AND FLOATS AWAY.")
    exit

    VampireTreasure:
    if VampireTreasureTaken == Yes { exit }
    if Search != Searching { exit }
    Text("YOU FIND THE VAMPIRE'S TREASURE TROVE.  DO YOU TAKE IT?")
    YesNoMenu()
    goto [TakeVampireTreasure, Done][Scratch1]

TakeVampireTreasure:
    ClearMonsters()
    Treasure(gold=9000, gems=12, items=[ClericalScrollWith2Spells, ClericalScrollWith2Spells, ClericalScrollWith2Spells, ClericalScrollWith2Spells, Shield1, ClericalScrollWith2Spells, PotionOfHealing*3, WandOfMagicMissiles])
    Combat()
    VampireTreasureTaken = Yes
    exit

GhoulPack:
    SetupMonster(Ghoul, distance=2, closeup=Ghoul)
    Approach()
    Delay()
    Approach()
    Text("A PACK OF HUNGRY GHOULS LOPES TOWARD YOU.")
    WaitReturnPressed()
    Scratch1 = Random(2)
    Scratch1 += 1
    Scratch8 = PartyStrength()
    Scratch1 = Scratch8 / Scratch1
    if Scratch1 == 0 { Scratch1 = 1 }
    ClearMonsters()
    LoadMonster(Ghoul, count=Scratch1, icon=Ghoul)
    Combat()
    exit

MysticManHint:
    if Facing == North and MysticManState == MysticJoined {
        Text("THE MYSTIC MAN SPEAKS, 'THE VAMPIRE IS BEHIND THE NORTHERN WALL.'")
    }
    exit

RandomEncounter:
    MonsterDist = GroupDistTable[Scratch9]
    Scratch3 = GroupTable1[Scratch9]
    Scratch4 = GroupTable2[Scratch9]
    Scratch5 = GroupTable3[Scratch9]
    if Scratch6 == 1 { return }
    PartySurprise(Scratch1, Scratch2)
    Surprise(Scratch1, Scratch2, 0, 0)
    if Surprise != PartyIsSurprised { goto SurpriseCheck }
    goto [SkeletonAttack, ZombieAttack, WightAttack][Scratch9]

    SurpriseCheck:
    if Surprise > MonstersAreSurprised {
        goto PartyConfronted
    } else if Surprise == MonstersAreSurprised {
        goto PartySneaksUp
    } else if Surprise < MonstersAreSurprised {
        goto FindUndead
    }

    PartyConfronted:
    MaxDist = 0
    $AuxText = "YOU ARE SUDDENLY CONFRONTED BY "
    $AuxNameCopy7 = $AuxMonsterName
    goto EncounterChoice

    PartySneaksUp:
    $AuxText = "YOU HAVE MANAGED TO SNEAK UP ON SOME "
    $AuxNameCopy8 = $AuxMonsterName
    EncounterMenu(Scratch5, distance=MonsterDist, closeup=Scratch5, result=Scratch1, combat=Mirror,
        wait=Attack, flee=Flee, advance=CloseIn, parlay=Talk, near=$AuxText, escape=0, strength=0)
    goto [Done, StartFight, FleeRandomly, PartyNoticed][Scratch1]

PartyNoticed:
    Surprise = NoSurprise
    Text("YOU ARE NOTICED.")
    goto EncounterChoice

FindUndead:
    Text("YOU FIND SOME UNDEAD.")

EncounterChoice:
    EncounterMenu(Scratch5, distance=MonsterDist, closeup=Scratch5, result=Scratch1, combat=Mirror,
        wait=Mirror, flee=Mirror, advance=Mirror, parlay=Talk, escape=Scratch4, strength=Scratch4)
    goto [Done, StartFight, FleeRandomly, ParlayFails][Scratch1]

ParlayFails:
    Scratch1 = Parlay(0, 0, 0, 0, 0)
    Text("THEY DON'T LISTEN.")

StartFight:
    Beaten = 0
    if Scratch7 == 0 { Scratch7 = 1 }
    ClearMonsters()
    LoadMonster(Scratch3, count=Scratch7, icon=Scratch5)
    if Scratch3 == 20 { goto LoadRandomUndead }

FightGroup:
    Combat()
    CheckFled()
    if Scratch1 == 2 { goto FleeRandomly }
    goto [SkeletonsBeaten, ZombiesBeaten, WightGroupFight, GiantSkeletonStep, GiantZombieStep,
        WightBossStep, BoneTowerSeenSet, FurTowerSeenSet, SkeletonAmbushOver, LightningTowerSeenSet][Scratch9]

CheckFled:
    Scratch1 = 0
    if CombatEnd > PartyDied { Scratch1 = 2 }
    return

FleeRandomly:
    Facing = Random(3)
    Scratch1 = Random(4)
    PartyX = FleeXTable[Scratch1]
    PartyY = FleeYTable[Scratch1]
    NoCoords = 1

RedrawExit:
    CallRedraw()
    exit

LoadRandomUndead:
    Scratch6 = Scratch7 * 4
    LoadMonster(Zombie, count=Scratch6, icon=Zombie)
    Scratch1 = Random(4)
    if Scratch1 < 2 {
        LoadMonster(JujuZombie, count=2, icon=Zombie)
    } else if Scratch1 > 2 {
        LoadMonster(Ghoul, count=2, icon=Ghoul)
    } else if Scratch1 == 2 {
        LoadMonster(Mummy, count=1, icon=Mummy)
    }
    goto FightGroup

def WaitReturnPressed() {
    AuxMonster = Menu("PRESS <RETURN> OR BUTTON TO CONTINUE")
    return
}

def SkeletonsApproach() {
    SetupMonster(Skeleton, distance=2, closeup=Skeleton)
    goto ApproachTwice
}

def ZombiesApproach() {
    SetupMonster(Zombie, distance=2, closeup=Zombie)
    goto ApproachTwice
}

def WightsApproach() {
    SetupMonster(Troll, distance=2, closeup=Troll)
    goto ApproachTwice
}

ApproachTwice:
    Approach()
    Delay()
    Approach()
    return

def YesNoMenu() {
    Scratch1 = Menu("YES", "NO")
    return
}
