	Treasure 0 0 0 #2328 0 12 0 30
	Combat
	Save 255 VampireTreasureTaken
	Exit

GhoulPack:
	SetupMonster 72 2 72
	Approach
	Delay
	Approach
	PrintClear "A PACK OF HUNGRY GHOULS LOPES TOWARD YOU."
	Gosub PressReturn
	Random 2 Scratch1
	Add 1 Scratch1 Scratch1
	PartyStrength Scratch8
	Div Scratch8 Scratch1 Scratch1
	if Scratch1 == 0: Save 1 Scratch1
	ClearMonsters
	LoadMonster 72 Scratch1 72
	Combat
	Exit

MysticManHint:
	if Facing == 0 && MysticManState == 250: PrintClear "THE MYSTIC MAN SPEAKS, 'THE VAMPIRE IS BEHIND THE NORTHERN WALL.'"
	Exit

RandomEncounter:
	GetTable GroupDistTable Scratch9 MonsterDist
	GetTable GroupTable1 Scratch9 Scratch3
	GetTable GroupTable2 Scratch9 Scratch4
	GetTable GroupTable3 Scratch9 Scratch5
	if Scratch6 == 1: Return
	PartySurprise Scratch1 Scratch2
	Surprise Scratch1 Scratch2 0 0
	if Surprise != 1: Goto SurpriseCheck
	OnGoto Scratch9 3 SkeletonAttack ZombieAttack WightAttack

SurpriseCheck:
	if Surprise > 2: Goto PartyConfronted
	also if Surprise == 2: Goto PartySneaksUp
	also if Surprise < 2: Goto FindUndead

PartyConfronted:
	Save 0 MaxDist
	Save "YOU ARE SUDDENLY CONFRONTED BY " $AuxText
	Save $AuxMonsterName $AuxNameCopy7
	Goto EncounterChoice

PartySneaksUp:
	Save "YOU HAVE MANAGED TO SNEAK UP ON SOME " $AuxText
	Save $AuxMonsterName $AuxNameCopy8
	EncounterMenu Scratch5 MonsterDist Scratch5 Scratch1 1 0 2 3 4 $AuxText "" "" 0 0
	OnGoto Scratch1 4 Done StartFight FleeRandomly PartyNoticed

PartyNoticed:
	Save 0 Surprise
	PrintClear "YOU ARE NOTICED."
	Goto EncounterChoice

FindUndead:
	PrintClear "YOU FIND SOME UNDEAD."

EncounterChoice:
	EncounterMenu Scratch5 MonsterDist Scratch5 Scratch1 1 1 1 1 4 "" "" "" Scratch4 Scratch4
	OnGoto Scratch1 4 Done StartFight FleeRandomly ParlayFails

ParlayFails:
	Parlay 0 0 0 0 0 Scratch1
	PrintClear "THEY DON'T LISTEN."

StartFight:
	Save 0 Beaten
	if Scratch7 == 0: Save 1 Scratch7
	ClearMonsters
	LoadMonster Scratch3 Scratch7 Scratch5
	if Scratch3 == 20: Goto LoadRandomUndead

FightGroup:
	Combat
	Gosub CheckFled
	if Scratch1 == 2: Goto FleeRandomly
	OnGoto Scratch9 10 SkeletonsBeaten ZombiesBeaten WightGroupFight GiantSkeletonStep GiantZombieStep WightBossStep BoneTowerSeenSet FurTowerSeenSet SkeletonAmbushOver LightningTowerSeenSet

CheckFled:
	Save 0 Scratch1
	if CombatEnd > 128: Save 2 Scratch1
	Return

FleeRandomly:
	Random 3 Facing
	Random 4 Scratch1
	GetTable FleeXTable Scratch1 PartyX
	GetTable FleeYTable Scratch1 PartyY
	Save 1 NoCoords

RedrawExit:
	Call CallRedraw
	Exit

LoadRandomUndead:
	Mul Scratch7 4 Scratch6
	LoadMonster 35 Scratch6 35
	Random 4 Scratch1
	if Scratch1 < 2: LoadMonster 29 2 35
	also if Scratch1 > 2: LoadMonster 72 2 72
	also if Scratch1 == 2: LoadMonster 114 1 114
	Goto FightGroup

PressReturn:
	HMenu AuxMonster 1 "PRESS <RETURN> OR BUTTON TO CONTINUE"
	Return

SkeletonsApproach:
	SetupMonster 34 2 34
	Goto ApproachTwice

ZombiesApproach:
	SetupMonster 35 2 35
	Goto ApproachTwice

WightsApproach:
	SetupMonster 20 2 20

ApproachTwice:
	Approach
	Delay
	Approach
	Return

YesNoMenu:
	HMenu Scratch1 2 "YES" "NO"
	Return
