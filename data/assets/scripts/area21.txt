    YesNoMenu()
    goto [BoardBoat, StayAtKeep][Scratch1]
    goto BoardBoat

BoardBoat:
    if SkyColor21 == 8 { goto SailToPhlan }
    Picture(Ship)

    SailToPhlan:
    Text("YOU BOARD A BOAT.")

EnterDone:
    Scratch10 = ~Inside & Cell

on step {
    Sky = SkyColor21
    Scratch10 = ~Inside & Cell
    RedrawView()
    Text("")
    Picture(None)
    Scratch6 = 0
    AuxCounter = 0

    FindCellLoop:
    if AuxCounter == Scratch10 { goto CellDispatch }
    if AuxCounter > 19 { exit }
    AuxCounter += 1
    goto FindCellLoop

    CellDispatch:
    goto [ClearExit, DeadElf, KeepGates, PatrolRoad, Stable, FungusRoom, Smithy, Barracks, Insects,
        OrcRaid, Armory, Chapel, FerranSpirit, HiddenRoom, GearStands, SmithyHammer, BarracksSearch,
        MagicGear, CoordsOff, EmptyHallCheck][AuxCounter]
    exit
}

on camp {
    if PatrolsFought21 > 4 { goto NoRest }
    RestEvery = 2
    RestChance = 1
    exit
}

NoRest:
    RestEvery = 0
    RestChance = 0
    exit

on rest {
    RestChance = 0
    Scratch5 = 0
    Scratch3 = 0
    Scratch6 = 0
    goto UndeadPatrol
}

    Text("THE BOAT DISEMBARKS YOU AT SOKAL KEEP.")
    PressToContinue()

    goto NoRest
}

    Protection(ElfScrollLine3)
    Text("...THE LAST PART IS EATEN AWAY.")

PatrolRoad:

    Scratch3 = 0

    if PatrolsFought21 < 4 { goto PatrolRoll }

    Delay()
    Approach()
    Delay()
    Approach()

    Text("AN UNDEAD PATROL CONFRONTS YOU.  DO YOU GIVE THE PASSWORD?")
    YesNoMenu()
    goto [PatrolLeaves, PatrolFight][Scratch1]

    PatrolMenu:
    EncounterMenu(Skeleton, distance=0, closeup=Skeleton, result=Scratch1, combat=Attack, wait=CloseIn,
        flee=Attack, advance=Mirror, parlay=Talk, near="A PACK OF UNDEAD SUDDENLY APPEARS.",
        escape=0, strength=0)

PatrolParlay:

    NewLine()

        $AuxText = "SAMOSUD"

        $AuxText = "SHESTNI"

FungusRoom:

    Text("WITH THEIR POWERS THEY RULED AND UNITED ALL ELSE, DRIVING FORWARD TO DESTROY US ALL. THE SAGE MENDOR WORKED HARD TO GATHER RECORD OF ALL THESE THINGS, BUT THEY ARE LOST NOW, HIS LIBRARY OVERRUN.")

def FleeIfLost() {
    if CombatEnd > PartyDied { goto RunAway }
    return
}

RunAway:
    Facing = Random(3)
    AuxTemp9 = Random(3)
    PartyX = FleeXTable[AuxTemp9]
    PartyY = FleeYTable[AuxTemp9]
    Text("")
    CallRedraw()

ClearExit:
    Picture(None)
    exit

def PressToContinue() {
    AuxMenuChoice = Menu("PRESS <RETURN> OR BUTTON TO CONTINUE")
    return
}

def YesNoMenu() {
    Scratch1 = Menu("YES", "NO")
    return
}

def SetSky() {
    SkyColor21 = 11
    if Hours >= 14 { SkyColor21 = 8 }
    Sky = SkyColor21
    return
}

LoadEncounter:
    MonsterDist = EncMonsterDistTable[Scratch1]
    AuxRoll = EncMonsterTable[Scratch1]
    AuxMonster = EncMoraleTable[Scratch1]
    AuxEncResult = EncCountTable[Scratch1]
    AuxEncResult2 = EncPictureTable[Scratch1]
    if Scratch6 != 0 { goto LoadEncounterGroup }
    ClearMonsters()
    Scratch6 = 1

    LoadEncounterGroup:
    LoadMonster(AuxRoll, count=AuxEncResult, icon=AuxEncResult2)
    if AuxPartyStrength == 0 { goto EncounterCombat }
    Scratch1 += 1
    AuxPartyStrength -= 1
    goto LoadEncounter

    EncounterCombat:
    Morale = AuxMonster
    Combat()
    FleeIfLost()
    goto [FrogsBeaten, InsectsDone, FerranBeaten, ClearExit, PatrolBeaten, ClearExit, ClearExit, OrcRaidDone][Scratch1]
    exit # the original runs on into data at AEF6
