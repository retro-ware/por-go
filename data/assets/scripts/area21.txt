	Exit

PressToContinue:

	Return

YesNoMenu:

	Choice = Menu "YES" "NO"
	Return

SetSky:
	SkyColor21 = 11
	if Hours >= 14: SkyColor21 = 8
	Sky = SkyColor21
	Return

LoadEncounter:
	MonsterDist = EncMonsterDistTable[Scratch1]
	AuxRoll = EncMonsterTable[Scratch1]
	AuxMonster = EncMoraleTable[Scratch1]
	AuxEncResult = EncCountTable[Scratch1]
	AuxEncResult2 = EncPictureTable[Scratch1]
	if Scratch6 != 0: Goto LoadEncounterGroup
	ClearMonsters
	Scratch6 = 1

LoadEncounterGroup:
	LoadMonster AuxRoll count=AuxEncResult icon=AuxEncResult2
	if AuxPartyStrength == 0: Goto EncounterCombat
	Scratch1 += 1
	AuxPartyStrength -= 1
	Goto LoadEncounter

EncounterCombat:
	Morale = AuxMonster
	Combat
	Gosub FleeIfLost
	OnGoto Scratch1 FrogsBeaten InsectsDone FerranBeaten ClearExit PatrolBeaten ClearExit ClearExit OrcRaidDone
	Exit // extra: the original runs on into data at AEF6
